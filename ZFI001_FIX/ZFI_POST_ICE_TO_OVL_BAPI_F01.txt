FORM fm_post_document .
*----------------------------------------------------------------------*
* CHANGE 30.09.2026 - Fix for duplicate posting of ONGC documents
*   (e.g. ONGC doc 8226000223 posted twice as 8826004360 / 8826004365)
*   1. Each ONGC document is locked (ENQUEUE_E_TABLE on ZFIBKPF) while
*      it is processed, so two parallel runs cannot post it together.
*   2. ZFIBKPF is re-read from DB just before posting; if an OVL
*      document number (ZBELNR) is already there, it is not posted.
*   3. ZFIBKPF / ZFI_BSEG are updated per document BEFORE the commit
*      (earlier: once for all documents at the end of the run), so an
*      FI document can no longer be committed without its ZFIBKPF entry.
*   4. Z-table rows are built from the DB row, so ZFIBKPF / ZFI_BSEG
*      fields that do not exist in BKPF / BSEG are no longer cleared.
*----------------------------------------------------------------------*

  IF  gv_no IS INITIAL .
    gv_no = 1.
  ENDIF.
****  Data declaration .
  DATA:
    lv_index TYPE sy-tabix.

  DATA  : ls_head       TYPE          bapiache09,
          ls_gl         TYPE          bapiacgl09,
          ls_ven        TYPE          bapiacap09,
          ls_ctmr       TYPE          bapiacar09,
          ls_curr       TYPE          bapiaccr09,
          ls_return     TYPE          bapiret2,
          ls_cmt_ret    TYPE          bapiret2,
          ls_alv        TYPE          zfi_s_alv,
          lt_gl         TYPE TABLE OF bapiacgl09,
          lt_ven        TYPE TABLE OF bapiacap09,
          lt_curr       TYPE TABLE OF bapiaccr09,
          lt_ctmr       TYPE TABLE OF bapiacar09,
          lt_tax        TYPE TABLE OF bapiactx09,
          lt_bapiacwt09 TYPE TABLE OF bapiacwt09,
          lt_return     TYPE TABLE OF bapiret2.

  DATA : obj_type TYPE bapiache09-obj_type,
         obj_key  TYPE bapiache09-obj_key,
         obj_sys  TYPE bapiache09-obj_sys.

  DATA : ls_bkpf     TYPE bkpf,
         ls_bseg     TYPE bseg,
         ls_bseg_i   TYPE bseg.

  DATA : lt_bseg     TYPE TABLE OF zfi_bseg,
         lt_bkpf_new TYPE TABLE OF zfibkpf,
         ls_bkpf_new TYPE zfibkpf,
         ls_bkpf_db  TYPE zfibkpf.
  DATA : lt_extension TYPE bapiparex_tab_ac,
         ls_extension TYPE bapiparex.
  DATA : lv_varkey    TYPE rstable-varkey.
  FIELD-SYMBOLS : <fs_bseg> TYPE zfi_bseg.

  SORT   : gt_bkpf BY bukrs belnr gjahr,
           gt_bseg BY bukrs belnr gjahr.

  LOOP AT  gt_bkpf INTO ls_bkpf .

    REFRESH : lt_gl,
              lt_ven,
              lt_return,
              lt_curr,
              lt_bseg,
              lt_ctmr,
              lt_tax,
              lt_bapiacwt09,
              lt_extension,
              lt_bkpf_new.
    CLEAR : ls_head, ls_bkpf_db, ls_return, ls_cmt_ret, obj_key.

*** Step 1 ==> Lock the ONGC document for this run
    CLEAR lv_varkey.
    CONCATENATE sy-mandt ls_bkpf-bukrs ls_bkpf-belnr ls_bkpf-gjahr
           INTO lv_varkey RESPECTING BLANKS.

    CALL FUNCTION 'ENQUEUE_E_TABLE'
      EXPORTING
        mode_rstable   = 'E'
        tabname        = 'ZFIBKPF'
        varkey         = lv_varkey
        _scope         = '1'
      EXCEPTIONS
        foreign_lock   = 1
        system_failure = 2
        OTHERS         = 3.
    IF sy-subrc <> 0.
      ls_alv-index   = gv_no.
      ls_alv-icon    = '@0A@'.
      ls_alv-bukrs   = ls_bkpf-bukrs.
      ls_alv-belnr   = ls_bkpf-belnr.
      ls_alv-gjahr   = ls_bkpf-gjahr.
      ls_alv-blart   = ls_bkpf-blart.
      CONCATENATE 'Document is being processed by' sy-msgv1 '- not posted'
             INTO ls_alv-message SEPARATED BY space.
      ls_alv-number  = '000'.
      ls_alv-id      = 'ZFI'.
      ls_alv-type    = 'E'.
      ADD 1 TO gv_no.
      APPEND ls_alv TO git_alv.
      CLEAR ls_alv.
      CONTINUE.
    ENDIF.

*** Step 2 ==> Fresh duplicate check on DB (OVL document already created?)
    SELECT SINGLE * FROM zfibkpf BYPASSING BUFFER INTO @ls_bkpf_db
      WHERE bukrs = @ls_bkpf-bukrs
      AND   belnr = @ls_bkpf-belnr
      AND   gjahr = @ls_bkpf-gjahr.
    IF ls_bkpf_db-zbelnr IS NOT INITIAL.
      ls_alv-index     = gv_no.
      ls_alv-icon      = '@0A@'.
      ls_alv-bukrs     = ls_bkpf-bukrs.
      ls_alv-belnr     = ls_bkpf-belnr.
      ls_alv-gjahr     = ls_bkpf-gjahr.
      ls_alv-blart     = ls_bkpf-blart.
      ls_alv-bukrs_new = ls_bkpf_db-zbukrs.
      ls_alv-belnr_new = ls_bkpf_db-zbelnr.
      ls_alv-gjahr_new = ls_bkpf_db-zgjahr.
      ls_alv-message   = 'Document is already posted'(002).
      ls_alv-number    = '000'.
      ls_alv-id        = 'ZFI'.
      ls_alv-type      = 'E'.
      ADD 1 TO gv_no.
      APPEND ls_alv TO git_alv.
      CLEAR ls_alv.
      PERFORM fm_dequeue USING lv_varkey.
      CONTINUE.
    ENDIF.
    IF ls_bkpf_db IS INITIAL.
      MOVE-CORRESPONDING ls_bkpf TO ls_bkpf_db.
    ENDIF.

    ls_head-username     = sy-uname.
    ls_head-trans_date = ls_bkpf-wwert.
    ls_head-obj_type     = 'BKPFF'.
    ls_head-header_txt   = ls_bkpf-bktxt.
    ls_head-ref_doc_no   = ls_bkpf-xblnr.
    ls_head-pstng_date   = ls_bkpf-budat.
    ls_head-doc_date     = ls_bkpf-bldat.
    ls_head-comp_code    = ls_bkpf-bukrs.
    ls_head-fisc_year    = ls_bkpf-gjahr.
    ls_head-fis_period   = ls_bkpf-monat.
    ls_head-doc_type     = 'EZ'.
    ls_head-vatdate      = ls_bkpf-vatdate.

    CLEAR : ls_bseg_i.
    READ TABLE gt_bseg INTO ls_bseg_i WITH KEY bukrs = ls_bkpf-bukrs
                                               belnr = ls_bkpf-belnr
                                               gjahr = ls_bkpf-gjahr
                                               BINARY SEARCH.
    IF  sy-subrc = 0 .
      lv_index = sy-tabix.

      LOOP AT gt_bseg INTO ls_bseg FROM lv_index.
        IF  ls_bseg-bukrs <> ls_bseg_i-bukrs
          OR ls_bseg-belnr <> ls_bseg_i-belnr
          OR ls_bseg-gjahr <> ls_bseg_i-gjahr.
          EXIT.
        ENDIF.
        CLEAR : ls_gl,
                ls_curr,
                ls_ven,
                ls_ctmr.

        CASE ls_bseg-koart.
          WHEN 'S'.
            PERFORM fm_change_gl USING  ls_bseg
                                        ls_bkpf
                               CHANGING ls_gl
                                        ls_curr
                                        lt_curr.
            APPEND ls_gl TO lt_gl.
            ls_extension-valuepart1 = sy-repid.

            CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
              EXPORTING
                input  = ls_bseg-hkont
              IMPORTING
                output = ls_bseg-hkont.

            ls_extension-valuepart4 = ls_bseg-hkont.

            ls_extension-valuepart2 = 'PERNR'.
            IF ls_bseg-pernr IS NOT INITIAL.
              ls_extension-valuepart3 = ls_bseg-pernr .
            ELSE.
              ls_extension-valuepart3 = so_pernr-low.
            ENDIF.
            APPEND ls_extension TO lt_extension.
            CLEAR ls_extension.

          WHEN 'K' .
            PERFORM fm_change_vendor USING ls_bseg
                                           ls_bkpf
                                  CHANGING ls_ven
                                           ls_curr
                                           lt_curr.
            APPEND ls_ven TO lt_ven.

          WHEN 'D'.
            CLEAR : ls_ctmr.
            PERFORM fm_change_customer USING ls_bseg
                                             ls_bkpf
                                    CHANGING ls_ctmr
                                             ls_curr
                                             lt_curr.
            APPEND ls_ctmr TO lt_ctmr.

            "-----Added by Manisha bh.Dt:12.05.2017------*
          WHEN 'A'.
            PERFORM fm_update_asset USING ls_bseg
                                          ls_bkpf
                                 CHANGING ls_head
                                          ls_gl
                                          lt_gl
                                          ls_curr
                                          lt_curr.
        ENDCASE.
        CLEAR : ls_bseg.
      ENDLOOP.
    ENDIF.

*** ZFI_BSEG lines of this document as they are on DB (keeps all Z fields)
    SELECT * FROM zfi_bseg INTO TABLE @lt_bseg
      WHERE bukrs = @ls_bkpf-bukrs
      AND   belnr = @ls_bkpf-belnr
      AND   gjahr = @ls_bkpf-gjahr.

*** Step 3 ==> Simulate
    REFRESH : lt_return.
    " Code Remediation changes S4 2025_1_A Conversion **BEGIN OF CHANGE BY SAP_ABAP 05.06.2026  FOR ATC
    CALL FUNCTION 'BAPI_ACC_DOCUMENT_CHECK' "#EC CI_USAGE_OK[2628704]
      " Code Remediation changes S4 2025_1_A Conversion * *END OF CHANGE BY SAP_ABAP 05.06.2026 FOR ATC
      EXPORTING
        documentheader    = ls_head     "#EC CI_USAGE_OK[2438131]
      TABLES
        accountgl         = lt_gl
        accountreceivable = lt_ctmr
        accountpayable    = lt_ven
        accounttax        = lt_tax
        currencyamount    = lt_curr
        return            = lt_return
        extension2        = lt_extension
        accountwt         = lt_bapiacwt09.

    READ TABLE lt_return INTO ls_return WITH KEY type = 'E'.
    IF sy-subrc <> 0.
*** Step 4 ==> Post
      REFRESH : lt_return.
      " Code Remediation changes S4 2025_1_A Conversion **BEGIN OF CHANGE BY SAP_ABAP 05.06.2026  FOR ATC
      CALL FUNCTION 'BAPI_ACC_DOCUMENT_POST'  "#EC CI_USAGE_OK
          "#EC CI_USAGE_OK[2628704]
  "#EC CI_USAGE_OK[2438131]
        " Code Remediation changes S4 2025_1_A Conversion * *END OF CHANGE BY SAP_ABAP 05.06.2026 FOR ATC
        EXPORTING
          documentheader    = ls_head
        IMPORTING
          obj_type          = obj_type
          obj_key           = obj_key
          obj_sys           = obj_sys
        TABLES
          accountgl         = lt_gl
          accountreceivable = lt_ctmr
          accountpayable    = lt_ven
          accounttax        = lt_tax
          currencyamount    = lt_curr
          return            = lt_return
          extension2        = lt_extension
          accountwt         = lt_bapiacwt09.

      CLEAR ls_return.
      READ TABLE lt_return INTO ls_return WITH KEY type   = 'S'
                                                   id     = 'RW'
                                                   number = '605'.
      IF sy-subrc = 0.
*** Step 5 ==> Posted: write ZFIBKPF / ZFI_BSEG first, then commit both
        CLEAR : ls_bkpf_new.
        ls_bkpf_new         = ls_bkpf_db.
        ls_bkpf_new-zbelnr  = ls_return-message_v2+0(10).
        ls_bkpf_new-zbukrs  = ls_return-message_v2+10(4).
        ls_bkpf_new-zgjahr  = ls_return-message_v2+14(4).
        ls_bkpf_new-message = ls_return-message.
        ls_bkpf_new-type    = 'S'.
        APPEND ls_bkpf_new TO lt_bkpf_new.

        LOOP AT  lt_bseg ASSIGNING <fs_bseg>.
          <fs_bseg>-zbelnr  = ls_bkpf_new-zbelnr.
          <fs_bseg>-zbukrs  = ls_bkpf_new-zbukrs.
          <fs_bseg>-zgjahr  = ls_bkpf_new-zgjahr.
          <fs_bseg>-zbuzei  = <fs_bseg>-buzei.
          <fs_bseg>-message = ls_return-message.
        ENDLOOP.

        CALL FUNCTION 'ZFI_UPDATE_ACT'
          TABLES
            it_bkpf     = lt_bkpf_new
            it_bseg     = lt_bseg
          EXCEPTIONS
            update_fail = 1
            OTHERS      = 2.
        IF sy-subrc <> 0.
*         Z-table could not be updated -> do not post the FI document
          CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
          CLEAR ls_return.
          ls_return-type    = 'E'.
          ls_return-id      = 'ZFI'.
          ls_return-number  = '000'.
          ls_return-message = 'ZFIBKPF/ZFI_BSEG update failed - document not posted'.
          PERFORM fm_save_error USING ls_bkpf ls_bkpf_db ls_return
                             CHANGING lt_bseg.
        ELSE.
          CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
            EXPORTING
              wait   = 'X'
            IMPORTING
              return = ls_cmt_ret.
          IF ls_cmt_ret-type CA 'EA'.
*           FI update terminated -> FI document does not exist, reset flag
            CLEAR ls_return.
            ls_return-type    = 'E'.
            ls_return-id      = 'ZFI'.
            ls_return-number  = '000'.
            ls_return-message = 'FI update terminated - check SM13, document not posted'.
            PERFORM fm_save_error USING ls_bkpf ls_bkpf_db ls_return
                               CHANGING lt_bseg.
          ELSE.
            ls_alv-icon      = '@08@'.   "FOR GREEN COLOR
            ls_alv-index     = gv_no.
            MOVE-CORRESPONDING ls_return TO ls_alv.
            ls_alv-bukrs     = ls_bkpf-bukrs.
            ls_alv-belnr     = ls_bkpf-belnr.
            ls_alv-gjahr     = ls_bkpf-gjahr.
            ls_alv-blart     = ls_bkpf-blart.
            ls_alv-bukrs_new = ls_bkpf_new-zbukrs.
            ls_alv-belnr_new = ls_bkpf_new-zbelnr.
            ls_alv-gjahr_new = ls_bkpf_new-zgjahr.
            ADD 1 TO gv_no.
            APPEND ls_alv TO git_alv.
            CLEAR ls_alv.
          ENDIF.
        ENDIF.

      ELSE.
*** Posting failed
        CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
        LOOP AT lt_return INTO ls_return WHERE type = 'E'
                                          AND  number <> '609'.
          PERFORM fm_save_error USING ls_bkpf ls_bkpf_db ls_return
                             CHANGING lt_bseg.
          EXIT.
        ENDLOOP.
      ENDIF.

    ELSE.
*** Simulation failed
      LOOP AT lt_return INTO ls_return WHERE type = 'E'
                                        AND  number <> '609'
                                        AND message_v1 <> 'BKPFF'.
        PERFORM fm_save_error USING ls_bkpf ls_bkpf_db ls_return
                           CHANGING lt_bseg.
        EXIT.
      ENDLOOP.
    ENDIF.

    PERFORM fm_dequeue USING lv_varkey.
    CLEAR : ls_bkpf.
  ENDLOOP.

ENDFORM.
*&---------------------------------------------------------------------*
*&      Form  FM_SAVE_ERROR
*&---------------------------------------------------------------------*
*       Error log in ALV and error status in ZFIBKPF / ZFI_BSEG
*       (called while the document is locked and ZBELNR is initial)
*----------------------------------------------------------------------*
FORM fm_save_error USING    is_bkpf    TYPE bkpf
                            is_bkpf_db TYPE zfibkpf
                            is_return  TYPE bapiret2
                   CHANGING ct_bseg    TYPE tt_zfi_bseg.

  DATA : ls_alv      TYPE zfi_s_alv,
         ls_bkpf_new TYPE zfibkpf,
         lt_bkpf_new TYPE TABLE OF zfibkpf.
  FIELD-SYMBOLS : <fs_bseg> TYPE zfi_bseg.

  ls_alv-icon      = '@0A@'.    "FOR RED COLOR
  ls_alv-index     = gv_no.
  MOVE-CORRESPONDING is_return TO ls_alv.
  ls_alv-bukrs     = is_bkpf-bukrs.
  ls_alv-belnr     = is_bkpf-belnr.
  ls_alv-gjahr     = is_bkpf-gjahr.
  ls_alv-blart     = is_bkpf-blart.
  ADD 1 TO gv_no.
  APPEND ls_alv TO git_alv.

  ls_bkpf_new         = is_bkpf_db.
  CLEAR : ls_bkpf_new-zbelnr, ls_bkpf_new-zbukrs, ls_bkpf_new-zgjahr.
  ls_bkpf_new-message = is_return-message.
  ls_bkpf_new-type    = 'E'.
  APPEND ls_bkpf_new TO lt_bkpf_new.

  LOOP AT ct_bseg ASSIGNING <fs_bseg>.
    CLEAR : <fs_bseg>-zbelnr, <fs_bseg>-zbukrs, <fs_bseg>-zgjahr, <fs_bseg>-zbuzei.
    <fs_bseg>-message = is_return-message.
  ENDLOOP.

  CALL FUNCTION 'ZFI_UPDATE_ACT'
    TABLES
      it_bkpf     = lt_bkpf_new
      it_bseg     = ct_bseg
    EXCEPTIONS
      update_fail = 1
      OTHERS      = 2.
  IF sy-subrc = 0.
    COMMIT WORK AND WAIT.
  ELSE.
    ROLLBACK WORK.
  ENDIF.

ENDFORM.

    EXPORTING
      mode_rstable = 'E'
      tabname      = 'ZFIBKPF'
      varkey       = iv_varkey
      _scope       = '1'.

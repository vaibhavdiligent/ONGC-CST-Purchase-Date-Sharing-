*&---------------------------------------------------------------------*
*&      Form  F_SUB_UPDATE_TABLE
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*  -->  p1        text
*  <--  p2        text
*----------------------------------------------------------------------*
FORM f_sub_update_table .
  CONSTANTS:lc_ver(3) TYPE c VALUE '000'.
  LOOP AT ta_upload INTO wa_upload.
    MOVE-CORRESPONDING wa_upload TO wa_s911.
    wa_s911-vrsio = lc_ver."'000'.
    CONCATENATE wa_upload-spmon+0(4) wa_upload-spmon+5(2) INTO wa_s911-spmon.
    CONCATENATE wa_upload-spwoc+0(4) wa_upload-spwoc+5(2) INTO wa_s911-spwoc.
    APPEND wa_s911 TO ta_s911.
  ENDLOOP.
  IF ta_s911 IS NOT INITIAL.
    MODIFY s911 FROM TABLE ta_s911.
    IF sy-subrc EQ 0.
      MESSAGE 'Data updated Successfully'(002) TYPE 'S'.
    ENDIF.
  ENDIF.

ENDFORM.

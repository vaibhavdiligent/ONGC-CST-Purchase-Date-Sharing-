function ssf_krn_sign_by_as.
*"----------------------------------------------------------------------
*"*"Lokale Schnittstelle:
*"  IMPORTING
*"     VALUE(SSFMANDT) TYPE  SSFARGS-MANDT DEFAULT SY-MANDT
*"     VALUE(SSFAPPLICATION) LIKE  SSFARGS-APPLIC DEFAULT ' DIRCT'
*"     VALUE(SSFPROFILEID) LIKE  SSFARGS-PROFILEID DEFAULT ''
*"     VALUE(SSFPROFILE) LIKE  SSFARGS-PROFILE DEFAULT ''
*"     VALUE(SSFPROFILEPW) LIKE  SSFARGS-PROFILEPW DEFAULT ''
*"     VALUE(SSFTOOLKIT) LIKE  SSFPARMS-SSFTOOLKIT DEFAULT ' '
*"     VALUE(STR_FORMAT) LIKE  SSFPARMS-SSFFORMAT DEFAULT 'PKCS7'
*"     VALUE(B_INC_CERTS) LIKE  SSFPARMS-BINCCERTS DEFAULT ' '
*"     VALUE(B_DETACHED) LIKE  SSFPARMS-BDETACHED DEFAULT ' '
*"     VALUE(B_INENC) LIKE  SSFPARMS-BINENC DEFAULT 'X'
*"     VALUE(IO_SPEC) LIKE  SSFPARMS-IOSPEC DEFAULT 'T'
*"     VALUE(OSTR_INPUT_DATA_L) LIKE  SSFPARMS-INDATALEN
*"     VALUE(STR_HASHALG) LIKE  SSFPARMS-SSFHASHALG OPTIONAL
*"  EXPORTING
*"     VALUE(OSTR_SIGNED_DATA_L) LIKE  SSFPARMS-SIGDATALEN
*"     VALUE(CRC) LIKE  SSFPARMS-SSFCRC
*"     VALUE(SRRC) LIKE  SSFINFO-RESULT
*"     VALUE(STR_SIGNERID) LIKE  SSFINFO-ID
*"  TABLES
*"      OSTR_INPUT_DATA
*"      OSTR_SIGNED_DATA STRUCTURE  SSFBIN
*"  EXCEPTIONS
*"      SSF_KRN_ERROR
*"      SSF_KRN_NOOP
*"      SSF_KRN_NOMEMORY
*"      SSF_KRN_OPINV
*"      SSF_KRN_NOSSFLIB
*"      SSF_KRN_INPUT_DATA_ERROR
*"      SSF_KRN_INVALID_PAR
*"      SSF_KRN_INVALID_PARLEN
*"      SSF_FB_INPUT_PARAMETER_ERROR
*"----------------------------------------------------------------------

  data: idlin_length type i,
        idlin_offset type i,
        offpos type i,
        idl(ssfno_length) type n,
        nsl(ssfno_length) type n,
        profl(ssfno_length) type n,
        pwl(ssfno_length) type n,
        nresult(ssfno_length) type n,
        id like ssfinfo-id,
        ns like ssfinfo-namespace,
        prof like ssfinfo-profile,
        pw like ssfinfo-password,
        str_format_l type i,
        str_hashalg_l type i.

  data: sigtab like ssfidlin occurs 0 with header line.

* check the boolean parameters
  check_boolpara b_inc_certs.
  check_boolpara b_detached.
  check_boolpara b_inenc.
* check the io_spec parameter
  check_ssf_iospec io_spec.

* get Application's SSF parameters
  if ssfapplication <> '!DIRCT'.
    call function 'SSF_GET_PARAMETER'
         exporting
              mandt                   = ssfmandt
              application             = ssfapplication
         importing
              ssftoolkit              = ssftoolkit
              str_format              = str_format
              str_profileid           = id
              str_profile             = prof
              str_profilepw           = pw
              str_hashalg             = str_hashalg
         exceptions
              ssf_parameter_not_found = 0
              others                  = 0.
  elseif SSFPROFILE is initial.
    perform get_as_profile changing id ns prof pw. " get AS' profile
  else.
    id   = SSFPROFILEID.
    ns   = SSF_CONSTNS-LOCALDIR.
    prof = SSFPROFILE.
    pw   = SSFPROFILEPW.
  endif.

* initialize length variables
  describe field sigtab-secinfos length idlin_length IN CHARACTER MODE.
  idlin_offset = idlin_length - 20.

* set length of SSF format parameter
  str_format_l = strlen( str_format ).
  str_hashalg_l = strlen( str_hashalg ).

* fill sigtab with signer data for application server
  refresh sigtab.
  clear sigtab.

* get AS' profile
  idl = strlen( id ).
  nsl = strlen( ns ).
  profl = strlen( prof ).
  pwl = strlen( pw ).
  str_signerid = id.                   " export parameter

* fill sigtab
  nresult = ssf_srresults-result_undefined.
  move nresult(ssfno_length) to sigtab+idlin_offset.

  offpos = 0.
  move idl(ssfno_length) to sigtab(ssfno_length).
  offpos = offpos + ssfno_length.
  move space to sigtab+offpos(1).
  offpos = offpos + 1.
  if idl > 0.
    move id(idl) to sigtab+offpos(idl).
    offpos = offpos + idl.
    move space to sigtab+offpos(1).
    offpos = offpos + 1.
  endif.
  move nsl(ssfno_length) to sigtab+offpos(ssfno_length).
  offpos = offpos + ssfno_length.
  move space to sigtab+offpos(1).
  offpos = offpos + 1.
  if nsl > 0.
    move ns(nsl) to sigtab+offpos(nsl).
    offpos = offpos + nsl.
    move space to sigtab+offpos(1).
    offpos = offpos + 1.
  endif.
  move profl(ssfno_length) to sigtab+offpos(ssfno_length).
  offpos = offpos + ssfno_length.
  move space to sigtab+offpos(1).
  offpos = offpos + 1.
  if profl > 0.
    move prof(profl) to sigtab+offpos(profl).
    offpos = offpos + profl.
    move space to sigtab+offpos(1).
    offpos = offpos + 1.
  endif.
  move pwl(ssfno_length) to sigtab+offpos(ssfno_length).
  offpos = offpos + ssfno_length.
  move space to sigtab+offpos(1).
  offpos = offpos + 1.
  if pwl > 0.
    move pw(pwl) to sigtab+offpos(pwl).
    offpos = offpos + pwl.
    move space to sigtab+offpos(1).
  endif.

  append sigtab.
  clear: pw.

* call C function 'SSFSIGN'
  call 'SSF_ABAP_SERVICE'                                  "#EC CI_CCALL
       id 'OPCODE'          field   ssf_opcodes-sign
       id 'SECTOOLKIT'      field   ssftoolkit
       id 'STRFORMAT'       field   str_format
       id 'STRFORMATL'      field   str_format_l
       id 'BINCCERTS'       field   b_inc_certs
       id 'BDETACHED'       field   b_detached
       id 'BINENC'          field   b_inenc
       id 'IOSPEC'          field   io_spec
       id 'OSTRINPUTDATAL'  field   ostr_input_data_l
       id 'OSTRSIGNEDDATAL' field   ostr_signed_data_l
       id 'CRC'             field   crc
       id 'OSTRINPUTDATA'   field   ostr_input_data-*sys*
       id 'SIGNERLIST'      field   sigtab-*sys*
       id 'OSTRSIGNEDDATA'  field   ostr_signed_data-*sys*
       id 'STRHASHALG'      field   str_hashalg
       id 'STRHASHALGL'     field   str_hashalg_l.

CHECK_KRN_ERRORS_SIGN 'SSF_KRN_SIGN_BY_AS'.

* transfer results from sigtab to signer_list table
* clear password information at the same time
  loop at sigtab.
    move sigtab+idlin_offset(ssfno_length) to nresult.
    srrc = nresult.
  endloop.

* LAUFD / LAUFI : this enhancement runs INSIDE the payment-medium LUW,
* BEFORE COMMIT WORK - so REGUT (written via update task) is not on the DB
* yet, and the SAPFPAYM globals PM_LAUFD/PM_LAUFI come through BLANK here.

* (ZFI_PAYMEDIUM_DMEE_20), which has it reliably from I_FPAYH, via
* EXPORT ... TO MEMORY ID 'ZBCM_LAUF'.

* LFDNR : the real REGUT-LFDNR is not available before COMMIT, so derive

* NOTE(UAT): this assumes files are signed in REGUT creation order, so the
* sequence equals REGUT-LFDNR (which ZFI_BNK_APP1/ZFI_BNK_APP use to link
* file <-> signature). Verify with a real multi-file run; if the order can
* differ, populate ZFI_PAYM_FILE from REGUT in a post-COMMIT step instead.

  WHERE laufd = lv_ilaufd AND laufi = lv_ilaufi.

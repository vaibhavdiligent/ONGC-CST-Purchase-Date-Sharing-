      upload_ole              = 2

  ENDIF.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  SUB_PROCESS_DATA
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*  -->  p1        text
*  <--  p2        text
*----------------------------------------------------------------------*
FORM sub_process_data .
  DATA : lv_count(5) TYPE n.
  LOOP AT t_upload INTO w_upload.
    lv_count = lv_count + 1.
    MOVE-CORRESPONDING w_upload TO w_s911.
    CONCATENATE p_indic lv_count INTO w_s911-spwoc.
    CONDENSE w_S911-spwoc.
    w_s911-mandt = sy-mandt.
    MODIFY s911 FROM w_s911.
    CLEAR w_s911.
    MESSAGE 'Data update successfully' TYPE 'S'.
  ENDLOOP.
ENDFORM.

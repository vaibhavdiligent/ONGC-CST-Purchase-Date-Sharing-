START-OF-SELECTION.

  SELECT * FROM
    zfi_bcm_payordr
    INTO TABLE lt_payordr
    WHERE pyord   IN s_pyord
      AND bukrs   IN s_bukrs
      AND laufd   IN s_laufd
      AND laufi   IN s_laufi
      AND zstatus IN s_statu
      AND rzawe   IN s_rzawe
      AND lifnr   IN s_lifnr
      AND kunnr   IN s_kunnr
      AND srtgb   IN s_srtgb
      and secco   IN s_secco.

  IF lt_payordr IS NOT INITIAL.
    SELECT * FROM reguh
      INTO TABLE lt_reguh
      FOR ALL ENTRIES IN lt_payordr
      WHERE laufd = lt_payordr-laufd
      AND   laufi = lt_payordr-laufi
      AND   xvorl = ' '
      AND   zbukr = lt_payordr-zbukr
      AND   lifnr = lt_payordr-lifnr
      AND   kunnr = lt_payordr-kunnr
      AND   pyord = lt_payordr-pyord.
*      AND   empfg = lt_payordr-empfg
*      AND   vblnr = lt_payordr-vblnr.

    SELECT lifnr name1
      FROM lfa1
      INTO TABLE lt_lfa1
      FOR ALL ENTRIES IN lt_payordr
      WHERE lifnr EQ lt_payordr-lifnr.

    SELECT kunnr name1
      FROM kna1
      INTO TABLE lt_kna1
      FOR ALL ENTRIES IN lt_payordr
      WHERE kunnr EQ lt_payordr-kunnr.

* Batch link (no BCM): the F110 payment run maps to the payment medium
* through REGUHM (LAUFD/LAUFI = F110 run) which carries BATCHNO and the
* medium run (LAUFD_M/LAUFI_M).
    SELECT * FROM reguhm
       INTO TABLE lt_reguhm
       FOR ALL ENTRIES IN lt_payordr
       WHERE laufd = lt_payordr-laufd
       AND   laufi = lt_payordr-laufi
       AND   xvorl = ' '
       AND   zbukr = lt_payordr-zbukr
       AND   lifnr = lt_payordr-lifnr
       AND   kunnr = lt_payordr-kunnr.

    IF lt_reguhm IS NOT INITIAL.
*     File send/received status from ZFI_PAYM_FILE (one row per medium
*     run, keyed by LAUFD/LAUFI = REGUHM-LAUFD_M/LAUFI_M). This is the
*     table the inbound SBI return-file interface updates (RECEIVED).
      SELECT * FROM zfi_paym_file
        INTO TABLE lt_paymf
        FOR ALL ENTRIES IN lt_reguhm
        WHERE laufd = lt_reguhm-laufd_m
        AND   laufi = lt_reguhm-laufi_m.
    ENDIF.
  ENDIF.

** Get Domain values
  CALL FUNCTION 'DDIF_DOMA_GET'
    EXPORTING
      name      = 'ZZSTATUS1'
      state     = 'A'
      langu     = 'E'
    TABLES
      dd07v_tab = lt_dd07v.
  IF sy-subrc <> 0.
* Implement suitable error handling here
  ENDIF.

  LOOP AT lt_payordr INTO ls_payordr.
    MOVE-CORRESPONDING ls_payordr TO ls_final.
** Get Domain Fixed values
    READ TABLE lt_dd07v INTO ls_dd07v WITH KEY domvalue_l = ls_final-zstatus.
    IF sy-subrc = 0.
      ls_final-desc = ls_dd07v-ddtext.
    ENDIF.
** REGUH Table data
    READ TABLE  lt_reguh INTO ls_reguh WITH KEY laufd = ls_payordr-laufd
                                                laufi = ls_payordr-laufi
                                                zbukr = ls_payordr-zbukr
                                                lifnr = ls_payordr-lifnr
                                                kunnr = ls_payordr-kunnr
                                                pyord = ls_payordr-pyord." BINARY SEARCH.
    IF sy-subrc = 0.
      MOVE ls_reguh-zbnks TO ls_final-zbnks.
      MOVE ls_reguh-zbnkn TO ls_final-zbnkn.
      MOVE ls_reguh-zbnky TO ls_final-zbnky.
    ENDIF.

    IF ls_payordr-lifnr IS NOT INITIAL.
      READ TABLE lt_lfa1 INTO ls_lfa1 WITH KEY lifnr = ls_payordr-lifnr.
      IF sy-subrc = 0.
        MOVE ls_lfa1-name1 TO ls_final-name1.
      ENDIF.
    ELSE.
      READ TABLE lt_kna1 INTO ls_kna1 WITH KEY kunnr = ls_payordr-kunnr.
      IF sy-subrc = 0.
        MOVE ls_kna1-name1 TO ls_final-name1.
      ENDIF.
    ENDIF.
** Batch data from REGUHM (batch no + medium run); send state from ZFI_PAYM_FILE
    CLEAR : ls_reguhm, ls_paymf.
    READ TABLE lt_reguhm INTO ls_reguhm WITH KEY laufd = ls_payordr-laufd
                                                 laufi = ls_payordr-laufi
                                                 zbukr = ls_payordr-zbukr
                                                 lifnr = ls_payordr-lifnr
                                                 kunnr = ls_payordr-kunnr.
    IF sy-subrc = 0.
      MOVE ls_reguhm-batchno TO ls_final-batch_no.
      MOVE ls_reguhm-laufd_m TO ls_final-laufd_f.
      MOVE ls_reguhm-laufi_m TO ls_final-laufi_f.
      READ TABLE lt_paymf INTO ls_paymf WITH KEY laufd = ls_reguhm-laufd_m
                                                 laufi = ls_reguhm-laufi_m.
    ENDIF.
** File status is PER PAYMENT, not per file: ZFI_PAYM_FILE-RECEIVED is a
** file-level flag (one row per medium run) and would wrongly show every
** payment of that file as "Received". A payment's bank response has
** actually come back only when its own ZSTATUS has moved off '001'
** (002 = success, 005 = rejected). Payments still '001' just show the
** send state (Sent / Not Sent).
    IF ls_payordr-zstatus = '002' OR ls_payordr-zstatus = '005'.
      ls_final-txt30 = 'Received'.
    ELSEIF ls_paymf-sent = 'X'.
      IF ls_paymf-sent_error = 'X'.
        ls_final-txt30 = 'Sent (Error)'.
      ELSE.
        ls_final-txt30 = 'Sent'.
      ENDIF.
    ELSE.
      ls_final-txt30 = 'Not Sent'.
    ENDIF.
    APPEND ls_final TO lt_final.
    CLEAR : ls_payordr, ls_final, ls_kna1, ls_lfa1.
  ENDLOOP.

END-OF-SELECTION.

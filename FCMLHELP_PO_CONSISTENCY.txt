*---------------------------------------------------------------------*
*       FORM DISTRIB                                                  *
*---------------------------------------------------------------------*
*       ........                                                      *
*---------------------------------------------------------------------*
*  -->  PT_EKBE                                                       *
*  -->  P_DELTA_ML                                                    *
*  -->  P_CURTP                                                       *
*  -->  P_COUNTS_DISTRIB_SETS_ML                                      *
*  -->  P_CORRECT_WEWRT                                               *
*---------------------------------------------------------------------*
FORM distrib TABLES pt_ekbe STRUCTURE ekbe
             USING  p_delta_ml LIKE mlbecr-arewr
                    p_curtp LIKE mlbecr-curtp
                    p_correct_wewrt LIKE boole-boole
             CHANGING ps_holz LIKE boole-boole
                      p_counts_distrib_sets_ml TYPE i.
*&---------------------------------------------------------------------*
*&      Form  DISTRIB
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*

  DATA: ft_ekbe_iv LIKE ekbe OCCURS 0 WITH HEADER LINE.
  DATA: f_arewr_shkzg  LIKE mlbecr-shkzg,
        f_arewr_help   LIKE mlbecr-arewr,
        f_arewr_delta  LIKE mlbecr-arewr,
        f_arewr_save   LIKE mlbecr-arewr,
        lh_stunr_space LIKE mlwere-stunr,
        lh_zaehk_space LIKE mlwere-zaehk.
  DATA: toleranz LIKE mlbecr-arewr VALUE 0.

*********************************************************************
* Initializations
*********************************************************************
  CLEAR: p_counts_distrib_sets_ml, ps_holz.
  IF p_delta_ml < 0.
    f_arewr_shkzg = 'H'.
  ELSE.
    f_arewr_shkzg = 'S'.
  ENDIF.
  f_arewr_save = abs( p_delta_ml ).
*********************************************************************
*  Prepare internal table ft_ekbe_iv with IV sets (or GR sets if no
*  invoices are available)
*********************************************************************
  IF p_correct_wewrt IS INITIAL.
    LOOP AT pt_ekbe WHERE vgabe = '2'
                    OR    vgabe = '3'.
      APPEND pt_ekbe TO ft_ekbe_iv.
    ENDLOOP.
  ELSE.
    LOOP AT pt_ekbe WHERE vgabe = '1'.
      APPEND pt_ekbe TO ft_ekbe_iv.
    ENDLOOP.
  ENDIF.

*********************************************************************
*  Order with PO, position, date and time (oldest first)
*********************************************************************
  SORT ft_ekbe_iv BY cpudt cputm.

*********************************************************************
*  Loop over ft_ekbe_iv and distribute f_arewr_delta to arewr.
*********************************************************************
  LOOP AT ft_ekbe_iv.
    SELECT * FROM  mlwere
             WHERE  ebeln = ft_ekbe_iv-ebeln
             AND    ebelp = ft_ekbe_iv-ebelp
             AND    stunr = h_stunr_space
             AND    zaehk = h_zaehk_space
             AND    zekkn = ft_ekbe_iv-zekkn.
      SELECT * FROM mlbecr
               WHERE mlvnr = mlwere-mlvnr
               AND   vgabe = ft_ekbe_iv-vgabe
               AND   gjahr = ft_ekbe_iv-gjahr
               AND   belnr = ft_ekbe_iv-belnr
               AND   buzei = ft_ekbe_iv-buzei
               AND   curtp = p_curtp.
        EXIT.
      ENDSELECT.
      IF sy-subrc = 0.
        EXIT.
      ENDIF.
    ENDSELECT.
    IF sy-subrc <> 0.
      EXIT.                            "Zur Sicherheit ganz raus.
    ENDIF.
    f_arewr_delta = 0.
***************************************************************
*     Calculate f_arewr_delta (the amount added to ft_ekbe_iv-arewr)
*     SHKZG has to be accounted for in the calculation
***************************************************************
    IF ( f_arewr_save > 0 ).
      p_counts_distrib_sets_ml = p_counts_distrib_sets_ml + 1.
      IF p_correct_wewrt IS INITIAL.
        IF  ( mlbecr-arewr < toleranz ) AND
            ( f_arewr_shkzg <> ft_ekbe_iv-shkzg ).
          " do nothing
        ELSE.
          IF ( f_arewr_shkzg = ft_ekbe_iv-shkzg ).
            f_arewr_delta = f_arewr_save.     " add f_arewr_delta
            f_arewr_save  = 0.
          ELSE.
            f_arewr_help = mlbecr-arewr - toleranz.
            IF (  f_arewr_save  > f_arewr_help ).
              f_arewr_delta = 0 - f_arewr_help.
              f_arewr_save  = f_arewr_save - abs( f_arewr_delta ).
            ELSE.
              f_arewr_delta =  - f_arewr_save.
              f_arewr_save  = 0.
            ENDIF.
          ENDIF.
        ENDIF.
      ELSE.
        IF  ( mlbecr-wrbtr < toleranz ) AND
            ( f_arewr_shkzg <> ft_ekbe_iv-shkzg ).
          " do nothing
        ELSE.
          IF ( f_arewr_shkzg = ft_ekbe_iv-shkzg ).
            f_arewr_delta =  f_arewr_save.     " add f_arewr_delta
            f_arewr_save  = 0.
          ELSE.
            f_arewr_help = mlbecr-wrbtr - toleranz.
            IF (  f_arewr_save  > f_arewr_help ).
              f_arewr_delta = 0 - f_arewr_help.
              f_arewr_save  = f_arewr_save - abs( f_arewr_delta ).
            ELSE.
              f_arewr_delta = 0 - f_arewr_save.
              f_arewr_save  = 0.
            ENDIF.
          ENDIF.
        ENDIF.
      ENDIF.
    ENDIF.
*********************************************************************
*     Update mlbecr with corrected arewr/wrbtr
*********************************************************************
    IF p_test IS INITIAL.
      IF f_arewr_delta <> 0.
        CLEAR: mlbecr_old, mlbecr_new.
        MOVE-CORRESPONDING mlbecr TO mlbecr_old.
        MOVE-CORRESPONDING mlbecr TO mlbecr_new.
        IF p_correct_wewrt IS INITIAL.
          UPDATE mlbecr SET arewr = arewr + f_arewr_delta
                        WHERE mlvnr = mlwere-mlvnr
                        AND   vgabe = ft_ekbe_iv-vgabe
                        AND   gjahr = ft_ekbe_iv-gjahr
                        AND   belnr = ft_ekbe_iv-belnr
                        AND   buzei = ft_ekbe_iv-buzei
                        AND   curtp = p_curtp.
          mlbecr_new-arewr = mlbecr_new-arewr + f_arewr_delta.
        ELSE.
          UPDATE mlbecr SET wrbtr = wrbtr + f_arewr_delta
                        WHERE mlvnr = mlwere-mlvnr
                        AND   vgabe = ft_ekbe_iv-vgabe
                        AND   gjahr = ft_ekbe_iv-gjahr
                        AND   belnr = ft_ekbe_iv-belnr
                        AND   buzei = ft_ekbe_iv-buzei
                        AND   curtp = p_curtp.
          mlbecr_new-wrbtr = mlbecr_new-wrbtr + f_arewr_delta.
        ENDIF.
        CALL FUNCTION 'FCMLHELP_CORRECTION_PROTOCOL'
          EXPORTING
            if_mlbecr_before = mlbecr_old
            if_mlbecr_after  = mlbecr_new.

      ENDIF.                           " f_arewr_delta
    ENDIF.                             " p_test is initial
**********************************************************************
    IF f_arewr_save = 0.
      EXIT.
    ENDIF.

  ENDLOOP.
  IF f_arewr_save <> 0.
    IF p_fbk1 IS INITIAL.
      FORMAT COLOR 6.
      WRITE: / 'Value cannot be distributed completely (not enough sets)'.
      HIDE ausgz.
      FORMAT RESET.
    ENDIF.
    ps_holz = 'X'.
  ENDIF.

ENDFORM.                               " DISTRIB

               AND   belnr = ft_ekbz_iv-belnr
               AND   buzei = ft_ekbz_iv-buzei
               AND   curtp = p_curtp.
        EXIT.
      ENDSELECT.
      IF sy-subrc = 0.
        EXIT.
      ENDIF.
    ENDSELECT.
    IF sy-subrc <> 0.
      EXIT.
    ENDIF.
    f_frawe_delta = 0.
***************************************************************
*     Calculate f_frawe_delta (the amount added to ft_ekbz_iv-arewr)
*     SHKZG has to be accounted for in the calculation
***************************************************************
    IF ( f_frawe_save > 0 ).

        IF  ( mlbecr-arewr < toleranz ) AND
            ( f_frawe_shkzg <> ft_ekbz_iv-shkzg ).
          " do nothing
        ELSE.
          IF ( f_frawe_shkzg = ft_ekbz_iv-shkzg ).
            f_frawe_delta = f_frawe_save.     " add f_frawe_delta
            f_frawe_save  = 0.
          ELSE.
            f_frawe_help = mlbecr-arewr - toleranz.
            IF (  f_frawe_save  > f_frawe_help ).
              f_frawe_delta = 0 - f_frawe_help.
              f_frawe_save  = f_frawe_save - abs( f_frawe_delta ).
            ELSE.
              f_frawe_delta = 0 - f_frawe_save.
              f_frawe_save  = 0.

          ENDIF.
        ENDIF.
      ELSE.
        IF  ( mlbecr-wrbtr < toleranz ) AND
            ( f_frawe_shkzg <> ft_ekbz_iv-shkzg ).
          " do nothing
        ELSE.

      IF f_frawe_delta <> 0.
        CLEAR: mlbecr_old, mlbecr_new.
        MOVE-CORRESPONDING mlbecr TO mlbecr_old.

*---------------------------------------------------------------------*
*       FORM DO_THE_CHECK                                             *
*---------------------------------------------------------------------*
*       ........                                                      *
*---------------------------------------------------------------------*
FORM do_the_check USING p_lfbnr LIKE ekbe-lfbnr
                        p_lfpos LIKE ekbe-lfpos
                        p_lfgja LIKE ekbe-lfgja.
*&---------------------------------------------------------------------*
*&      Form  DO_THE_CHECK
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
************************************************************************
****  test for critical cases
****  correct further currencies ***************************************
  DATA: h_wemng                    LIKE ebefu-wemng,
        h_were_menge               LIKE ebefu-wemng,
        h_wewrt                    LIKE ebefu-wewrt,
        h_were_wert                LIKE ebefu-wewrt,
        h_remng                    LIKE ebefu-remng,
        h_rewrt                    LIKE ebefu-rewrt,
        h_arewr                    LIKE ebefu-arewr,
        h_frwmg                    LIKE ekbnk-frwmg,
        h_frwwr                    LIKE ekbnk-frwwr,
        h_frrmg                    LIKE ekbnk-frrmg,
        h_frrwr                    LIKE ekbnk-frrwr,
        h_frawe                    LIKE ekbnk-frawe,
        h_delta_ml                 LIKE ebefu-arewr,
        h_delta_ml_1               LIKE ebefu-arewr,
        h_delta_ml_2               LIKE ebefu-arewr,
        h_wewrt_delta              LIKE ebefu_cr-wewrt,
        h_arewr_delta              LIKE ebefu_cr-arewr,
        h_rewrt_delta              LIKE ebefu_cr-rewrt,
        h_wewrt_delta_1            LIKE ebefu_cr-wewrt,
        h_arewr_delta_1            LIKE ebefu_cr-arewr,
        h_rewrt_delta_1            LIKE ebefu_cr-rewrt,
        h_wewrt_delta_2            LIKE ebefu_cr-wewrt,
        h_arewr_delta_2            LIKE ebefu_cr-arewr,
        h_rewrt_delta_2            LIKE ebefu_cr-rewrt,
        h_waers_ml_1               LIKE ebefu-waers,
        h_waers_ml_2               LIKE ebefu-waers,
        h_count                    TYPE i,
        s_invoice_found            LIKE boole-boole,
        s_correct_wewrt            LIKE boole-boole,
        s_correct_frwwr            LIKE boole-boole,
        s_holz                     LIKE boole-boole,
        s_error                    LIKE boole-boole,
        h_counts_distrib_sets_ml   TYPE i,
        h_counts_distrib_sets_ml_1 TYPE i,
        h_counts_distrib_sets_ml_2 TYPE i.

  CLEAR: h_count, s_correct_wewrt,
         h_wemng, h_wewrt, h_rewrt, h_remng, h_arewr,
         h_delta_ml, h_delta_ml_1, h_delta_ml_2,
         h_wewrt_delta, h_arewr_delta, h_wewrt_delta,
         h_wewrt_delta_1, h_arewr_delta_1, h_wewrt_delta_1,
         h_wewrt_delta_2, h_arewr_delta_2, h_wewrt_delta_2.

  READ TABLE t_ebefu WITH KEY ebeln = t_ekpo-ebeln
                              ebelp = t_ekpo-ebelp
                              lfbnr = p_lfbnr
                              lfpos = p_lfpos
                              lfbja = p_lfgja.
  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  SORT:
*    ct_ekbe_cr,                                             " 2261450/3443002
    t_ekbe_help BY ebeln ebelp lfgja lfbnr lfpos vgabe.

  h_remng = t_ebefu-remng.
  h_wemng = t_ebefu-wemng + t_ebefu-wesbb.                " 3590725-a / 3443002-a

  IF ( t_ekpo-wepos IS INITIAL ) AND ( h_wemng <> 0 ).
    WRITE: 'Inconsistent EKPO data: WEPOS = " ", WEMNG = ',
            h_wemng, ': '.
    HIDE ausgz.
    WRITE: 'Purchase order', t_ekpo-ebeln, '/', t_ekpo-ebelp.
    HIDE ausgz.
    erring_ekpo = 'X'.
    EXIT.
  ENDIF.

  h_were_menge = h_wemng - h_remng.                         " 3590725-a / 3443002-a


* Check the ML currencies (MLBECR)
  LOOP AT t_ebefu_cr WHERE ebeln = t_ekpo-ebeln
                     AND   ebelp = t_ekpo-ebelp
                     AND   curtp <> '00'
                     AND   curtp <> '10'.
    h_count = h_count + 1.
    h_rewrt = t_ebefu_cr-rewrt.
    h_wewrt = t_ebefu_cr-wewrt.
    IF ( t_ekpo-wepos IS INITIAL OR NOT t_ekpo-weunb IS INITIAL )
       AND ( h_wewrt <> 0 ).
      WRITE: 'Inconsistent EKPO data: WEPOS = "', t_ekpo-wepos,
             '", WEUNB = "', t_ekpo-weunb, '", WEWRT = "', h_wewrt,
             ': '.
      HIDE ausgz.
      WRITE:   'Purchase order', t_ekpo-ebeln, '/', t_ekpo-ebelp.
      HIDE ausgz.
      erring_ekpo = 'X'.
      EXIT.
    ENDIF.

    h_arewr = t_ebefu_cr-arewr.
    h_were_wert = t_ebefu_cr-wewrt - t_ebefu_cr-arewr.
*   Case 3 (no GR expected) *******************************************
    IF ( t_ekpo-wepos IS INITIAL OR NOT t_ekpo-weunb IS INITIAL )
       AND h_arewr <> 0.
*     We correct AREWR!
      erring_ekpo = 'X'.                                  "PUE11122000
      h_delta_ml = 0 - h_arewr.
      WRITE: / 'Case 3 (No GR expected or GR not valuated):'.
      HIDE ausgz.
      WRITE:   'Purchase order', t_ekpo-ebeln, '/', t_ekpo-ebelp.
      HIDE ausgz.
      WRITE: / 'AREWR not zero:', h_delta_ml CURRENCY t_ebefu_cr-waers,
               t_ebefu_cr-waers.
      HIDE ausgz.
      ULINE.
      HIDE ausgz.
      PERFORM clear_arewr TABLES t_ekbe_help
                          USING  t_ebefu_cr-curtp.
*   Case 1 (Quantity zero, values not) ********************************
    ELSEIF h_were_menge = 0 AND h_were_wert <> 0.
      erring_ekpo = 'X'.                                   "PUE11122000
      IF h_remng = 0 AND h_rewrt = 0.
*       Was there any invoice?
        READ TABLE t_ekbe_help WITH KEY ebeln = t_ekpo-ebeln
                                        ebelp = t_ekpo-ebelp
                                        lfgja = p_lfgja
                                        lfbnr = p_lfbnr
                                        lfpos = p_lfpos
                                        vgabe = '2'
                                        BINARY SEARCH
                                        TRANSPORTING NO FIELDS.
        IF sy-subrc <> 0.
*         No invoice! We have to correct WEWRT!
          s_correct_wewrt = 'X'.
          h_delta_ml = 0 - t_ebefu_cr-wewrt.
        ENDIF.
      ENDIF.
      IF s_correct_wewrt IS INITIAL.
*       We correct AREWR!
        h_delta_ml = h_were_wert.
      ENDIF.
      IF h_delta_ml <> 0.
*       Attention! If new arewr value is greater then value in EKBE
*       and sign is different, do no update! Value has to be spread
*       on several invoice entries (activate distribution switch)!
        PERFORM distrib TABLES t_ekbe_help
                        USING h_delta_ml
                              t_ebefu_cr-curtp
                              s_correct_wewrt
                        CHANGING s_holz
                                 h_counts_distrib_sets_ml.
        IF NOT s_holz IS INITIAL.
          IF p_fbk1 = 'X'.
            " fallback1: distribute to VGABE=1 records instead (3443002-d)
            s_correct_wewrt = 'X'.

            FORMAT COLOR 5.
            WRITE: / 'Using fallback strategy to distribute difference to GR.'.
            HIDE ausgz.
            FORMAT RESET.

            PERFORM distrib TABLES t_ekbe_help
                            USING h_delta_ml
                                  t_ebefu_cr-curtp
                                  s_correct_wewrt           " for vgabe=1
                            CHANGING s_holz
                                     h_counts_distrib_sets_ml.
          ENDIF.
        ENDIF.
*       Remember values for output (max. of two ML currencies)
        IF h_count = '1'.
          h_counts_distrib_sets_ml_1 = h_counts_distrib_sets_ml.
          h_delta_ml_1 = h_delta_ml.
          h_waers_ml_1 = t_ebefu_cr-waers.
        ELSE.
          h_counts_distrib_sets_ml_2 = h_counts_distrib_sets_ml.

FORM get_ebefu_cr TABLES   it_ebefu_cr STRUCTURE t_ebefu_cr
                           it_ekbe STRUCTURE t_ekbe
                  USING    i_ebeln LIKE ekpo-ebeln
                           i_ebelp LIKE ekpo-ebelp
                  CHANGING o_error TYPE c.
*
* Aufbauen t_ebefu_cr aus ML-Tabellen, nur weitere Währungen
*
  DATA: lt_ekbe_cr     LIKE cki_ekbecr OCCURS 0 WITH HEADER LINE,
        lt_ekbz_cr     LIKE cki_ekbzcr OCCURS 0 WITH HEADER LINE,
        ls_big_trouble LIKE boole-boole.
  DATA: BEGIN OF lt_curtp OCCURS 0,
          curtp LIKE cki_ekbecr-curtp,
          waers LIKE cki_ekbecr-waers,
        END OF lt_curtp.
  DATA: old_curtp LIKE cki_ekbecr-curtp.
  DATA: l_wewrt LIKE ebefu-wewrt.
  DATA: l_rewrt LIKE ebefu-rewrt.
  DATA: l_arewr LIKE ebefu-arewr.

  CLEAR: o_error.
  CALL FUNCTION 'CKMW_READ_PO_HISTORY'
    EXPORTING
      ebeln          = i_ebeln
      ebelp          = i_ebelp
      bnk            = ' '  "Keine Bezugsnebenkosten
    TABLES
      t_ekbe_cr      = lt_ekbe_cr
      t_ekbz_cr      = lt_ekbz_cr
    EXCEPTIONS
      data_not_found = 1
      OTHERS         = 3.
  IF sy-subrc NE 0.
    FORMAT COLOR 5.
    WRITE: / 'Error reading ml tables: ', i_ebeln, i_ebelp.
    HIDE ausgz.
    FORMAT RESET.
    o_error = 'X'.
    EXIT.
  ENDIF.
  REFRESH: it_ebefu_cr.
* Parallelwährungen bestimmen
  REFRESH lt_curtp.
  SORT lt_ekbe_cr BY curtp.
  CLEAR: old_curtp.
  LOOP AT lt_ekbe_cr.
    IF lt_ekbe_cr-curtp NE '00' AND
       lt_ekbe_cr-curtp NE '10'.
      IF lt_ekbe_cr-curtp NE old_curtp.
        old_curtp = lt_ekbe_cr-curtp.
        lt_curtp-curtp = lt_ekbe_cr-curtp.
        lt_curtp-waers = lt_ekbe_cr-waers.
        APPEND lt_curtp.
      ENDIF.
    ENDIF.
  ENDLOOP.
* Wert pro Währung aufsummieren
  LOOP AT lt_curtp.
    CLEAR: l_wewrt, l_rewrt, l_arewr.
    LOOP AT it_ekbe WHERE ebeln = i_ebeln
                      AND ebelp = i_ebelp.
      CHECK it_ekbe-vgabe = '1' OR
            it_ekbe-vgabe = '2' OR
            it_ekbe-vgabe = '3'.
      READ TABLE lt_ekbe_cr
       WITH KEY ebeln = it_ekbe-ebeln
                ebelp = it_ekbe-ebelp
                vgabe = it_ekbe-vgabe
                zekkn = it_ekbe-zekkn
                gjahr = it_ekbe-gjahr
                belnr = it_ekbe-belnr
                buzei = it_ekbe-buzei
                curtp = lt_curtp-curtp.
      IF sy-subrc <> 0 AND
         NOT ( it_ekbe-dmbtr = 0 AND it_ekbe-wrbtr = 0 ).
        FORMAT COLOR 5.
        WRITE: / 'BIG TROUBLE WITH EBEFU_CR:', i_ebeln, i_ebelp.
        HIDE ausgz.
        FORMAT RESET.
        ls_big_trouble = 'X'.
        EXIT.
      ENDIF.
      CHECK sy-subrc = 0.
      IF it_ekbe-vgabe EQ '1'.
        IF it_ekbe-shkzg EQ 'S'.
          ADD lt_ekbe_cr-wrbtr TO l_wewrt.
        ELSE.
          SUBTRACT lt_ekbe_cr-wrbtr FROM l_wewrt.
        ENDIF.
      ELSEIF it_ekbe-vgabe EQ '2' OR
             it_ekbe-vgabe EQ '3'.
        IF it_ekbe-shkzg EQ 'S'.
          ADD lt_ekbe_cr-arewr TO l_arewr.
          ADD lt_ekbe_cr-wrbtr TO l_rewrt.
        ELSE.
          SUBTRACT lt_ekbe_cr-arewr FROM l_arewr.
          SUBTRACT lt_ekbe_cr-wrbtr FROM l_rewrt.
        ENDIF.
      ENDIF.
    ENDLOOP.
    IF NOT ls_big_trouble IS INITIAL.
      REFRESH: it_ebefu_cr.
      EXIT.
    ENDIF.
    CLEAR it_ebefu_cr.
    MOVE-CORRESPONDING it_ekbe TO it_ebefu_cr.
    it_ebefu_cr-curtp = lt_curtp-curtp.
    it_ebefu_cr-waers = lt_curtp-waers.
    it_ebefu_cr-wewrt = l_wewrt.
    it_ebefu_cr-rewrt = l_rewrt.
    it_ebefu_cr-arewr = l_arewr.
    APPEND it_ebefu_cr.
  ENDLOOP.

ENDFORM.                               " GET_EBEFU_CR
*&---------------------------------------------------------------------*
*&      Form  GET_EKBNK_CR
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*----------------------------------------------------------------------*
FORM get_ekbnk_cr TABLES   it_ekbnk_cr STRUCTURE t_ekbnk_cr
                           it_ekbz STRUCTURE ekbz
                  USING    i_ebeln LIKE ekpo-ebeln
                           i_ebelp LIKE ekpo-ebelp
                  CHANGING o_error TYPE c.
*
*   Aufbauen t_ekbnk_cr aus ML-Tabellen, nur weitere Währungen
*
  DATA: lt_ekbe_cr     LIKE cki_ekbecr OCCURS 0 WITH HEADER LINE,
        lt_ekbz_cr     LIKE cki_ekbzcr OCCURS 0 WITH HEADER LINE,
        ls_big_trouble LIKE boole-boole.

  CLEAR: o_error.
  CALL FUNCTION 'CKMW_READ_PO_HISTORY'
    EXPORTING
      ebeln          = i_ebeln
      ebelp          = i_ebelp
      bnk            = 'X'  "Bezugsnebenkosten
    TABLES
      t_ekbe_cr      = lt_ekbe_cr
      t_ekbz_cr      = lt_ekbz_cr
    EXCEPTIONS
      data_not_found = 1
      OTHERS         = 3.
  IF sy-subrc NE 0.
    FORMAT COLOR 5.
    WRITE: / 'Error reading ml tables: ', i_ebeln, i_ebelp.
    HIDE ausgz.
    FORMAT RESET.
    o_error = 'X'.
    EXIT.
  ENDIF.

  REFRESH: it_ekbnk_cr.
* Parallelwährungen bestimmen
  DATA: BEGIN OF lt_curtp OCCURS 0,
          curtp LIKE cki_ekbecr-curtp,
          waers LIKE cki_ekbecr-waers,
        END OF lt_curtp.
  REFRESH lt_curtp.
  SORT lt_ekbz_cr BY curtp.
  DATA: old_curtp LIKE cki_ekbecr-curtp.
  CLEAR: old_curtp.
  LOOP AT lt_ekbz_cr.
    IF lt_ekbz_cr-curtp NE '00' AND
       lt_ekbz_cr-curtp NE '10'.
      IF lt_ekbz_cr-curtp NE old_curtp.
        old_curtp = lt_ekbz_cr-curtp.
        lt_curtp-curtp = lt_ekbz_cr-curtp.
        lt_curtp-waers = lt_ekbz_cr-waers.
        APPEND lt_curtp.
      ENDIF.
    ENDIF.
  ENDLOOP.
* Wert pro Währung aufsummieren
  DATA: l_frwwr LIKE ekbnk-frwwr.      "WE-Wert Fracht
  DATA: l_frrwr LIKE ekbnk-frrwr.      "RE-Wert Fracht
  DATA: l_frawe LIKE ekbnk-frawe.      "RE-Ausgleichswert Fracht
  DATA: l_stunr_old LIKE ekbnk-stunr.
  DATA: l_zaehk_old LIKE ekbnk-zaehk.
  DATA: l_first_loop.
  CLEAR: l_stunr_old, l_zaehk_old, l_first_loop.
  SORT it_ekbz BY ebeln ebelp stunr zaehk.
  LOOP AT lt_curtp.
    CLEAR: l_frwwr, l_frrwr, l_frawe.
    l_first_loop = 'X'.
    LOOP AT it_ekbz WHERE ebeln = i_ebeln
                      AND ebelp = i_ebelp.
      CHECK it_ekbz-vgabe = '1' OR
            it_ekbz-vgabe = '2' OR
            it_ekbz-vgabe = '3'.
      CHECK it_ekbz-bewtp = 'F' OR     "BNK WE Verrechnung
            it_ekbz-bewtp = 'G' OR     "BNK RE Verrechnung
            it_ekbz-bewtp = 'I' OR     "BNK KP Verrechnung
            it_ekbz-bewtp = 'K' OR     "KtoP Kontenpflege
            it_ekbz-bewtp = 'M' OR     "BzRe
            it_ekbz-bewtp = 'P'.       "NB-B
      IF l_first_loop = 'X'.
*       Initialization first loop.
        CLEAR: l_first_loop.
        l_stunr_old = it_ekbz-stunr.
        l_zaehk_old = it_ekbz-zaehk.
        CLEAR: it_ekbnk_cr.
        MOVE-CORRESPONDING it_ekbz TO it_ekbnk_cr.
        it_ekbnk_cr-curtp = lt_curtp-curtp.
        it_ekbnk_cr-waers = lt_curtp-waers.
      ELSE.
*       Control Break (Gruppenwechsel) at new Price Condition
        IF it_ekbz-stunr <> l_stunr_old OR
           it_ekbz-zaehk <> l_zaehk_old.
*         Create Record
          it_ekbnk_cr-frwwr = l_frwwr.
          it_ekbnk_cr-frrwr = l_frrwr.
          it_ekbnk_cr-frawe = l_frawe.
          APPEND it_ekbnk_cr.
*         Initialize for next Price Condition
          CLEAR: l_frwwr, l_frrwr, l_frawe.
          l_stunr_old = it_ekbz-stunr.
          l_zaehk_old = it_ekbz-zaehk.
          CLEAR: it_ekbnk_cr.
          MOVE-CORRESPONDING it_ekbz TO it_ekbnk_cr.
          it_ekbnk_cr-curtp = lt_curtp-curtp.
          it_ekbnk_cr-waers = lt_curtp-waers.
        ENDIF.
      ENDIF.
      READ TABLE lt_ekbz_cr
       WITH KEY ebeln = it_ekbz-ebeln
                ebelp = it_ekbz-ebelp
                vgabe = it_ekbz-vgabe
                stunr = it_ekbz-stunr
                zaehk = it_ekbz-zaehk
                gjahr = it_ekbz-gjahr
                belnr = it_ekbz-belnr
                buzei = it_ekbz-buzei
                curtp = lt_curtp-curtp.
      IF sy-subrc <> 0 AND
         NOT ( it_ekbz-dmbtr = 0 AND it_ekbz-wrbtr = 0 ).
        FORMAT COLOR 5.
        WRITE: / 'BIG TROUBLE WITH EKBNK_CR:', i_ebeln, i_ebelp.
        HIDE ausgz.
        FORMAT RESET.
        ls_big_trouble = 'X'.
        EXIT.
      ENDIF.
*     Soll/Haben-KZ in EKBZ genau umgekehrt wie in EKBE!!
*     ABER nur bei Wareneingängen (VGABE = 1)!!!
      IF it_ekbz-vgabe EQ '1'.
        IF it_ekbz-shkzg EQ 'H'.
          ADD lt_ekbz_cr-wrbtr TO l_frwwr.
        ELSE.
          SUBTRACT lt_ekbz_cr-wrbtr FROM l_frwwr.
        ENDIF.
      ELSEIF it_ekbz-vgabe EQ '2' OR
             it_ekbz-vgabe EQ '3'.
        IF it_ekbz-shkzg EQ 'S'.
          ADD lt_ekbz_cr-arewr TO l_frawe.
          ADD lt_ekbz_cr-wrbtr TO l_frrwr.
        ELSE.
          SUBTRACT lt_ekbz_cr-arewr FROM l_frawe.
          SUBTRACT lt_ekbz_cr-wrbtr FROM l_frrwr.
        ENDIF.
      ENDIF.
    ENDLOOP.
    IF NOT ls_big_trouble IS INITIAL.
      REFRESH: it_ekbnk_cr.
      EXIT.
    ENDIF.
*   Append last Record
    it_ekbnk_cr-frwwr = l_frwwr.
    it_ekbnk_cr-frrwr = l_frrwr.
    it_ekbnk_cr-frawe = l_frawe.
    APPEND it_ekbnk_cr.
  ENDLOOP.

ENDFORM.                               " GET_EBEFU_CR

                 USING    p_curtp LIKE mlbecr-curtp.

  DATA: lh_arewr_ml LIKE ebefu-arewr.

  LOOP AT pt_ekbe_help.
    SELECT * FROM  mlwere
             WHERE  ebeln = pt_ekbe_help-ebeln
             AND    ebelp = pt_ekbe_help-ebelp
             AND    stunr = h_stunr_space
             AND    zaehk = h_zaehk_space
             AND    zekkn = pt_ekbe_help-zekkn.
      SELECT * FROM mlbecr
               WHERE mlvnr = mlwere-mlvnr

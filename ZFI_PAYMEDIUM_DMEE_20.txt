FUNCTION zfi_paymedium_dmee_20.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(I_FPAYH) LIKE  FPAYH STRUCTURE  FPAYH
*"     VALUE(I_FPAYHX) LIKE  FPAYHX STRUCTURE  FPAYHX
*"     VALUE(I_FORMAT_PARAMS) LIKE  FPM_SELPAR-PARAM
*"     VALUE(I_FILENAME) LIKE  REGUT-FSNAM
*"     VALUE(I_XFILESYSTEM) TYPE  DFILESYST
*"  TABLES
*"      T_FILE_OUTPUT STRUCTURE  FPM_FILE
*"  CHANGING
*"     REFERENCE(C_FILENAME) LIKE  REGUT-FSNAM
*"----------------------------------------------------------------------
*data: bla(1) type n.
  DATA: wa_item  TYPE dmee_paym_if_type,
        params   TYPE fpm_selpar-param.
  DATA: error_output   TYPE STANDARD TABLE OF fimsg.
  data: it_fpm_fields  type standard table of dmee_fpm_fields.

  DATA: x_filesheet_tab TYPE xfeld,      "sum tablel for filesheet
        x_filesheet     TYPE xfeld.      "create filesheet
  data : p_treeid type fpayhx-formi.

  data : g_constant(1) type c value 'T',
         g_date(10),
         g_sdate(10),
         g_time(06),
         g_filename     type REGUT-FSNAM,
         wa_zfi_path    type zfi_path,
         g_ztxn        type zfi_s2s_txn-ztxn,
         wa_zfi_s2s_txn type zfi_s2s_txn.

* List of sheet key fields
  DATA: int_sheet_keys TYPE TABLE OF dmee_tree_sheet.

  params  = i_format_params.
  wa_item-fpayh  = i_fpayh.
  wa_item-fpayhx = i_fpayhx.
* wa_dmee_paym_if_param-****  = I_FORMAT_PARAMS

  CALL FUNCTION 'DMEE_START'
    EXPORTING
      i_tree_type = 'PAYM'
      i_tree_id   = i_fpayhx-formi
      item        = wa_item
      uparam      = i_format_params
    TABLES
      file_output = t_file_output.

  DATA: lwa_dmee_head TYPE dmee_tree_head.

  p_treeid = i_fpayhx-formi.
  SELECT SINGLE * FROM dmee_tree_head INTO lwa_dmee_head
    WHERE tree_type = 'PAYM'
    AND   tree_id   = p_treeid
    AND   version   = '000'.           "active version

  select single * from zfi_s2s_txn into wa_zfi_s2s_txn where rzawe = i_fpayh-rzawe.
  if sy-subrc = 0.

    g_ztxn = wa_zfi_s2s_txn-ztxn.
    write i_fpayh-laufd to g_date DDMMYY.
    write sy-datum to g_sdate DD/MM/YYYY.
    replace all occurences of '.' in g_sdate with '' .
    condense g_sdate.
    g_time = sy-uzeit.
* Begin of <RD1K976705> on 21062011
*    if i_fpayh-zbukr = 'MUM' and i_fpayh-hbkid = 'HB_21'.
*      concatenate g_ztxn '.' g_sdate '.' '94427' g_date i_fpayh-laufi g_time into g_filename.
*
*    elseif i_fpayh-zbukr = 'DDN' and i_fpayh-hbkid = 'HB_10'.
*      concatenate g_ztxn '.' g_sdate '.' '93128' g_date i_fpayh-laufi g_time into g_filename.
*
*    endif.

    select single * from zfi_path into wa_zfi_path where sysid = syst-sysid
                                                     and zbukr = i_fpayh-zbukr
                                                     and hbkid = i_fpayh-hbkid.
    if sy-subrc = 0.
* Begin of <RD1K973468>
      concatenate g_ztxn '.' g_sdate '.' wa_zfi_path-corpid g_date i_fpayh-laufi g_time into g_filename.
* End of <RD1K976705> on 21062011
      concatenate wa_zfi_path-fsnam g_filename '.txt' into g_filename.
*     concatenate wa_zfi_path-fsnam g_filename  into g_filename.
* End of <RD1K973468>
    endif.
*  concatenate '/usr/sap/RQ1/epay/' g_filename '.txt' into g_filename.
*  concatenate '/usr/sap/RD1/DigSignApp/data/output/signed/' g_filename '.txt' into g_filename.

    C_FILENAME = g_filename.  "'SPAN'.

*   Hand the run key (LAUFD/LAUFI) to the ZFI_PAYM_FILE enhancement ZBCM4
*   in the signing FM SSF_KRN_SIGN_BY_AS: there PM_LAUFD/PM_LAUFI are blank
*   and REGUT is not yet committed, so it cannot get the run key itself.
*   Here I_FPAYH carries it reliably. LAUFD/LAUFI are run-level (same for
*   every file of the run), so a single memory slot is enough.
    EXPORT laufd = i_fpayh-laufd
           laufi = i_fpayh-laufi
      TO MEMORY ID 'ZBCM_LAUF'.

  endif.

*                                                                      *
* BCM signature mapping - create approver entries in ZFI_BATCH_SIGN
*   - BATCH_NO = concatenated REGUT key (must match ZFI_BNK_APP1 logic):
*       ZBUKR(4) BANKS(3) LAUFD(8) LAUFI(6) XVORL(1) DTKEY(16) LFDNR(3)
*   - Approvers read from ZFI_BNK_RULE (ZRULE_ID = paying company code):
*       Rule 90700005 -> SNRO 1 signers
*       Rule 90700006 -> SNRO 2 signers
*                                                                      *
  DATA: lv_batch_no TYPE zfi_batch_sign-batch_no,
        l_bukrs     TYPE bukrs,        "ZBUKR  - 4
        l_banks     TYPE banks,        "BANKS  - 3
        l_laufd     TYPE laufd,        "LAUFD  - 8
        l_laufi     TYPE laufi,        "LAUFI  - 6
        l_xvorl     TYPE xvorl,        "XVORL  - 1
        l_dtkey     TYPE dtkey_d,      "DTKEY  - 16
        l_lfdnr     TYPE lfdnu,        "LFDNR  - 3
        ls_sign     TYPE zfi_batch_sign,
        lt_rule     TYPE STANDARD TABLE OF zfi_bnk_rule,
        ls_rule     TYPE zfi_bnk_rule.

*      REGUT key parts (local variables, lengths match REGUT)
* Known sources from the payment header:
  l_bukrs = i_fpayh-zbukr.
  l_laufd = i_fpayh-laufd.
  l_laufi = i_fpayh-laufi.
  l_xvorl = i_fpayh-xvorl.
  L_BANKS = i_fpayh-ZLAND.
  l_dtkey = i_fpayh-HBKID.
*  l_lfdnr =
* LFDNR must match the value the ZFI_PAYM_FILE enhancement (ZBCM4) stores,
* so the batch_no built here is found later. REGUT is not committed at this
* point (payment-medium LUW, before COMMIT), so MAX(REGUT) would return a
* stale value and every file would get LFDNR 1. Derive it the same way as
* ZBCM4: MAX(LFDNR)+1 over the ZFI_PAYM_FILE rows already committed for
* this run (prior files commit their row before the next file is signed).
* NOTE(UAT): assumes files are processed in REGUT creation order.
  DATA lv_cnt_lf TYPE i.
  select count(*) into lv_cnt_lf from zfi_paym_file
    WHERE laufd = l_laufd
      AND laufi = l_laufi.

  l_LFDNR = lv_cnt_lf + 1.
* >>> To be populated in the system (mapping to be done manually):
*  l_banks = ...        "Bank Country/Region Key
*  l_dtkey = ...        "Additional TemSe key field (e.g. HB_01)
*  l_lfdnr = ...        "Sequence number of the file within the run

  CLEAR lv_batch_no.
  CONCATENATE l_bukrs l_banks l_laufd l_laufi
              l_xvorl l_dtkey l_lfdnr
         INTO lv_batch_no RESPECTING BLANKS.

*      SNRO 1 approvers : Rule 90700005
  CLEAR lt_rule.
  SELECT * FROM zfi_bnk_rule INTO TABLE lt_rule
    WHERE zrule    = '90700005'
    AND   zrule_id = i_fpayh-zbukr.
  LOOP AT lt_rule INTO ls_rule.
    CLEAR ls_sign.
    ls_sign-batch_no = lv_batch_no.
    ls_sign-signer   = ls_rule-zuser.
    ls_sign-snro     = '1'.
    ls_sign-cdate    = sy-datum.
    ls_sign-ctime    = sy-uzeit.
    MODIFY zfi_batch_sign FROM ls_sign.
  ENDLOOP.

*      SNRO 2 approvers : Rule 90700006
  CLEAR lt_rule.
  SELECT * FROM zfi_bnk_rule INTO TABLE lt_rule
    WHERE zrule    = '90700006'
    AND   zrule_id = i_fpayh-zbukr.
  LOOP AT lt_rule INTO ls_rule.
    CLEAR ls_sign.
    ls_sign-batch_no = lv_batch_no.
    ls_sign-signer   = ls_rule-zuser.
    ls_sign-snro     = '2'.
    ls_sign-cdate    = sy-datum.
    ls_sign-ctime    = sy-uzeit.
    MODIFY zfi_batch_sign FROM ls_sign.
  ENDLOOP.
*                                                                      *


* -- simple file sheet
  IF lwa_dmee_head-filesheet = '1'.
    x_filesheet     = 'X'.
    x_filesheet_tab = space.
* -- file sheet with sums
  ELSEIF lwa_dmee_head-filesheet = '2'.
    x_filesheet     = 'X'.
    x_filesheet_tab = 'X'.
*   determine list of key fiels
    SELECT * FROM dmee_tree_sheet INTO TABLE int_sheet_keys
      WHERE tree_type = 'PAYM'
      AND   tree_id   = p_treeid
      AND   version   = '000'
      AND   x_key     = 'X'.           "key field
* -- no file sheet
  ELSE.
    x_filesheet     = space.
    x_filesheet_tab = space.
  ENDIF.




*BREAK-POINT.
*  PERFORM init_filesheet_keys USING i_fpayhx-formi.

ENDFUNCTION.

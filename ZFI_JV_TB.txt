*--------------------------------------------------------------*
*FORM get_opening.
*--------------------------------------------------------------*
* Opening balance per venture/account for p_year into lt_jvto1
* (HSLVT = company code INR, KSLVT = index based USD).
* JVA runs on ACDOCA from the START entry in JVAONACDOCAACTIV (GJACTCD).
* - p_year up to the START year: carry-forward of ledger 4A in JVTO1.
* - p_year after the START year: JVTO1 carry-forward of the START year
*   plus all ACDOCA postings (periods 001-016) from the START year up to
*   p_year - 1.
* The ACDOCA carry-forward (period 000), which the switch views use from
* the START year on, is not used: it does not hold correct balances per
* venture (opening of 2026 for MM1601 did not net to zero).
*--------------------------------------------------------------*
FORM get_opening.

  DATA : lv_start_year TYPE gjahr,
         lv_bcf_year   TYPE gjahr,
         lv_prev_year  TYPE gjahr,
         ls_open       TYPE ty_jvto1.

  REFRESH lt_jvto1.

  SELECT SINGLE gjahr
    FROM jvaonacdocaactiv
    INTO @lv_start_year
    WHERE activity_id = 'START'.
  IF sy-subrc NE 0 OR p_year LE lv_start_year.
    lv_bcf_year = p_year.
  ELSE.
    lv_bcf_year = lv_start_year.
  ENDIF.

  SELECT SUM( hslvt ) AS hslvt,
         SUM( kslvt ) AS kslvt,
         rjvnam,
         racct
    FROM jvto1
    INTO CORRESPONDING FIELDS OF TABLE @lt_jvto1
    WHERE ryear  =  @lv_bcf_year
      AND rbukrs =  @p_bukrs
      AND rjvnam IN @p_jvnam
      AND rldnr  =  '4A'
      AND rrcty  =  '0'
      AND rrecin IN @p_rrecin
    GROUP BY rjvnam, racct.

  IF lv_bcf_year LT p_year.
    lv_prev_year = p_year - 1.

    SELECT SUM( hsl ) AS hsl,
           SUM( ksl ) AS ksl,
           rjvnam,
           racct
      FROM jv_jvso1_acdoca
      INTO TABLE @DATA(lt_acd_mov)
      WHERE ryear  BETWEEN @lv_bcf_year AND @lv_prev_year
        AND rbukrs =  @p_bukrs
        AND rjvnam IN @p_jvnam
        AND poper  BETWEEN '001' AND '016'
        AND rldnr  =  '4A'
        AND rrcty  =  '0'
        AND rrecin IN @p_rrecin
      GROUP BY rjvnam, racct.

    LOOP AT lt_acd_mov INTO DATA(ls_acd_mov).
      CLEAR ls_open.
      ls_open-rjvnam = ls_acd_mov-rjvnam.
      ls_open-racct  = ls_acd_mov-racct.
      ls_open-hslvt  = ls_acd_mov-hsl.
      ls_open-kslvt  = ls_acd_mov-ksl.
      COLLECT ls_open INTO lt_jvto1.
    ENDLOOP.
  ENDIF.

ENDFORM.

    CALL FUNCTION 'CONVERSION_EXIT_INVDT_INPUT'
      EXPORTING
        input  = lv_budat
      IMPORTING
        output = lv_budat.

    SELECT SINGLE ukurs

           AND fcurr = 'USD'
           AND tcurr = 'INR'
           AND gdatu = lv_budat.

  ENDIF.
  gv_ukurs = gv_ukurs2.

  IF   lv_month2+1(2)  = '02'.
    lv_days = '28'.

      output = lv_budat2.

*&---------------------------------------------------------------------*
*&      Form  FETCHDATA_AL11
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*  -->  p1        text
*  <--  p2        text
*----------------------------------------------------------------------*
FORM fetchdata_al11 .

  IF ta_file IS NOT INITIAL.

    DELETE ta_file WHERE name NP co_pattern.

    CLEAR : wa_file.

    LOOP AT ta_file INTO wa_file.

      CLEAR : gv_inc_file, gv_arc_file.

      CONCATENATE gv_inc wa_file-name INTO gv_inc_file SEPARATED BY '\'.
      CONCATENATE gv_arc wa_file-name INTO gv_arc_file SEPARATED BY '\'.

      OPEN DATASET gv_inc_file FOR INPUT IN TEXT MODE ENCODING DEFAULT.

      OPEN DATASET gv_arc_file FOR OUTPUT IN TEXT MODE ENCODING DEFAULT.
      IF sy-subrc = 0.
        DO.
          READ DATASET gv_inc_file INTO wa_tab.
          IF sy-subrc <> 0.
            EXIT.
          ENDIF.

          gv_string = wa_tab.
          WHILE gv_string <> space.
            IF wa_tab+0(3) CA 'id'.
              CLEAR : gv_string.
              TRANSFER wa_tab TO gv_arc_file.
            ELSE.
              SPLIT wa_tab AT '|' INTO wa_al11-id
                                       wa_al11-lifnr
                                       wa_al11-taxno
                                       wa_al11-sr_company_code
                                       wa_al11-sr_purchase_org
                                       wa_al11-company_code
                                       wa_al11-payment_term_code
                                       wa_al11-purchasing_organization_code
                                       wa_al11-reco_account_code
                                       wa_al11-industry_key
                                       wa_al11-currency_code
                                       wa_al11-email
                                       wa_al11-contact_number
                                       wa_al11-payment_mode
                                       wa_al11-gr_based_iv
                                       wa_al11-bank_country
                                       wa_al11-account_number
                                       wa_al11-ifsc
                                       wa_al11-account_holder_name
                                       wa_al11-bank_name
                                       wa_al11-branch_address
                                       wa_al11-bic
                                       wa_al11-micr
                                       wa_al11-swift
                                       wa_al11-iban.
              APPEND wa_al11 TO ta_al11.
              CLEAR : wa_al11, gv_string.
            ENDIF.
          ENDWHILE.
          CLEAR : wa_tab.
        ENDDO.
      ELSE.
        MESSAGE text-003 TYPE 'E'.
      ENDIF.

      DELETE DATASET gv_inc_file.
      CLOSE DATASET gv_inc_file.
      CLOSE DATASET gv_arc_file.
      CLEAR : wa_file.
    ENDLOOP.
    CLEAR : gv_inc_file, gv_arc_file, gv_string.
  ELSE.
    MESSAGE text-002 TYPE 'E'.
    LEAVE LIST-PROCESSING.
  ENDIF.
  CLEAR wa_al11.

ENDFORM.

FORM calling_bdc .

  IF ta_al11 IS NOT INITIAL.
    LOOP AT ta_al11 ASSIGNING FIELD-SYMBOL(<fs>).
      <fs>-lifnr = |{ <fs>-lifnr ALPHA = IN }|.
    ENDLOOP.
  ENDIF.

* FETCH EXISTING COMPANY CODE
*--------------------------------------------------------------------*
  SELECT lifnr,
         bukrs,
         qland
    FROM lfb1
    INTO TABLE @DATA(lt_lfb1)
    FOR ALL ENTRIES IN @ta_al11
    WHERE lifnr = @ta_al11-lifnr
      AND bukrs = @ta_al11-company_code.

*--------------------------------------------------------------------*
* FETCH EXISTING PURCHASE ORG
*--------------------------------------------------------------------*
  SELECT lifnr,
         ekorg,
         kzaut
    FROM lfm1
    INTO TABLE @DATA(lt_lfm1)
    FOR ALL ENTRIES IN @ta_al11
    WHERE lifnr = @ta_al11-lifnr
      AND ekorg = @ta_al11-purchasing_organization_code.

*--------------------------------------------------------------------*
* FETCH ACCOUNT GROUP + SSI TYPE
*--------------------------------------------------------------------*
  SELECT lifnr,
         ktokk
    FROM lfa1
    INTO TABLE @DATA(lt_lfa1)
    FOR ALL ENTRIES IN @ta_al11
    WHERE lifnr = @ta_al11-lifnr.

*--------------------------------------------------------------------*
* FETCH PAN
*--------------------------------------------------------------------*
  SELECT lifnr,
         j_1ipanno,
         j_1issist
    FROM j_1imovend
    INTO TABLE @DATA(lt_j_1imovend)
    FOR ALL ENTRIES IN @ta_al11
    WHERE lifnr = @ta_al11-lifnr.

*--------------------------------------------------------------------*
* FETCH WHT DETAILS
*--------------------------------------------------------------------*
  SELECT lifnr,
         bukrs,
         witht,
         wt_withcd,
         wt_subjct,
         qsrec,
         wt_exnr,
         wt_exrt,
         wt_wtexrs,
         wt_exdf,
         wt_exdt
    FROM lfbw
    INTO TABLE @DATA(lt_lfbw)
    FOR ALL ENTRIES IN @ta_al11
    WHERE lifnr = @ta_al11-lifnr
      AND bukrs = @ta_al11-sr_company_code.

  SELECT *
    FROM zfi_tax_tab
    INTO TABLE @DATA(lt_zfi_tax_tab)
    FOR ALL ENTRIES IN @ta_al11
    WHERE lifnr = @ta_al11-lifnr.

*--------------------------------------------------------------------*
* SORT
*--------------------------------------------------------------------*
  SORT lt_lfb1 BY lifnr bukrs.
  SORT lt_lfm1 BY lifnr ekorg.
  SORT lt_lfa1 BY lifnr.
  SORT lt_j_1imovend  BY lifnr.
  SORT lt_lfbw BY lifnr bukrs witht.
  SORT lt_zfi_tax_tab BY lifnr.

*--------------------------------------------------------------------*
* LOOP INPUT FILE
*--------------------------------------------------------------------*
  LOOP AT ta_al11 INTO wa_al11.

*--------------------------------------------------------------------*
* CHECK COMPANY CODE
*--------------------------------------------------------------------*
    READ TABLE lt_lfb1 INTO DATA(wa_lfb1) WITH KEY lifnr = wa_al11-lifnr
                                                   bukrs = wa_al11-company_code BINARY SEARCH.
    DATA(lv_lfb1) = sy-subrc.

*--------------------------------------------------------------------*
* CHECK PURCHASE ORG
*--------------------------------------------------------------------*

                                                   ekorg = wa_al11-purchasing_organization_code BINARY SEARCH.
    DATA(lv_lfm1) = sy-subrc.

*--------------------------------------------------------------------*
* ALREADY EXTENDED
*--------------------------------------------------------------------*

      wa_sft-id         = wa_al11-id.
      wa_sft-vendor     = wa_al11-lifnr.
      wa_sft-vendortype = 'VEND_EXT'.
      wa_sft-date       = sy-datum.
      wa_sft-msg        = 'INFO'.
      wa_sft-message    = 'Vendor already extended'.

*--------------------------------------------------------------------*
* FETCH ACCOUNT GROUP + SSI
*--------------------------------------------------------------------*

* FETCH PAN

*&---------------------------------------------------------------------*

*----------------------------------------------------------------------*
FORM send_data .

  SELECT SINGLE *
    FROM zparam_table
    INTO @DATA(wa_zparam)
    WHERE id_name = 'CIM_VNRSP'
    AND param_name NE ''.

  IF sy-subrc IS NOT INITIAL.
    MESSAGE 'Path for AL11 not maintained' TYPE 'I'.
  ENDIF.

  IF ta_sft IS NOT INITIAL.

    SELECT lifnr,ekorg,werks,parvw,parza,lifn2 FROM wyt3
      INTO TABLE @DATA(ta_partner)
      FOR ALL ENTRIES IN @ta_sft
      WHERE lifnr = @ta_sft-vendor.

    wa_sftp1-id     = 'Id' .
    wa_sftp1-vendor  = 'SAPVendorNo'.
    wa_sftp1-vendortype = 'VendorType'.
    wa_sftp1-date   = 'CreatedOnDate'.
    wa_sftp1-plant  = 'Plant'.
    wa_sftp1-part_func = 'PartnerFunction'.
    wa_sftp1-part_count = 'PartnerCounter'.
    wa_sftp1-ref = 'ReftoOthervVendor'.
    wa_sftp1-message = 'Message'.
    APPEND wa_sftp1 TO ta_sftp1.
    CLEAR wa_sftp1.

    LOOP AT ta_sft INTO wa_sft.
      wa_sftp1-id    =  wa_sft-id.
      wa_sftp1-vendor = wa_sft-vendor.
      wa_sftp1-vendortype = wa_sft-vendortype.
      wa_sftp1-date   = wa_sft-date.
      READ TABLE ta_partner INTO DATA(wa_partner) WITH KEY lifnr = wa_sft-vendor.
      IF sy-subrc = 0.
        DATA(lv_index) = sy-tabix.
        LOOP AT ta_partner INTO DATA(wa_partner2) FROM lv_index.
          IF wa_partner-lifnr <> wa_partner2-lifn2.
            EXIT.
          ELSE.
            wa_sftp1-part_func = |{  wa_sftp1-part_func }-{ wa_partner-parvw }|.
          ENDIF.
        ENDLOOP.
        CLEAR : lv_index,wa_partner2.

        wa_sftp1-plant = wa_partner-werks.
        wa_sftp1-part_count = wa_partner-parza.
        wa_sftp1-ref = wa_partner-lifn2.
      ENDIF.

      wa_sftp1-message = wa_sft-message.
      APPEND wa_sftp1 TO ta_sftp1.
      CLEAR: wa_sftp1,wa_sft.
    ENDLOOP.
  ENDIF.

  htab = cl_abap_char_utilities=>horizontal_tab.
  DATA(lv_count) = lines( ta_sftp1 ).
  IF ta_sftp1[] IS NOT INITIAL AND lv_count > 1.

    DATA(lv_variant) = syst-slset.

    CLEAR : gv_al11_path.

    lv_date = sy-datlo.

    CONCATENATE lv_date+6(2) lv_date+4(2) lv_date+0(4) INTO lv_ddmmyyyy.

    IF lv_variant IS INITIAL.
      gv_al11_path = |{ wa_zparam-param_name }{ '\' }{ co_filename }{ lv_ddmmyyyy }{ sy-uzeit }{ '.csv' }|.
    ELSE.
      gv_al11_path = |{ wa_zparam-param_name }{ '\' }{ co_filename }{ lv_ddmmyyyy }{ sy-uzeit }{ '.csv' }|.
    ENDIF.

    OPEN DATASET gv_al11_path IN TEXT MODE FOR OUTPUT ENCODING UTF-8.

    IF sy-subrc = 0.
      LOOP AT ta_sftp1 INTO wa_sftp1.

        CONCATENATE
          wa_sftp1-id
          wa_sftp1-vendor
          wa_sftp1-vendortype
          wa_sftp1-date
          wa_sftp1-plant
          wa_sftp1-part_func
          wa_sftp1-part_count
          wa_sftp1-ref
          wa_sftp1-message
         INTO lv_string SEPARATED BY co_comma.

        TRANSFER lv_string TO gv_al11_path.
        CLEAR: lv_string, wa_sftp1.
      ENDLOOP.
    ENDIF.
    CLOSE DATASET gv_al11_path.
    MESSAGE text-004 TYPE 'S'.

  ELSE.
    MESSAGE text-005 TYPE 'I'.
  ENDIF.


ENDFORM.

FORM error_log .

  IF lt_alv IS NOT INITIAL.
    LOOP AT lt_alv INTO ls_alv.
      CONCATENATE ls_alv-pyord
                  ls_alv-zstatus
                  ls_alv-zrej_reason
                  ls_alv-bktxt
                  ls_alv-zuonr
                  ls_alv-zutrno
                  ls_alv-zpaydate
                  ls_alv-messg
                  ls_alv-zerr_filename INTO ls_data-data SEPARATED BY ','.
      APPEND ls_data TO lt_data.
      CLEAR ls_data.
    ENDLOOP.
  ENDIF.

  CALL FUNCTION 'SCMS_TEXT_TO_BINARY'
    IMPORTING
      output_length = ls_params-indatalen
    TABLES
      text_tab      = lt_data
      binary_tab    = lt_bin_data.
  IF sy-subrc <> 0.
* Implement suitable error handling here
  ENDIF.

* Convert Binary data to Raw String
  CALL FUNCTION 'SCMS_BINARY_TO_XSTRING'
    EXPORTING
      input_length = ls_params-indatalen
    IMPORTING
      buffer       = lv_binary
    TABLES
      binary_tab   = lt_bin_data
    EXCEPTIONS
      failed       = 1
      OTHERS       = 2.

  ls_bank_data-prog_name = 'ZFI_BCM_VEN_SBIFILE_CPY'.
  ls_bank_data-error_log = lv_binary.
  ls_bank_data-error_log_len = ls_params-indatalen.

  MODIFY zfi_bnk_input_f FROM ls_bank_data.
  COMMIT WORK AND WAIT.
ENDFORM.                    " ERROR_LOG

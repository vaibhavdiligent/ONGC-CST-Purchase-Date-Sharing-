FORM fetchdata_al11 .

  IF ta_file IS NOT INITIAL.

    DELETE ta_file WHERE name NP co_pattern.

    CLEAR : wa_file.

    LOOP AT ta_file INTO wa_file.

      CLEAR : gv_inc_file, gv_arc_file.

      " Code Remediation S4 AL11 file path change **BEGIN OF CHANGE BY ABAP7 25.10.2026
*     CONCATENATE gv_inc wa_file-name INTO gv_inc_file SEPARATED BY '\'.
*     CONCATENATE gv_arc wa_file-name INTO gv_arc_file SEPARATED BY '\'.
      CONCATENATE gv_inc wa_file-name INTO gv_inc_file SEPARATED BY '/'.
      CONCATENATE gv_arc wa_file-name INTO gv_arc_file SEPARATED BY '/'.
      " Code Remediation S4 AL11 file path change **END OF CHANGE BY ABAP7 25.10.2026

      OPEN DATASET gv_inc_file FOR INPUT IN TEXT MODE ENCODING DEFAULT.

      OPEN DATASET gv_arc_file FOR OUTPUT IN TEXT MODE ENCODING DEFAULT.
      IF sy-subrc = 0.
        DO.
          READ DATASET gv_inc_file INTO wa_tab.
          IF sy-subrc <> 0.
            EXIT.
          ENDIF.

          gv_string = wa_tab.
          WHILE gv_string <> space.
            IF wa_tab+0(3) CA 'id'.
              CLEAR : gv_string.
              TRANSFER wa_tab TO gv_arc_file.
            ELSE.
              SPLIT wa_tab AT '|' INTO wa_al11-id
                                       wa_al11-lifnr
                                       wa_al11-bukrs
                                       wa_al11-ekorg
                                       wa_al11-sperr
                                       wa_al11-sperr1
                                       wa_al11-sperm
                                       wa_al11-sperm1
                                       wa_al11-sperq
                                       wa_al11-zahls.
              APPEND wa_al11 TO ta_al11.
              CLEAR : wa_al11, gv_string.
            ENDIF.
          ENDWHILE.
          CLEAR : wa_tab.
        ENDDO.
      ELSE.
        MESSAGE text-003 TYPE 'E'.
      ENDIF.

      DELETE DATASET gv_inc_file.
      CLOSE DATASET gv_inc_file.
      CLOSE DATASET gv_arc_file.
      CLEAR : wa_file.
    ENDLOOP.
    CLEAR : gv_inc_file, gv_arc_file, gv_string.
  ELSE.
    MESSAGE text-002 TYPE 'E'.
    LEAVE LIST-PROCESSING.
  ENDIF.
  CLEAR wa_al11.

ENDFORM.

FORM process_data .

  LOOP AT ta_al11 INTO wa_al11.

FORM bdc_dynpro USING program dynpro.
  CLEAR wa_bdcdata.
  wa_bdcdata-program  = program.
  wa_bdcdata-dynpro   = dynpro.
  wa_bdcdata-dynbegin = 'X'.
  APPEND wa_bdcdata TO it_bdcdata.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  BDC_XK05
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*      -->WA_AL11  text
*----------------------------------------------------------------------*
FORM bdc_xk05  USING wa_al11 TYPE ty_data.

  REFRESH it_bdcdata.
  REFRESH it_msg.

  PERFORM bdc_dynpro USING 'SAPMF02K' '0500'.
  PERFORM bdc_field USING 'BDC_OKCODE' '/00'.
  PERFORM bdc_field USING 'RF02K-LIFNR' wa_al11-lifnr.
  PERFORM bdc_field USING 'RF02K-BUKRS' wa_al11-bukrs.
  PERFORM bdc_field USING 'RF02K-EKORG' wa_al11-ekorg.

  PERFORM bdc_dynpro USING 'SAPMF02K' '0510'.
  PERFORM bdc_field USING 'LFA1-SPERR' wa_al11-sperr.
  PERFORM bdc_field USING 'LFB1-SPERR' wa_al11-sperr1.

  IF wa_al11-ekorg IS NOT INITIAL.
    PERFORM bdc_field USING 'LFA1-SPERM' wa_al11-sperm.
    PERFORM bdc_field USING 'LFM1-SPERM' wa_al11-sperm1.
  ENDIF.
  IF wa_al11-sperq IS NOT INITIAL.
    wa_al11-sperq = |{ wa_al11-sperq WIDTH = 2 PAD = '0' ALIGN = RIGHT }|.
  ENDIF.

  PERFORM bdc_field USING 'LFA1-SPERQ' wa_al11-sperq.
  PERFORM bdc_field USING 'BDC_OKCODE' '=UPDA'.

  CALL TRANSACTION 'XK05' USING it_bdcdata MODE 'N' UPDATE 'S' MESSAGES INTO it_msg.

  PERFORM collect_messages USING wa_al11-lifnr 'XK05'.

ENDFORM.

*&      Form  COLLECT_MESSAGES
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*      -->P_LIFNR  text
*      -->P_tcode   text
*----------------------------------------------------------------------*
FORM collect_messages  USING    p_lifnr TYPE lifnr
                                p_tcode TYPE tcode.

  DATA : lv_text TYPE string.

  LOOP AT it_msg INTO wa_msg.

    CALL FUNCTION 'MESSAGE_TEXT_BUILD'
      EXPORTING
        msgid               = wa_msg-msgid
        msgnr               = wa_msg-msgnr
        msgv1               = wa_msg-msgv1
        msgv2               = wa_msg-msgv2
        msgv3               = wa_msg-msgv3
        msgv4               = wa_msg-msgv4
      IMPORTING
        message_text_output = lv_text.

    CLEAR wa_log.
    wa_log-lifnr = p_lifnr.
    wa_log-tcode = p_tcode.

    CASE wa_msg-msgtyp.
      WHEN 'S'.
        wa_log-msg = |SUCCESS: { lv_text }|.
      WHEN 'E'.
        wa_log-msg = |ERROR: { lv_text }|.
      WHEN 'W'.
        wa_log-msg = |WARNING: { lv_text }|.
      WHEN 'I'.
        wa_log-msg = |INFO: { lv_text }|.
      WHEN 'A'.
        wa_log-msg = |ABORT: { lv_text }|.
      WHEN OTHERS.
        wa_log-msg = lv_text.
    ENDCASE.

    APPEND wa_log TO it_log.

  ENDLOOP.

ENDFORM.

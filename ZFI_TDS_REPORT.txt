FORM get_bkpf .
  REFRESH:lt_bkpf.
*     S.O.C - Perf: earlier ALL documents of the company code in the
*     posting period were read (and a separate IF for an empty P_BUKRS,
*     which is OBLIGATORY, so that branch could never run). Only
*     documents that have a withholding tax line can reach the output,
*     so read just those. WAERS/KURSF are taken here so the display loop
*     no longer selects BKPF once per document.
  SELECT bukrs, belnr, gjahr, bldat, budat, xblnr, stblg, bktxt,
         waers, kursf
    FROM bkpf
   WHERE bukrs =  @p_bukrs
     AND budat IN @postdate
     AND EXISTS ( SELECT belnr FROM with_item
                   WHERE bukrs    =  bkpf~bukrs
                     AND belnr    =  bkpf~belnr
                     AND gjahr    =  bkpf~gjahr
                     AND wt_qsshh <> 0 )
    INTO TABLE @lt_bkpf.
*     E.O.C
ENDFORM.                    " GET_BKPF

FORM get_withitem .
  CLEAR:lt_with_item.
  REFRESH:lt_with_item.

  IF lt_bkpf IS NOT INITIAL.
    SELECT belnr
           bukrs
           gjahr
           witht
           wt_qsshh
           wt_acco
           wt_qbshh
      FROM with_item INTO TABLE lt_with_item
      FOR ALL ENTRIES IN lt_bkpf
      WHERE bukrs = lt_bkpf-bukrs
        AND belnr = lt_bkpf-belnr
        AND gjahr = lt_bkpf-gjahr
        AND wt_qsshh NE 0.
  ENDIF.
ENDFORM.                    " GET_WITHITEM

FORM get_vendor .
  IF lt_bseg IS NOT INITIAL.
    SELECT lifnr
           land1
           name1
           stcd3
      FROM lfa1
      INTO TABLE lt_lfa1
      FOR ALL ENTRIES IN lt_bseg
      WHERE lifnr = lt_bseg-lifnr
        AND lifnr IN vendor.

  ENDIF.

*     S.O.C - Perf: vendors from WITH_ITEM-WT_ACCO were read with a
*     SELECT SINGLE per row inside GET_DISPLAY. Read them once here.
*     (No VENDOR range here - the old SELECT SINGLE did not apply it.)
  IF lt_with_item IS NOT INITIAL.
    SELECT lifnr, land1, name1, stcd3
      FROM lfa1
       FOR ALL ENTRIES IN @lt_with_item
     WHERE lifnr = @lt_with_item-wt_acco
      INTO TABLE @lt_lfa1_acco.
  ENDIF.

  IF lt_lfa1 IS NOT INITIAL OR lt_lfa1_acco IS NOT INITIAL.
*     Country texts for both vendor sets (WT_ACCO vendors were missing)
    DATA(lt_land1) = VALUE rsdsselopt_t(
      FOR ls_l IN lt_lfa1      ( sign = 'I' option = 'EQ' low = ls_l-land1 ) ).
    lt_land1 = VALUE #( BASE lt_land1
      FOR ls_a IN lt_lfa1_acco ( sign = 'I' option = 'EQ' low = ls_a-land1 ) ).
    SORT lt_land1 BY low.
    DELETE ADJACENT DUPLICATES FROM lt_land1 COMPARING low.
    SELECT spras, land1, landx
      FROM t005t
     WHERE land1 IN @lt_land1
       AND spras = 'E'
      INTO TABLE @lt_t005t.
  ENDIF.
*     E.O.C
ENDFORM.

  SORT lt_with_item BY belnr bukrs gjahr.
  SORT lt_lfa1 BY lifnr.
  SORT lt_bseg BY belnr gjahr bukrs.
  SORT lt_ekko BY ebeln.
  SORT lt_t059z BY witht.
  SORT lt_t005t BY land1.

  DATA: a TYPE i VALUE 0.
  LOOP AT  lt_with_item INTO ls_with_item WHERE   witht = 'IZ' OR witht = 'CZ' OR witht = 'SZ'.
    ls_final-witht = ls_with_item-witht.
    ls_final-wt_qsshh = ls_with_item-wt_qsshh.
    ls_final-wt_qbshh = ls_with_item-wt_qbshh.

                                             gjahr = ls_with_item-gjahr
                               bukrs = ls_with_item-bukrs BINARY SEARCH.

    "S.O.C BY ROHIT ON 13-08-2025
    a = 0.
    IF ls_with_item-wt_acco NE ls_bseg-lifnr.

      ls_bseg-lifnr = ls_with_item-wt_acco.
      a = 1.
    ENDIF.

    "E.O.C BY ROHIT ON 13-08-2025
    IF a = 0.
      READ TABLE lt_lfa1 INTO ls_lfa1 WITH KEY  lifnr  = ls_bseg-lifnr BINARY SEARCH.
      IF sy-subrc = 0.

        ls_final-name1 = ls_lfa1-name1.
        ls_final-stcd3 = ls_lfa1-stcd3.

        CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
          EXPORTING
            input  = ls_lfa1-lifnr
          IMPORTING
            output = ls_lfa1-lifnr.
        ls_final-lifnr = ls_lfa1-lifnr.
        ls_final-land1 = ls_lfa1-land1.
      ENDIF.
    ELSEIF a = 1.
*     Perf: pre-read in GET_VENDOR instead of SELECT SINGLE per row
      READ TABLE lt_lfa1_acco INTO ls_lfa1 WITH TABLE KEY lifnr = ls_bseg-lifnr.

      IF sy-subrc = 0.

        ls_final-name1 = ls_lfa1-name1.
        ls_final-stcd3 = ls_lfa1-stcd3.

        CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
          EXPORTING
            input  = ls_lfa1-lifnr
          IMPORTING
            output = ls_lfa1-lifnr.
        ls_final-lifnr = ls_lfa1-lifnr.
        ls_final-land1 = ls_lfa1-land1.
      ENDIF.
    ENDIF.
    READ TABLE lt_t005t INTO ls_t005t WITH KEY land1 = ls_lfa1-land1 BINARY SEARCH.
    IF sy-subrc = 0.
      ls_final-landx = ls_t005t-landx.
    ENDIF.
    READ TABLE lt_bkpf INTO ls_bkpf  WITH KEY belnr = ls_with_item-belnr
                                              bukrs = ls_with_item-bukrs
                               gjahr = ls_with_item-gjahr BINARY SEARCH.
    IF sy-subrc = 0 .

           ebelp

      FOR ALL ENTRIES IN lt_ekko
      WHERE ebeln = lt_ekko-ebeln.
  ENDIF.

  LOOP AT lt_ekpo INTO ls_ekpo.
    COLLECT ls_ekpo INTO lt_ekpo1.
  ENDLOOP.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  ADD_FCAT
*&---------------------------------------------------------------------*
*       Append one column to the ALV field catalog I_FCAT
*----------------------------------------------------------------------*
FORM add_fcat USING pv_pos    TYPE any
                    pv_field  TYPE any
                    pv_text   TYPE any
                    pv_reftab TYPE any
                    pv_reffld TYPE any
                    pv_len    TYPE any.
  CLEAR wa_fcat.
  wa_fcat-col_pos       = pv_pos.
  wa_fcat-fieldname     = pv_field.
  wa_fcat-tabname       = 'LT_FINAL9'.
  wa_fcat-seltext_s     = pv_text.
  wa_fcat-seltext_m     = pv_text.
  wa_fcat-seltext_l     = pv_text.
  wa_fcat-reptext_ddic  = pv_text.
  wa_fcat-ddictxt       = 'L'.
  wa_fcat-outputlen     = pv_len.
  IF pv_reftab IS NOT INITIAL.
    wa_fcat-ref_tabname   = pv_reftab.
    wa_fcat-ref_fieldname = pv_reffld.
  ELSE.
*   Local CHAR fields (EFFWR, TDS, AMT_TDS, DATE_TDS, GSTIN) - no DDIC ref
    wa_fcat-inttype   = 'C'.
    wa_fcat-datatype  = 'CHAR'.
    wa_fcat-intlen    = pv_len.
    wa_fcat-ddic_outputlen = pv_len.
  ENDIF.
  IF pv_field = 'EFFWR'.
    wa_fcat-just = 'R'.
  ENDIF.
  APPEND wa_fcat TO i_fcat.
  CLEAR wa_fcat.
ENDFORM.

 " HTTP Client Variables
DATA: lo_http_client TYPE REF TO if_http_client,
      lv_xml         TYPE string,          " Changed from lv_json to lv_xml
      lv_json type string,
      lv_status_code TYPE i,
      lv_reason      TYPE string,
      lv_response    TYPE string,
      lt_payload type table of ty_cpi_payload.
 CONSTANTS: lc_destination TYPE rfcdest VALUE 'CPI_API_OB'. " Configured in SM59
"Fill payload
 lt_payload = VALUE #(
  FOR ls_asrs IN gt_asrs WHERE ( plant = iv_plant )
  ( VALUE #(
      BASE CORRESPONDING #( ls_asrs )
      processname = 'ASR_PUSH_MESSAGE' ) ) ).
 " 3. Convert Internal Table to XML String
  CALL TRANSFORMATION id
    SOURCE items = lt_payload
    RESULT XML lv_xml.
   " 4. Instantiate HTTP Client
  cl_http_client=>create_by_destination(
    EXPORTING
      destination              = lc_destination
    IMPORTING
      client                   = lo_http_client
    EXCEPTIONS
      argument_not_found       = 1
      destination_not_found    = 2
      destination_no_authority = 3
      plugin_not_active        = 4
      internal_error           = 5
      OTHERS                   = 6
  ).
     IF sy-subrc <> 0.
    MESSAGE 'Error creating HTTP Client.' TYPE 'E'.
    RETURN.
  ENDIF.
    " 5. Configure HTTP Headers for XML
  lo_http_client->request->set_method( if_http_request=>co_request_method_post ).
  lo_http_client->request->set_header_field( name = 'Content-Type' value = 'application/xml' ).
  lo_http_client->request->set_header_field( name = 'Accept'       value = 'application/xml' ).
  lo_http_client->request->set_header_field( name = 'X-Plant' value = CONV string( iv_plant ) ).

  " Set XML payload
  lo_http_client->request->set_cdata( lv_xml ).
*  lo_http_client->request->set_cdata( lv_json ).

  " 6. Send HTTP Request
  lo_http_client->send(
    EXCEPTIONS
      http_communication_failure = 1
      http_invalid_state         = 2
      http_processing_failed     = 3
      OTHERS                     = 4
  ).

  IF sy-subrc <> 0.
    lo_http_client->get_last_error( IMPORTING message = lv_reason ).
    MESSAGE |Communication Failure: { lv_reason }| TYPE 'E'.
    lo_http_client->close( ).
    RETURN.
  ENDIF.
   " 7. Receive Response
  lo_http_client->receive(
    EXCEPTIONS
      http_communication_failure = 1
      http_invalid_state         = 2
      http_processing_failed     = 3
      OTHERS                     = 4
  ).

  " Get Response details
  lo_http_client->response->get_status(
    IMPORTING
      code   = lv_status_code
      reason = lv_reason
  ).

  lv_response = lo_http_client->response->get_cdata( ).
  IF lv_status_code BETWEEN 200 AND 299.
   MESSAGE |Payload sent successfully in XML format! Status: { lv_status_code }| TYPE 'S'.
  ELSE.
    MESSAGE |Failed to send XML payload. Error: { lv_status_code } { lv_reason }| type 'E'.
  ENDIF.

  " Close Connection
  lo_http_client->close( ).
ENDFORM.

*&---------------------------------------------------------------------*
*& Report ZDUAL_MAINTENANCE_RPT
*&---------------------------------------------------------------------*
*& Reads the object list of a transport request (and its tasks) from a
*& source system via RFC and re-creates the selected objects in this
*& system, recording them in a local transport request.
*&
*& Supported object types (E071-OBJECT):
*&   1. DOMA / DOMD          Domain            (incl. fixed values)
*&   2. DTEL / DTED          Data element      (texts in all languages)
*&   3. TABL / TABD / TABT   Table / structure (incl. secondary indexes)
*&   4. PROG / REPS / REPT / INCL  Report, include, text elements
*&   5. FUGR / FUNC          Function group (TOP + includes + all FMs)
*&                           or single function module
*&   6. TRAN                 Transaction code (TSTC/TSTCT/TSTCP/TSTCC)
*&   7. TTYP / TTYD          Table type
*&   8. VIEW / VIED          Database / maintenance / help view
*&   9. SHLP / SHLD          Search help
*&  10. ENQU / ENQD          Lock object
*&  11. TYPE / TYPD          Type group
*&  12. MSAG / MESS          Message class / single message
*&  13. PARA                 SET/GET parameter ID
*&  14. NROB                 Number range object
*&  15. INTF / CLAS (+ CLSD CPUB CPRO CPRI CINC METH)  Interface / class
*&  16. DYNP / CUAD          Screens and GUI status (also copied
*&                           automatically with every program / FUGR)
*&  17. TOBJ                 Table maintenance object (SE54)
*&  18. VCLS                 View cluster
*&  19. SXCI                 Classic BAdI implementation
*&  20. TABU                 Table contents (keys from E071K)
*&  21. SUSO / AUTH          Authorization object / authorization field
*&
*& NOT COPIED (NO REMOTE WRITE-BACK API): SSFO, SFPF, ENHO/ENHS, SMIM
*&
*& DDIC objects are first written inactive and then activated together
*& in dependency order (DOMA -> DTEL -> TABL/TTYP -> VIEW/SHLP/ENQU),
*& so objects of the same request that depend on each other activate.
*& Source systems are read with remote-enabled APIs only: RFC_READ_TABLE,
*& DD_*_GET, RPY_PROGRAM_READ, RS_CUA_INTERNAL_FETCH and the version
*& management readers SVRS_GET_VERSION_*_40 (remote comparison).
*&---------------------------------------------------------------------*
REPORT zdual_maintenance_rpt.

*---------------------------------------------*
* HELPER CLASS

CLASS lcl_util DEFINITION.
  PUBLIC SECTION.
    CLASS-METHODS:
      " ADD A LINE TO THE RESULT LOG (FOR 'E' THE SY-MSG* TEXT IS APPENDED)
      add_msg IMPORTING iv_type TYPE csequence
                        iv_text TYPE csequence,
      " COLLECT AN OBJECT FOR THE LOCAL TRANSPORT REQUEST
      add_tr_obj IMPORTING iv_pgmid  TYPE csequence
                           iv_object TYPE csequence
                           iv_name   TYPE csequence,
      " CREATE THE TADIR ENTRY (PACKAGE ASSIGNMENT) IF IT DOES NOT EXIST
      create_tadir IMPORTING iv_object TYPE csequence
                             iv_name   TYPE csequence,
      " RFC_READ_TABLE ON THE SOURCE SYSTEM, RESULT MAPPED BY FIELD NAME
      rfc_read IMPORTING iv_table  TYPE csequence
                         it_where  TYPE tt_options
                         it_fields TYPE tt_names OPTIONAL
                         iv_quiet  TYPE abap_bool DEFAULT abap_false
               EXPORTING et_data   TYPE STANDARD TABLE
                         ev_subrc  TYPE sy-subrc.
ENDCLASS.

CLASS lcl_util IMPLEMENTATION.

  METHOD add_msg.
    DATA lv_text TYPE bapi_msg.

    lv_text = iv_text.
    IF iv_type = 'E' AND sy-msgid IS NOT INITIAL AND sy-msgty IS NOT INITIAL.
      MESSAGE ID sy-msgid TYPE 'S' NUMBER sy-msgno
              WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4 INTO DATA(lv_sap_text).
      lv_text = |{ iv_text }: { lv_sap_text }|.
    ENDIF.

    APPEND VALUE #( type    = iv_type
                    message = lv_text ) TO t_return.
    CLEAR: sy-msgid, sy-msgty, sy-msgno, sy-msgv1, sy-msgv2, sy-msgv3, sy-msgv4.
  ENDMETHOD.

  METHOD add_tr_obj.
    DATA ls_object TYPE ko200.

    ls_object-pgmid    = iv_pgmid.
    ls_object-object   = iv_object.
    ls_object-obj_name = iv_name.
    READ TABLE gt_objects WITH KEY pgmid    = ls_object-pgmid
                                   object   = ls_object-object
                                   obj_name = ls_object-obj_name
                          TRANSPORTING NO FIELDS.
    IF sy-subrc <> 0.
      APPEND ls_object TO gt_objects.
    ENDIF.
  ENDMETHOD.

  METHOD create_tadir.
    DATA: lv_object TYPE tadir-object,
          lv_name   TYPE tadir-obj_name.

    lv_object = iv_object.
    lv_name   = iv_name.

    SELECT SINGLE devclass FROM tadir INTO @DATA(lv_devclass)
      WHERE pgmid    = 'R3TR'
        AND object   = @lv_object
        AND obj_name = @lv_name.
    IF sy-subrc = 0 AND lv_devclass IS NOT INITIAL.
      RETURN.  " ALREADY ASSIGNED TO A PACKAGE - DO NOT MOVE IT
    ENDIF.

    CALL FUNCTION 'TR_TADIR_INTERFACE'
      EXPORTING
        wi_test_modus       = abap_false
        wi_tadir_pgmid      = 'R3TR'
        wi_tadir_object     = lv_object
        wi_tadir_obj_name   = lv_name
        wi_tadir_author     = sy-uname
        wi_tadir_devclass   = p_pk
        wi_tadir_masterlang = sy-langu
      EXCEPTIONS
        OTHERS              = 1.
    IF sy-subrc <> 0.
      add_msg( iv_type = 'E'
               iv_text = |{ lv_object } { lv_name } - TADIR ENTRY NOT CREATED| ).
    ENDIF.
  ENDMETHOD.

  METHOD rfc_read.
    DATA: lt_fields  TYPE STANDARD TABLE OF rfc_db_fld,
          lt_data    TYPE STANDARD TABLE OF tab512,
          lt_options TYPE STANDARD TABLE OF rfc_db_opt,
          lv_table   TYPE tabname,
          lv_msg     TYPE c LENGTH 200,
          lr_line    TYPE REF TO data.
    FIELD-SYMBOLS: <ls_line> TYPE any,
                   <lv_comp> TYPE any.

    CLEAR: et_data, ev_subrc.
    lv_table   = iv_table.
    lt_options = it_where.
    LOOP AT it_fields INTO DATA(lv_fieldname).
      APPEND VALUE #( fieldname = lv_fieldname ) TO lt_fields.
    ENDLOOP.

    CALL FUNCTION 'RFC_READ_TABLE' DESTINATION p_rd
      EXPORTING
        query_table           = lv_table
      TABLES
        options               = lt_options
        fields                = lt_fields
        data                  = lt_data
      EXCEPTIONS
        table_not_available   = 1
        table_without_data    = 2
        option_not_valid      = 3
        field_not_valid       = 4
        not_authorized        = 5
        data_buffer_exceeded  = 6
        system_failure        = 7 MESSAGE lv_msg
        communication_failure = 8 MESSAGE lv_msg
        OTHERS                = 9.
    ev_subrc = sy-subrc.
    IF ev_subrc = 2.
      RETURN.          " NO DATA - NOT AN ERROR
    ELSEIF ev_subrc <> 0.
      IF iv_quiet = abap_true.
        RETURN.
      ENDIF.
      add_msg( iv_type = 'E'
               iv_text = |RFC_READ_TABLE { lv_table } FAILED (RC { ev_subrc }) { lv_msg }| ).
      RETURN.
    ENDIF.

    " MAP EACH RETURNED LINE INTO THE TARGET STRUCTURE USING THE FIELD
    " OFFSETS / LENGTHS RETURNED BY RFC_READ_TABLE (NO HARD-CODED OFFSETS)
    CREATE DATA lr_line LIKE LINE OF et_data.
    ASSIGN lr_line->* TO <ls_line>.
    LOOP AT lt_data INTO DATA(ls_data).
      CLEAR <ls_line>.
      LOOP AT lt_fields INTO DATA(ls_field).
        ASSIGN COMPONENT ls_field-fieldname OF STRUCTURE <ls_line> TO <lv_comp>.
        IF sy-subrc = 0.
          DATA(lv_off) = CONV i( ls_field-offset ).
          DATA(lv_len) = CONV i( ls_field-length ).
          <lv_comp> = ls_data-wa+lv_off(lv_len).
        ENDIF.
      ENDLOOP.
      APPEND <ls_line> TO et_data.
    ENDLOOP.
  ENDMETHOD.

ENDCLASS.

*&---------------------------------------------------------------------*
*& FORM GET_TR_OBJECTS
*&---------------------------------------------------------------------*
*& READ E071 OF THE REQUEST AND ALL ITS TASKS FROM THE SOURCE SYSTEM.
*& (OBJECTS OF AN UNRELEASED REQUEST ARE STORED UNDER ITS TASKS)
*&---------------------------------------------------------------------*
FORM get_tr_objects.
  TYPES: BEGIN OF ty_e070,
           trkorr  TYPE e070-trkorr,
           strkorr TYPE e070-strkorr,
         END OF ty_e070.
  DATA: lt_e070    TYPE STANDARD TABLE OF ty_e070,
        lt_e071    TYPE STANDARD TABLE OF ty_e071,
        lt_tr      TYPE STANDARD TABLE OF trkorr WITH EMPTY KEY,
        lt_options TYPE tt_options,
        lv_subrc   TYPE sy-subrc.

  " STEP 1: REQUEST + ITS TASKS
  lcl_util=>rfc_read(
    EXPORTING iv_table  = 'E070'
              it_where  = VALUE #( ( text = |STRKORR = '{ p_tr }'| ) )
              it_fields = VALUE #( ( 'TRKORR' ) ( 'STRKORR' ) )
    IMPORTING et_data   = lt_e070
              ev_subrc  = lv_subrc ).

  APPEND p_tr TO lt_tr.
  LOOP AT lt_e070 INTO DATA(ls_e070).
    APPEND ls_e070-trkorr TO lt_tr.
  ENDLOOP.

  gt_trkorrs = lt_tr.

  " STEP 2: OBJECT LIST OF ALL OF THEM
  LOOP AT lt_tr INTO DATA(lv_tr).
    IF sy-tabix = 1.
      APPEND VALUE #( text = |TRKORR = '{ lv_tr }'| ) TO lt_options.
    ELSE.
      APPEND VALUE #( text = |OR TRKORR = '{ lv_tr }'| ) TO lt_options.
    ENDIF.
  ENDLOOP.

  lcl_util=>rfc_read(
    EXPORTING iv_table  = 'E071'
              it_where  = lt_options
              it_fields = VALUE #( ( 'TRKORR' ) ( 'AS4POS' ) ( 'PGMID' ) ( 'OBJECT' )
                                   ( 'OBJ_NAME' ) ( 'OBJFUNC' ) ( 'LOCKFLAG' )
                                   ( 'GENNUM' ) ( 'LANG' ) )
    IMPORTING et_data   = lt_e071
              ev_subrc  = lv_subrc ).

  IF t_return IS NOT INITIAL.
    PERFORM display_return.
    CLEAR t_return.
  ENDIF.

  " STEP 3: REMOVE REQUEST COMMENTS AND DUPLICATES (SAME OBJECT IN SEVERAL TASKS)
  DELETE lt_e071 WHERE pgmid = 'CORR'.
  SORT lt_e071 BY pgmid object obj_name.
  DELETE ADJACENT DUPLICATES FROM lt_e071 COMPARING pgmid object obj_name.
  SORT lt_e071 BY object obj_name.

  it_data = lt_e071.
ENDFORM.

*&---------------------------------------------------------------------*
*& FORM CHECK_LOCAL_STATUS
*&---------------------------------------------------------------------*
*& OBJECT EXISTS IN THIS SYSTEM? / CAN THIS REPORT COPY IT?
*&---------------------------------------------------------------------*
FORM check_local_status.
  DATA: ls_tadir  TYPE tadir,
        lv_exists TYPE flag.

  LOOP AT it_data ASSIGNING FIELD-SYMBOL(<fs_data>).
    CLEAR lv_exists.
    CALL FUNCTION 'TR_CHECK_EXIST'
      EXPORTING
        iv_pgmid             = <fs_data>-pgmid
        iv_object            = <fs_data>-object
        iv_obj_name          = <fs_data>-obj_name
        is_tadir             = ls_tadir
      IMPORTING
        e_exist              = lv_exists
      EXCEPTIONS
        tr_no_check_function = 1
        OTHERS               = 2.

    IF lv_exists = 'X'.
      <fs_data>-activity = 'AVAILABLE'.
    ELSE.
      <fs_data>-activity = 'NOT AVAILABLE'.
    ENDIF.

    CASE <fs_data>-object.
      WHEN 'DOMA' OR 'DOMD' OR 'DTEL' OR 'DTED' OR 'TABL' OR 'TABD' OR 'TABT'
        OR 'PROG' OR 'REPS' OR 'REPT' OR 'INCL' OR 'FUGR' OR 'FUNC' OR 'TRAN'
        OR 'TTYP' OR 'TTYD' OR 'VIEW' OR 'VIED' OR 'SHLP' OR 'SHLD' OR 'ENQU' OR 'ENQD'
        OR 'TYPE' OR 'TYPD' OR 'MSAG' OR 'MESS' OR 'PARA' OR 'NROB'
        OR 'INTF' OR 'INTD' OR 'CLAS' OR 'CLSD' OR 'CPUB' OR 'CPRO' OR 'CPRI' OR 'CINC' OR 'METH'
        OR 'DYNP' OR 'CUAD' OR 'TOBJ' OR 'VCLS' OR 'SXCI' OR 'TABU'
        OR 'SUSO' OR 'AUTH'.
        <fs_data>-support = 'SUPPORTED'.
      WHEN OTHERS.
        <fs_data>-support = 'NOT SUPPORTED'.
    ENDCASE.
  ENDLOOP.
ENDFORM.

*---------------------------------------------*
* USER COMMAND HANDLING
*---------------------------------------------*
FORM user_command USING r_ucomm     LIKE sy-ucomm
                        rs_selfield TYPE slis_selfield.

  CASE r_ucomm.
    WHEN 'COPY'.
      READ TABLE it_data WITH KEY sel = 'X' TRANSPORTING NO FIELDS.
      IF sy-subrc = 0.
        CLEAR: t_return, gt_objects, gt_ddic_act, gt_done_fm, gt_e071, gt_e071k, gt_deps.

        PERFORM process_selected.

        " REFRESH THE STATUS COLUMN AND SHOW THE LOG
        PERFORM check_local_status.
        rs_selfield-refresh = 'X'.
        PERFORM display_return.
      ELSE.
        MESSAGE 'NO ROWS SELECTED.' TYPE 'I'.
        RETURN.
      ENDIF.
    WHEN '&BACK' OR '&EXIT'.
      LEAVE PROGRAM.
  ENDCASE.

ENDFORM.

*---------------------------------------------*
* PROCESS SELECTED ROWS
*---------------------------------------------*
FORM process_selected.
  DATA: lt_names TYPE tt_objnames,
        lt_clas  TYPE tt_objnames,
        lv_name  TYPE trobj_name.

*--------------------------------------------------------------------*
* DICTIONARY: WRITE EVERYTHING INACTIVE, THEN ACTIVATE TOGETHER
*--------------------------------------------------------------------*
  " 1. DOMAINS
  PERFORM collect_names USING 'DOMA DOMD' CHANGING lt_names.
  LOOP AT lt_names INTO lv_name.
    lv_ddic = lv_name.
    PERFORM make_doma.
  ENDLOOP.

  " 2. DATA ELEMENTS
  PERFORM collect_names USING 'DTEL DTED' CHANGING lt_names.
  LOOP AT lt_names INTO lv_name.
    lv_ddic = lv_name.
    PERFORM make_detl.
  ENDLOOP.

  " 3. TABLES / STRUCTURES
  PERFORM collect_names USING 'TABL TABD TABT' CHANGING lt_names.
  LOOP AT lt_names INTO lv_name.
    lv_ddic = lv_name.
    PERFORM make_tabl.
  ENDLOOP.

  " 7. TABLE TYPES
  PERFORM collect_names USING 'TTYP TTYD' CHANGING lt_names.
  LOOP AT lt_names INTO lv_name.
    lv_ddic = lv_name.
    PERFORM make_ttyp.
  ENDLOOP.

  " 8. VIEWS
  PERFORM collect_names USING 'VIEW VIED' CHANGING lt_names.
  LOOP AT lt_names INTO lv_name.
    lv_ddic = lv_name.
    PERFORM make_view.
  ENDLOOP.

  " 9. SEARCH HELPS
  PERFORM collect_names USING 'SHLP SHLD' CHANGING lt_names.
  PERFORM merge_deps USING 'SHLP' CHANGING lt_names.
  LOOP AT lt_names INTO lv_name.
    lv_ddic = lv_name.
    PERFORM make_shlp.
  ENDLOOP.

  " 10. LOCK OBJECTS
  PERFORM collect_names USING 'ENQU ENQD' CHANGING lt_names.
  LOOP AT lt_names INTO lv_name.
    lv_ddic = lv_name.
    PERFORM make_enqu.
  ENDLOOP.

  PERFORM activate_ddic.

  " 11. TYPE GROUPS
  PERFORM collect_names USING 'TYPE TYPD' CHANGING lt_names.
  LOOP AT lt_names INTO lv_name.
    PERFORM make_type USING lv_name.
  ENDLOOP.

*--------------------------------------------------------------------*
* TEXTS / SETTINGS
*--------------------------------------------------------------------*
  " 12. MESSAGE CLASSES AND SINGLE MESSAGES (LIMU MESS = CLASS(20)+NO(3))
  PERFORM collect_names USING 'MSAG' CHANGING lt_names.
  LOOP AT lt_names INTO lv_name.
    PERFORM make_msag USING lv_name space.
  ENDLOOP.
  PERFORM collect_names USING 'MESS' CHANGING lt_names.
  LOOP AT lt_names INTO lv_name.
    PERFORM make_msag USING lv_name(20) lv_name+20(3).
  ENDLOOP.

  " 13. SET/GET PARAMETERS
  PERFORM collect_names USING 'PARA' CHANGING lt_names.
  LOOP AT lt_names INTO lv_name.
    PERFORM make_para USING lv_name.
  ENDLOOP.

  " 14. NUMBER RANGE OBJECTS
  PERFORM collect_names USING 'NROB' CHANGING lt_names.
  LOOP AT lt_names INTO lv_name.
    PERFORM make_nrob USING lv_name.
  ENDLOOP.

*--------------------------------------------------------------------*
* 15. INTERFACES, THEN CLASSES (ANY PART OF A CLASS COPIES THE CLASS)
*--------------------------------------------------------------------*
  PERFORM collect_names USING 'INTF INTD' CHANGING lt_names.
  LOOP AT lt_names INTO lv_name.
    PERFORM make_intf USING lv_name.
  ENDLOOP.

  CLEAR lt_clas.
  LOOP AT it_data INTO DATA(wa_selected) WHERE sel = 'X'.
    CASE wa_selected-object.
      WHEN 'CLAS' OR 'CLSD' OR 'CPUB' OR 'CPRO' OR 'CPRI'.
        INSERT CONV trobj_name( wa_selected-obj_name ) INTO TABLE lt_clas.
      WHEN 'METH'.                          " CLASS(30) + METHOD
        INSERT CONV trobj_name( wa_selected-obj_name(30) ) INTO TABLE lt_clas.
      WHEN 'CINC'.                          " ZCL_X=====CCIMP
        SPLIT wa_selected-obj_name AT '=' INTO lv_name DATA(lv_rest).
        INSERT lv_name INTO TABLE lt_clas.
    ENDCASE.
  ENDLOOP.
  LOOP AT lt_clas INTO lv_name.
    PERFORM make_clas USING lv_name.
  ENDLOOP.

*--------------------------------------------------------------------*
* 4. PROGRAMS & INCLUDES (SCREENS + GUI STATUS COPIED WITH THEM)
*--------------------------------------------------------------------*
  PERFORM collect_names USING 'PROG REPS REPT INCL' CHANGING lt_names.
  LOOP AT lt_names INTO lv_name.
    lv_prog = lv_name.
    PERFORM make_prog.
    WAIT UP TO 2 SECONDS.
  ENDLOOP.

*--------------------------------------------------------------------*
* 5. FUNCTION GROUPS / SINGLE FUNCTION MODULES
*--------------------------------------------------------------------*
  PERFORM collect_names USING 'FUGR' CHANGING lt_names.
  PERFORM merge_deps USING 'FUGR' CHANGING lt_names.
  LOOP AT lt_names INTO lv_name.
    lv_fgname = lv_name.
    PERFORM make_fugr_complete.
  ENDLOOP.

  PERFORM collect_names USING 'FUNC' CHANGING lt_names.
  LOOP AT lt_names INTO lv_name.
    lv_fname = lv_name.
    PERFORM make_single_func.
  ENDLOOP.

*--------------------------------------------------------------------*
* 16. SINGLE SCREENS (LIMU DYNP = PROG(40)+DYNNR(4)) / GUI STATUS
*--------------------------------------------------------------------*
  PERFORM collect_names USING 'DYNP' CHANGING lt_names.

  PERFORM collect_names USING 'TRAN' CHANGING lt_names.

  PERFORM collect_names USING 'TABU' CHANGING lt_names.

  ENDLOOP.

*--------------------------------------------------------------------*

*--------------------------------------------------------------------*
  LOOP AT it_data INTO wa_selected WHERE sel = 'X' AND support = 'NOT SUPPORTED'.
    CASE wa_selected-object.

        lcl_util=>add_msg( iv_type = 'W'
                           iv_text = |{ wa_selected-object } { wa_selected-obj_name } - ENHANCEMENTS NEED THE ENHANCEMENT FRAMEWORK IN THE TARGET, RE-CREATE IN SE19/SE20| ).
      WHEN 'SMIM'.
        lcl_util=>add_msg( iv_type = 'W'

*&---------------------------------------------------------------------*
*& FORM ADD_DEP
*&---------------------------------------------------------------------*
*& REMEMBER A CUSTOMER OBJECT THAT A COPIED OBJECT NEEDS, IF IT DOES
*& NOT EXIST IN THIS SYSTEM YET (IT IS THEN COPIED IN ITS OWN STEP)
*&---------------------------------------------------------------------*
FORM add_dep USING iv_object TYPE csequence
                   iv_name   TYPE csequence.
  DATA: lv_object TYPE trobjtype,
        lv_name   TYPE trobj_name,
        lv_dir    TYPE tadir-obj_name.

  lv_object = iv_object.
  lv_name   = iv_name.
  IF lv_name IS INITIAL OR ( lv_name(1) <> 'Z' AND lv_name(1) <> 'Y' AND lv_name(1) <> '/' ).
    RETURN.
  ENDIF.
  lv_dir = lv_name.
  SELECT SINGLE obj_name FROM tadir INTO @DATA(lv_exists)
    WHERE pgmid = 'R3TR' AND object = @lv_object AND obj_name = @lv_dir.
  IF sy-subrc = 0.
    RETURN.                               " ALREADY IN THIS SYSTEM
  ENDIF.
  INSERT VALUE #( object = lv_object name = lv_name ) INTO TABLE gt_deps.
ENDFORM.

*&---------------------------------------------------------------------*
*& FORM MERGE_DEPS
*&---------------------------------------------------------------------*
*& ADD THE AUTOMATIC DEPENDENCIES OF ONE TYPE TO THE NAMES TO COPY
*&---------------------------------------------------------------------*
FORM merge_deps USING    iv_object TYPE csequence
                CHANGING ct_names  TYPE tt_objnames.
  LOOP AT gt_deps INTO DATA(ls_dep) WHERE object = iv_object.
    INSERT ls_dep-name INTO TABLE ct_names.
    IF sy-subrc = 0.
      lcl_util=>add_msg( iv_type = 'I'
                         iv_text = |{ ls_dep-object } { ls_dep-name } - NOT IN THE REQUEST, COPIED AS DEPENDENCY| ).
    ENDIF.
  ENDLOOP.
ENDFORM.

*&---------------------------------------------------------------------*
*& FORM COLLECT_NAMES
*&---------------------------------------------------------------------*
*& UNIQUE NAMES OF THE SELECTED ROWS WITH ONE OF THE GIVEN OBJECT TYPES
*& (IV_TYPES = SPACE SEPARATED LIST, E.G. 'TABL TABD TABT')
*&---------------------------------------------------------------------*
FORM collect_names USING    iv_types TYPE csequence
                   CHANGING ct_names TYPE tt_objnames.
  DATA lt_types TYPE STANDARD TABLE OF string WITH EMPTY KEY.

  CLEAR ct_names.
  SPLIT iv_types AT space INTO TABLE lt_types.
  LOOP AT it_data INTO DATA(ls_data) WHERE sel = 'X'.
    IF line_exists( lt_types[ table_line = CONV string( ls_data-object ) ] ).
      INSERT CONV trobj_name( ls_data-obj_name ) INTO TABLE ct_names.
    ENDIF.
  ENDLOOP.
ENDFORM.

*& FORM MAKE_DOMA
*&---------------------------------------------------------------------*
*& 1. DOMAIN: READ FROM SOURCE, WRITE INACTIVE, QUEUE FOR ACTIVATION
*&---------------------------------------------------------------------*
FORM make_doma.
  DATA: ls_dd01v_a  TYPE dd01v,
        ls_dd01v_n  TYPE dd01v,
        lt_dd07v_a  TYPE STANDARD TABLE OF dd07v,
        lt_dd07v_n  TYPE STANDARD TABLE OF dd07v,
        ls_getstate TYPE dcdomaget,
        lv_msg      TYPE c LENGTH 200.

  " STEP 1: GET DOMAIN DEFINITION + FIXED VALUES FROM SOURCE SYSTEM
  ls_getstate-doma = 'M'.

  CALL FUNCTION 'DD_DOMA_GET' DESTINATION p_rd
    EXPORTING
      domain_name           = lv_ddic
      get_state             = ls_getstate
      langu                 = sy-langu
      prid                  = 0
      withtext              = 'X'
    IMPORTING
      dd01v_wa_a            = ls_dd01v_a
      dd01v_wa_n            = ls_dd01v_n
    TABLES
      dd07v_tab_a           = lt_dd07v_a
      dd07v_tab_n           = lt_dd07v_n
    EXCEPTIONS
      illegal_value         = 1
      op_failure            = 2
      system_failure        = 3 MESSAGE lv_msg
      communication_failure = 4 MESSAGE lv_msg
      OTHERS                = 5.
  IF sy-subrc <> 0.
    lcl_util=>add_msg( iv_type = 'E' iv_text = |DOMA { lv_ddic } - READ FROM SOURCE FAILED { lv_msg }| ).
    RETURN.
  ENDIF.

  " ONLY AN INACTIVE VERSION IN THE SOURCE -> USE IT
  IF ls_dd01v_a-domname IS INITIAL.
    ls_dd01v_a = ls_dd01v_n.
    lt_dd07v_a = lt_dd07v_n.
  ENDIF.
  IF ls_dd01v_a-domname IS INITIAL.
    lcl_util=>add_msg( iv_type = 'E' iv_text = |DOMA { lv_ddic } - NOT FOUND IN SOURCE SYSTEM| ).
    RETURN.
  ENDIF.

  " STEP 2: CREATE / UPDATE DOMAIN (INACTIVE)
  CALL FUNCTION 'DD_DOMA_PUT'
    EXPORTING
      dd01v_wa            = ls_dd01v_a
      domain_name         = lv_ddic
    TABLES
      dd07v_tab           = lt_dd07v_a
    EXCEPTIONS
      illegal_value       = 1
      op_failure          = 2
      object_inconsistent = 3
      OTHERS              = 4.
  IF sy-subrc <> 0.
    lcl_util=>add_msg( iv_type = 'E' iv_text = |DOMA { lv_ddic } - DOMAIN PUT FAILURE| ).
    RETURN.
  ENDIF.
  COMMIT WORK.

  " STEP 3: PACKAGE + TRANSPORT + ACTIVATION QUEUE
  lcl_util=>create_tadir( iv_object = 'DOMA' iv_name = lv_ddic ).
  lcl_util=>add_tr_obj( iv_pgmid = 'R3TR' iv_object = 'DOMA' iv_name = lv_ddic ).

*&---------------------------------------------------------------------*
*& FORM ACTIVATE_DDIC
*&---------------------------------------------------------------------*
*& ACTIVATE IN DEPENDENCY ORDER:
*&   DOMA -> DTEL -> TABL + TTYP (TWO PASSES, THEY CAN REFER TO EACH
*&   OTHER) -> VIEW -> SHLP -> ENQU
*&---------------------------------------------------------------------*
FORM activate_ddic.
  DATA: lt_retry TYPE STANDARD TABLE OF ty_act,
        lv_ok    TYPE abap_bool.

  LOOP AT gt_ddic_act INTO DATA(ls_act) WHERE object = 'DOMA'.
    PERFORM activate_one USING ls_act abap_true CHANGING lv_ok.
  ENDLOOP.
  LOOP AT gt_ddic_act INTO ls_act WHERE object = 'DTEL'.
    PERFORM activate_one USING ls_act abap_true CHANGING lv_ok.
  ENDLOOP.

  " FIRST PASS: FAILURES ARE ONLY COLLECTED
  LOOP AT gt_ddic_act INTO ls_act WHERE object = 'TABL' OR object = 'TTYP'.
    PERFORM activate_one USING ls_act abap_false CHANGING lv_ok.
    IF lv_ok = abap_false.
      APPEND ls_act TO lt_retry.
    ENDIF.
  ENDLOOP.
  " SECOND PASS: NOW REPORTED
  LOOP AT lt_retry INTO ls_act.
    PERFORM activate_one USING ls_act abap_true CHANGING lv_ok.
  ENDLOOP.

  LOOP AT gt_ddic_act INTO ls_act WHERE object = 'VIEW'.
    PERFORM activate_one USING ls_act abap_true CHANGING lv_ok.
  ENDLOOP.
  LOOP AT gt_ddic_act INTO ls_act WHERE object = 'SHLP'.
    PERFORM activate_one USING ls_act abap_true CHANGING lv_ok.
  ENDLOOP.
  LOOP AT gt_ddic_act INTO ls_act WHERE object = 'ENQU'.
    PERFORM activate_one USING ls_act abap_true CHANGING lv_ok.
  ENDLOOP.

  COMMIT WORK.
ENDFORM.

*&---------------------------------------------------------------------*
*& FORM ACTIVATE_ONE
*&---------------------------------------------------------------------*
FORM activate_one USING    is_act    TYPE ty_act
                           iv_report TYPE abap_bool
                  CHANGING cv_ok     TYPE abap_bool.
  DATA lv_rc TYPE sy-subrc.

  CASE is_act-object.
    WHEN 'DOMA'.
      CALL FUNCTION 'DDIF_DOMA_ACTIVATE'
        EXPORTING name = is_act-name IMPORTING rc = lv_rc
        EXCEPTIONS not_found = 1 put_failure = 2 OTHERS = 3.
    WHEN 'DTEL'.
      CALL FUNCTION 'DDIF_DTEL_ACTIVATE'
        EXPORTING name = is_act-name IMPORTING rc = lv_rc
        EXCEPTIONS not_found = 1 put_failure = 2 OTHERS = 3.
    WHEN 'TABL'.
      CALL FUNCTION 'DDIF_TABL_ACTIVATE'
        EXPORTING name = is_act-name IMPORTING rc = lv_rc
        EXCEPTIONS not_found = 1 put_failure = 2 OTHERS = 3.
    WHEN 'TTYP'.
      CALL FUNCTION 'DDIF_TTYP_ACTIVATE'
        EXPORTING name = is_act-name IMPORTING rc = lv_rc
        EXCEPTIONS not_found = 1 put_failure = 2 OTHERS = 3.
    WHEN 'VIEW'.
      CALL FUNCTION 'DDIF_VIEW_ACTIVATE'
        EXPORTING name = is_act-name IMPORTING rc = lv_rc
        EXCEPTIONS not_found = 1 put_failure = 2 OTHERS = 3.
    WHEN 'SHLP'.
      CALL FUNCTION 'DDIF_SHLP_ACTIVATE'
        EXPORTING name = is_act-name IMPORTING rc = lv_rc
        EXCEPTIONS not_found = 1 put_failure = 2 OTHERS = 3.
    WHEN 'ENQU'.
      CALL FUNCTION 'DDIF_ENQU_ACTIVATE'
        EXPORTING name = is_act-name IMPORTING rc = lv_rc
        EXCEPTIONS not_found = 1 put_failure = 2 OTHERS = 3.
  ENDCASE.

  IF sy-subrc <> 0 OR lv_rc > 4.
    cv_ok = abap_false.
    IF iv_report = abap_true.
      lcl_util=>add_msg( iv_type = 'E' iv_text = |{ is_act-object } { is_act-name } - ACTIVATION FAILED (RC { lv_rc }), CHECK SE11 ACTIVATION LOG| ).
    ENDIF.
  ELSE.
    cv_ok = abap_true.
    lcl_util=>add_msg( iv_type = 'S' iv_text = |{ is_act-object } { is_act-name } - CREATED AND ACTIVATED| ).
  ENDIF.
ENDFORM.

*&---------------------------------------------------------------------*

      with_includelist      = ' '
      only_source           = ' '
      only_texts            = ' '
      read_latest_version   = space
      with_lowercase        = 'X'          " KEEP ORIGINAL UPPER/LOWER CASE
    IMPORTING
      prog_inf              = ls_prog_inf
    TABLES
      include_tab           = lt_include_tab
      source                = lt_source
      source_extended       = lt_source_extended
      textelements          = lt_textelements
    EXCEPTIONS
      cancelled             = 1
      not_found             = 2
      permission_error      = 3
      system_failure        = 4 MESSAGE lv_msg
      communication_failure = 5 MESSAGE lv_msg
      OTHERS                = 6.
  IF sy-subrc <> 0.
    lcl_util=>add_msg( iv_type = 'E' iv_text = |PROG { lv_prog } - READ FROM SOURCE FAILED { lv_msg }| ).
    RETURN.
  ENDIF.

  lv_subc = ls_prog_inf-prog_type.
  " PROGRAM TITLE IS TEXT ELEMENT 'R'
  READ TABLE lt_textelements INTO DATA(ls_title) WITH KEY id = 'R'.
  IF sy-subrc = 0.
    lv_title = ls_title-entry.
  ENDIF.

  " STEP 2: CREATE, OR UPDATE WHEN IT ALREADY EXISTS
  SELECT SINGLE name FROM trdir INTO @DATA(lv_exists) WHERE name = @lv_prog.
  IF sy-subrc <> 0.

    CALL FUNCTION 'RPY_PROGRAM_INSERT'
      EXPORTING
        development_class = p_pk
        program_name      = lv_prog
        program_type      = lv_subc
        title_string      = lv_title
        transport_number  = p_ltr
        save_inactive     = space
        suppress_dialog   = 'X'
        uccheck           = 'X'
      TABLES
        source_extended   = lt_source_extended
      EXCEPTIONS
        already_exists    = 1
        cancelled         = 2
        name_not_allowed  = 3
        permission_error  = 4
        OTHERS            = 5.
    IF sy-subrc = 3.
      " NAMES RPY_PROGRAM_INSERT DOES NOT ACCEPT (E.G. FUNCTION GROUP
      " INCLUDES L<FG>F01) ARE WRITTEN DIRECTLY
      INSERT REPORT lv_prog FROM lt_source_extended PROGRAM TYPE lv_subc.
      IF sy-subrc <> 0.
        lcl_util=>add_msg( iv_type = 'E' iv_text = |PROG { lv_prog } - INSERT REPORT FAILED| ).
        RETURN.
      ENDIF.
    ELSEIF sy-subrc <> 0.
      lcl_util=>add_msg( iv_type = 'E' iv_text = |PROG { lv_prog } - REPORT PUT FAILURE| ).
      RETURN.
    ENDIF.
    lcl_util=>add_msg( iv_type = 'S' iv_text = |PROG { lv_prog } - CREATED| ).

  ELSE.

    CALL FUNCTION 'RPY_INCLUDE_UPDATE'
      EXPORTING
        include_name     = lv_prog
        title_string     = lv_title
        save_inactive    = space
      TABLES
        source_extended  = lt_source_extended
      EXCEPTIONS
        not_found        = 1
        cancelled        = 2
        permission_error = 3
        OTHERS           = 4.
    IF sy-subrc <> 0.
      lcl_util=>add_msg( iv_type = 'E' iv_text = |PROG { lv_prog } - REPORT UPDATE FAILURE| ).
      RETURN.
    ENDIF.
    lcl_util=>add_msg( iv_type = 'S' iv_text = |PROG { lv_prog } - UPDATED| ).

  ENDIF.

  " STEP 3: TEXT ELEMENTS (SELECTION TEXTS, TEXT SYMBOLS, TITLE)
  IF lt_textelements IS NOT INITIAL.
    INSERT TEXTPOOL lv_prog FROM lt_textelements LANGUAGE sy-langu.
    IF sy-subrc <> 0.
      lcl_util=>add_msg( iv_type = 'W' iv_text = |PROG { lv_prog } - TEXT ELEMENTS NOT SAVED| ).
    ELSE.
      lcl_util=>add_tr_obj( iv_pgmid = 'LIMU' iv_object = 'REPT' iv_name = lv_prog ).
    ENDIF.
  ENDIF.

  lcl_util=>add_tr_obj( iv_pgmid = 'LIMU' iv_object = 'REPS' iv_name = lv_prog ).
  COMMIT WORK.

  " STEP 4: SCREENS AND GUI STATUS (REPORT, MODULE POOL, FUNCTION GROUP)
  IF lv_subc CA '1MF'.
    DATA(lv_main) = lv_prog.
    PERFORM copy_screens_and_status USING lv_main.
  ENDIF.
ENDFORM.

*&---------------------------------------------------------------------*
*& FORM SET_FG_NAMES
*&---------------------------------------------------------------------*
*& DERIVE MAIN PROGRAM (SAPL<FG>) AND INCLUDE PREFIX (L<FG>) OF
*& LV_FGNAME, INCLUDING NAMESPACES (/NS/FG -> /NS/SAPLFG, /NS/LFG)
*&---------------------------------------------------------------------*
FORM set_fg_names.
  DATA: lv_fg TYPE string,
        lv_ns TYPE string,
        lv_gr TYPE string.

  lv_fg = condense( CONV string( lv_fgname ) ).
  IF lv_fg CP '/*/*'.
    FIND REGEX '^(/[^/]+/)(.+)$' IN lv_fg SUBMATCHES lv_ns lv_gr.
  ELSE.
    lv_gr = lv_fg.
  ENDIF.
  lv_mprog    = |{ lv_ns }SAPL{ lv_gr }|.
  lv_fgprefix = |{ lv_ns }L{ lv_gr }|.
  gv_fg_group = lv_gr.
  gv_fg_ns    = lv_ns.
ENDFORM.

*&---------------------------------------------------------------------*
*& FORM MAKE_FUGR_COMPLETE
*&---------------------------------------------------------------------*
*& 5a. FUNCTION GROUP: GROUP -> INCLUDES (TOP, F01...) -> FUNCTION
*&     MODULES -> MAIN PROGRAM (SO THE INCLUDE LIST MATCHES THE SOURCE)
*&---------------------------------------------------------------------*
FORM make_fugr_complete.
  TYPES: BEGIN OF ty_tfdir,
           funcname TYPE tfdir-funcname,
         END OF ty_tfdir.
  DATA: lt_tfdir TYPE STANDARD TABLE OF ty_tfdir,
        lv_subrc TYPE sy-subrc.

  PERFORM set_fg_names.

  " STEP 1: FUNCTION GROUP
  PERFORM make_functiongroup.
  IF gv_fg_ok = abap_false.
    RETURN.
  ENDIF.

  " STEP 2: INCLUDES OF THE GROUP (TOP, F01, O01, I01, ...)
  PERFORM make_includes.

  " STEP 3: ALL FUNCTION MODULES OF THE GROUP IN THE SOURCE SYSTEM
  lcl_util=>rfc_read(
    EXPORTING iv_table  = 'TFDIR'
              it_where  = VALUE #( ( text = |PNAME = '{ lv_mprog }'| ) )
              it_fields = VALUE #( ( 'FUNCNAME' ) )
    IMPORTING et_data   = lt_tfdir
              ev_subrc  = lv_subrc ).

  LOOP AT lt_tfdir INTO DATA(ls_tfdir).
    lv_fname = ls_tfdir-funcname.
    PERFORM make_func.
  ENDLOOP.

  " STEP 4: MAIN PROGRAM SAPL<FG> (CONTAINS THE INCLUDE STATEMENTS)
  lv_prog = lv_mprog.
  PERFORM make_prog.

  lcl_util=>add_tr_obj( iv_pgmid = 'R3TR' iv_object = 'FUGR' iv_name = lv_fgname ).
ENDFORM.

*&---------------------------------------------------------------------*
*& FORM MAKE_FUNCTIONGROUP
*&---------------------------------------------------------------------*
*& CREATE THE FUNCTION GROUP (IF IT DOES NOT EXIST) WITH ITS ORIGINAL
*& SHORT TEXT. GV_FG_OK = ABAP_FALSE ON RETURN = FAILED.
*&---------------------------------------------------------------------*
FORM make_functiongroup.
  TYPES: BEGIN OF ty_tlibt,
           spras TYPE tlibt-spras,
           area  TYPE tlibt-area,
           areat TYPE tlibt-areat,
         END OF ty_tlibt.
  DATA: lt_tlibt   TYPE STANDARD TABLE OF ty_tlibt,
        lv_stext   TYPE tftit-stext,
        lv_corrnum TYPE e071-trkorr,
        lv_subrc   TYPE sy-subrc.

  gv_fg_ok = abap_false.

  " ALREADY IN THIS SYSTEM?  (TLIBG IS THE FUNCTION GROUP DIRECTORY)
  SELECT SINGLE area FROM tlibg INTO @DATA(lv_area) WHERE area = @lv_fgname.
  IF sy-subrc = 0.
    lcl_util=>add_msg( iv_type = 'I' iv_text = |FUGR { lv_fgname } - ALREADY EXISTS, CONTENT WILL BE UPDATED| ).
    gv_fg_ok = abap_true.
    RETURN.
  ENDIF.

  " SHORT TEXT FROM SOURCE SYSTEM
  lcl_util=>rfc_read(
    EXPORTING iv_table  = 'TLIBT'
              it_where  = VALUE #( ( text = |AREA = '{ lv_fgname }'| ) )
              it_fields = VALUE #( ( 'SPRAS' ) ( 'AREA' ) ( 'AREAT' ) )
    IMPORTING et_data   = lt_tlibt
              ev_subrc  = lv_subrc ).
  READ TABLE lt_tlibt INTO DATA(ls_tlibt) WITH KEY spras = sy-langu.
  IF sy-subrc <> 0.
    READ TABLE lt_tlibt INTO ls_tlibt INDEX 1.
  ENDIF.
  lv_stext = ls_tlibt-areat.
  IF lv_stext IS INITIAL.
    lv_stext = lv_fgname.
  ENDIF.

  lv_corrnum = p_ltr.

  CALL FUNCTION 'RS_FUNCTION_POOL_INSERT'
    EXPORTING
      function_pool           = gv_fg_group
      short_text              = lv_stext
      namespace               = gv_fg_ns
      devclass                = p_pk
      corrnum                 = lv_corrnum
      suppress_corr_check     = space
    IMPORTING
      corrnum                 = lv_corrnum
    EXCEPTIONS
      name_already_exists     = 1
      name_not_correct        = 2
      function_already_exists = 3
      invalid_function_pool   = 4
      invalid_name            = 5
      too_many_functions      = 6
      no_modify_permission    = 7
      no_show_permission      = 8
      enqueue_system_failure  = 9
      canceled_in_corr        = 10
      undefined_error         = 11
      OTHERS                  = 12.
  CASE sy-subrc.
    WHEN 0 OR 1.
      lcl_util=>add_msg( iv_type = 'S' iv_text = |FUGR { lv_fgname } - CREATED| ).
      COMMIT WORK.
      gv_fg_ok = abap_true.
    WHEN OTHERS.
      lcl_util=>add_msg( iv_type = 'E' iv_text = |FUGR { lv_fgname } - FUNCTION GROUP PUT FAILURE| ).
  ENDCASE.
ENDFORM.

*&---------------------------------------------------------------------*
*& FORM MAKE_INCLUDES
*&---------------------------------------------------------------------*
*& COPY ALL OWN INCLUDES OF THE FUNCTION GROUP (TOP, F01, O01, ...).
*& SKIPPED: UXX AND U01.. (GENERATED BY THE FUNCTION BUILDER) AND
*& INCLUDES THAT DO NOT BELONG TO THIS GROUP (E.G. SAP STANDARD).
*&---------------------------------------------------------------------*
FORM make_includes.
  DATA: ls_prog_inf        TYPE rpy_prog,
        lt_include_tab     TYPE STANDARD TABLE OF rpy_repo,
        lt_source          TYPE STANDARD TABLE OF abapsource,
        lt_source_extended TYPE STANDARD TABLE OF abaptxt255,
        lt_textelements    TYPE STANDARD TABLE OF textpool,
        lv_suffix          TYPE string,
        lv_plen            TYPE i,
        lv_msg             TYPE c LENGTH 200.

  CALL FUNCTION 'RPY_PROGRAM_READ' DESTINATION p_rd
    EXPORTING
      language              = sy-langu
      program_name          = lv_mprog
      with_includelist      = 'X'
      only_source           = ' '
      only_texts            = ' '
      read_latest_version   = space
      with_lowercase        = 'X'
    IMPORTING
      prog_inf              = ls_prog_inf
    TABLES
      include_tab           = lt_include_tab
      source                = lt_source
      source_extended       = lt_source_extended
      textelements          = lt_textelements
    EXCEPTIONS
      cancelled             = 1
      not_found             = 2
      permission_error      = 3
      system_failure        = 4 MESSAGE lv_msg
      communication_failure = 5 MESSAGE lv_msg
      OTHERS                = 6.
  IF sy-subrc <> 0.
    lcl_util=>add_msg( iv_type = 'E' iv_text = |FUGR { lv_fgname } - READ OF { lv_mprog } FROM SOURCE FAILED { lv_msg }| ).
    RETURN.
  ENDIF.

  lv_plen = strlen( lv_fgprefix ).
  LOOP AT lt_include_tab INTO DATA(wa_include_tab).
    IF wa_include_tab-inclname NP |{ lv_fgprefix }*|.
      CONTINUE.                 " NOT AN INCLUDE OF THIS GROUP
    ENDIF.
    lv_suffix = wa_include_tab-inclname+lv_plen.
    IF lv_suffix CP 'U++'.
      CONTINUE.                 " UXX / FUNCTION MODULE INCLUDES
    ENDIF.
    lv_prog = wa_include_tab-inclname.
    PERFORM make_prog.
  ENDLOOP.
ENDFORM.

*&---------------------------------------------------------------------*
*& FORM MAKE_SINGLE_FUNC
*&---------------------------------------------------------------------*
*& 5b. SINGLE FUNCTION MODULE (LIMU FUNC). THE FUNCTION GROUP IS
*&     DETERMINED IN THE SOURCE SYSTEM AND CREATED FIRST IF MISSING.
*&---------------------------------------------------------------------*
FORM make_single_func.
  TYPES: BEGIN OF ty_tfdir,
           funcname TYPE tfdir-funcname,
           pname    TYPE tfdir-pname,
         END OF ty_tfdir.
  DATA: lt_tfdir TYPE STANDARD TABLE OF ty_tfdir,
        lv_subrc TYPE sy-subrc,
        lv_pname TYPE string.

  " ALREADY COPIED AS PART OF A SELECTED FUNCTION GROUP
  READ TABLE gt_done_fm WITH TABLE KEY table_line = lv_fname TRANSPORTING NO FIELDS.
  IF sy-subrc = 0.
    RETURN.
  ENDIF.

  lcl_util=>rfc_read(
    EXPORTING iv_table  = 'TFDIR'
              it_where  = VALUE #( ( text = |FUNCNAME = '{ lv_fname }'| ) )
              it_fields = VALUE #( ( 'FUNCNAME' ) ( 'PNAME' ) )
    IMPORTING et_data   = lt_tfdir
              ev_subrc  = lv_subrc ).
  READ TABLE lt_tfdir INTO DATA(ls_tfdir) INDEX 1.
  IF sy-subrc <> 0.
    lcl_util=>add_msg( iv_type = 'E' iv_text = |FUNC { lv_fname } - NOT FOUND IN SOURCE SYSTEM| ).
    RETURN.
  ENDIF.

  " SAPLZFG -> ZFG ,  /NS/SAPLZFG -> /NS/ZFG
  lv_pname = condense( CONV string( ls_tfdir-pname ) ).
  REPLACE FIRST OCCURRENCE OF 'SAPL' IN lv_pname WITH ''.
  lv_fgname = lv_pname.
  PERFORM set_fg_names.

  SELECT SINGLE area FROM tlibg INTO @DATA(lv_area) WHERE area = @lv_fgname.
  IF sy-subrc <> 0.
    " GROUP MISSING: CREATE IT WITH ITS INCLUDES, THEN THE MODULE
    PERFORM make_functiongroup.
    IF gv_fg_ok = abap_false.
      RETURN.
    ENDIF.
    PERFORM make_includes.
    PERFORM make_func.
    lv_prog = lv_mprog.
    PERFORM make_prog.
    lcl_util=>add_tr_obj( iv_pgmid = 'R3TR' iv_object = 'FUGR' iv_name = lv_fgname ).
  ELSE.
    PERFORM make_func.
  ENDIF.
ENDFORM.

*&---------------------------------------------------------------------*
*& FORM MAKE_FUNC
*&---------------------------------------------------------------------*
*& CREATE ONE FUNCTION MODULE:
*&   1. READ INTERFACE + SOURCE FROM SOURCE SYSTEM (ACTIVE VERSION)
*&   2. NEW FM: CREATE WITH FULL INTERFACE (IMPORT/EXPORT/CHANGING/
*&      TABLES/EXCEPTIONS, RFC / UPDATE FLAGS, SHORT TEXT)
*&   3. WRITE THE ORIGINAL SOURCE (255 CHAR LINES) INTO THE FM INCLUDE
*&   EXISTING FM: ONLY THE SOURCE IS UPDATED
*&---------------------------------------------------------------------*
FORM make_func.
  DATA: lv_objname   TYPE vrsd-objname,
        lv_info_line TYPE abaptext-line,
        lt_vsmodisrc TYPE STANDARD TABLE OF smodisrc,
        lt_uincl_tab TYPE STANDARD TABLE OF abaptxt255,
        lt_dincl_tab TYPE STANDARD TABLE OF abaptext,
        lt_enlfd_tab TYPE STANDARD TABLE OF enlfdir,
        lt_funct_tab TYPE STANDARD TABLE OF funct,
        lt_fupar_tab TYPE STANDARD TABLE OF sfupararef,
        lt_vsmodilog TYPE STANDARD TABLE OF smodilog,
        lt_tfdir_tab TYPE STANDARD TABLE OF tfdir,
        lt_tftit_tab TYPE STANDARD TABLE OF tftit,
        lt_trdir_tab TYPE STANDARD TABLE OF trdir.

  DATA: lt_import     TYPE STANDARD TABLE OF rsimp,
        ls_import     TYPE rsimp,
        lt_export     TYPE STANDARD TABLE OF rsexp,
        ls_export     TYPE rsexp,
        lt_changing   TYPE STANDARD TABLE OF rscha,
        ls_changing   TYPE rscha,
        lt_tables     TYPE STANDARD TABLE OF rstbl,
        ls_tables     TYPE rstbl,
        lt_exceptions TYPE STANDARD TABLE OF rsexc,
        lt_source     TYPE STANDARD TABLE OF rssource,
        lv_short_text TYPE tftit-stext,
        lv_include    TYPE rs38l-include,
        lv_remote     TYPE rs38l-remote,
        lv_utask      TYPE rs38l-utask.

  INSERT lv_fname INTO TABLE gt_done_fm.
  lv_objname = lv_fname.

  " STEP 1: READ ACTIVE VERSION OF THE FM FROM THE SOURCE SYSTEM
  CALL FUNCTION 'SVRS_GET_VERSION_FUNC_40'
    EXPORTING
      destination           = p_rd
      object_name           = lv_objname
      versno                = '00000'
    IMPORTING
      info_line             = lv_info_line
    TABLES
      vsmodisrc             = lt_vsmodisrc
      uincl_tab             = lt_uincl_tab      " FM SOURCE (L<FG>Uxx)
      dincl_tab             = lt_dincl_tab
      enlfd_tab             = lt_enlfd_tab
      funct_tab             = lt_funct_tab
      fupar_tab             = lt_fupar_tab      " FM INTERFACE
      vsmodilog             = lt_vsmodilog
      tfdir_tab             = lt_tfdir_tab      " RFC / UPDATE FLAGS
      tftit_tab             = lt_tftit_tab      " SHORT TEXT
      trdir_tab             = lt_trdir_tab
    EXCEPTIONS
      no_version            = 1
      system_failure        = 2
      communication_failure = 3
      OTHERS                = 4.
  IF sy-subrc <> 0 OR lt_uincl_tab IS INITIAL.
    lcl_util=>add_msg( iv_type = 'E' iv_text = |FUNC { lv_fname } - READ FROM SOURCE FAILED| ).
    RETURN.
  ENDIF.

  " FM ALREADY EXISTS HERE -> UPDATE SOURCE ONLY
  SELECT SINGLE pname, include FROM tfdir INTO @DATA(ls_local)
    WHERE funcname = @lv_fname.
  IF sy-subrc = 0.
    lv_include = |{ lv_fgprefix }U{ ls_local-include }|.
    IF ls_local-pname <> lv_mprog.
      " FM LIVES IN ANOTHER GROUP LOCALLY
      DATA(lv_local_prefix) = replace( val = CONV string( ls_local-pname ) sub = 'SAPL' with = 'L' ).
      lv_include = |{ lv_local_prefix }U{ ls_local-include }|.
    ENDIF.
    INSERT REPORT lv_include FROM lt_uincl_tab.
    IF sy-subrc <> 0.
      lcl_util=>add_msg( iv_type = 'E' iv_text = |FUNC { lv_fname } - SOURCE UPDATE FAILED ({ lv_include })| ).
    ELSE.
      lcl_util=>add_msg( iv_type = 'S' iv_text = |FUNC { lv_fname } - SOURCE UPDATED (INTERFACE KEPT AS IN THIS SYSTEM)| ).
      lcl_util=>add_tr_obj( iv_pgmid = 'LIMU' iv_object = 'FUNC' iv_name = lv_fname ).
    ENDIF.
    RETURN.
  ENDIF.

  " STEP 2: BUILD THE INTERFACE FROM FUPARAREF OF THE SOURCE SYSTEM
  SORT lt_fupar_tab BY paramtype pposition.
  LOOP AT lt_fupar_tab INTO DATA(ls_fupar).
    CASE ls_fupar-paramtype.
      WHEN 'I'.
        CLEAR ls_import.
        MOVE-CORRESPONDING ls_fupar TO ls_import.
        ls_import-types   = ls_fupar-type.
        ls_import-default = ls_fupar-defaultval.
        IF ls_fupar-type IS NOT INITIAL.
          ls_import-typ = ls_fupar-structure.
        ELSE.
          ls_import-dbfield = ls_fupar-structure.
        ENDIF.
        APPEND ls_import TO lt_import.

      WHEN 'E'.
        CLEAR ls_export.
        MOVE-CORRESPONDING ls_fupar TO ls_export.
        ls_export-types = ls_fupar-type.
        IF ls_fupar-type IS NOT INITIAL.
          ls_export-typ = ls_fupar-structure.
        ELSE.
          ls_export-dbfield = ls_fupar-structure.
        ENDIF.
        APPEND ls_export TO lt_export.

      WHEN 'C'.
        CLEAR ls_changing.
        MOVE-CORRESPONDING ls_fupar TO ls_changing.
        ls_changing-types   = ls_fupar-type.
        ls_changing-default = ls_fupar-defaultval.
        IF ls_fupar-type IS NOT INITIAL.
          ls_changing-typ = ls_fupar-structure.
        ELSE.
          ls_changing-dbfield = ls_fupar-structure.
        ENDIF.
        APPEND ls_changing TO lt_changing.

      WHEN 'T'.
        CLEAR ls_tables.
        MOVE-CORRESPONDING ls_fupar TO ls_tables.
        ls_tables-types = ls_fupar-type.
        IF ls_fupar-type IS NOT INITIAL.
          ls_tables-typ = ls_fupar-structure.
        ELSE.
          ls_tables-dbstruct = ls_fupar-structure.
        ENDIF.
        APPEND ls_tables TO lt_tables.

      WHEN 'X'.
        APPEND VALUE #( exception = ls_fupar-parameter ) TO lt_exceptions.
    ENDCASE.
  ENDLOOP.

  " SHORT TEXT (LOGON LANGUAGE, ELSE ANY)
  READ TABLE lt_tftit_tab INTO DATA(ls_tftit) WITH KEY spras = sy-langu.
  IF sy-subrc <> 0.
    READ TABLE lt_tftit_tab INTO ls_tftit INDEX 1.
  ENDIF.
  lv_short_text = ls_tftit-stext.

  " PROCESSING TYPE: 'R' = REMOTE ENABLED, UTASK = UPDATE MODULE
  READ TABLE lt_tfdir_tab INTO DATA(ls_tfdir) INDEX 1.
  IF ls_tfdir-fmode = 'R'.
    lv_remote = 'R'.
  ENDIF.
  lv_utask = ls_tfdir-utask.

  " STEP 3: CREATE THE FUNCTION MODULE (INTERFACE)
  CALL FUNCTION 'RS_FUNCTIONMODULE_INSERT'
    EXPORTING
      funcname                = lv_fname
      function_pool           = gv_fg_group    " FUNCTION GROUP MUST ALREADY EXIST
      namespace               = gv_fg_ns
      short_text              = lv_short_text
      remote_call             = lv_remote
      update_task             = lv_utask
      corrnum                 = p_ltr
      suppress_corr_check     = space
    IMPORTING
      function_include        = lv_include
    TABLES
      import_parameter        = lt_import
      export_parameter        = lt_export
      tables_parameter        = lt_tables
      changing_parameter      = lt_changing
      exception_list          = lt_exceptions
      source                  = lt_source
    EXCEPTIONS
      double_task             = 1
      error_message           = 2
      function_already_exists = 3
      invalid_function_pool   = 4
      invalid_name            = 5
      too_many_functions      = 6
      no_modify_permission    = 7
      no_show_permission      = 8
      enqueue_system_failure  = 9
      canceled_in_corr        = 10
      OTHERS                  = 11.
  IF sy-subrc <> 0.
    lcl_util=>add_msg( iv_type = 'E' iv_text = |FUNC { lv_fname } - FUNCTION PUT FAILURE| ).
    RETURN.
  ENDIF.

  " STEP 4: WRITE THE ORIGINAL SOURCE (FUNCTION ... ENDFUNCTION, FULL
  " 255 CHAR LINES) INTO THE GENERATED INCLUDE - SAME AS SE37 SAVE
  IF lv_include IS INITIAL.
    SELECT SINGLE include FROM tfdir INTO @DATA(lv_incno) WHERE funcname = @lv_fname.
    lv_include = |{ lv_fgprefix }U{ lv_incno }|.
  ENDIF.
  INSERT REPORT lv_include FROM lt_uincl_tab.
  IF sy-subrc <> 0.
    lcl_util=>add_msg( iv_type = 'E' iv_text = |FUNC { lv_fname } - CREATED, BUT SOURCE NOT WRITTEN ({ lv_include })| ).
  ELSE.
    lcl_util=>add_msg( iv_type = 'S' iv_text = |FUNC { lv_fname } - CREATED| ).
  ENDIF.

  lcl_util=>add_tr_obj( iv_pgmid = 'LIMU' iv_object = 'FUNC' iv_name = lv_fname ).
  COMMIT WORK.
ENDFORM.

*&---------------------------------------------------------------------*
*& FORM MAKE_TRAN
*&---------------------------------------------------------------------*
*& 6. TRANSACTION CODE: COPIES
*&    TSTC  - HEADER (PROGRAM, SCREEN, TYPE FLAGS)
*&    TSTCT - SHORT TEXTS (ALL LANGUAGES)
*&    TSTCP - PARAMETERS (PARAMETER / VARIANT / OO TRANSACTIONS)
*&    TSTCC - GUI SUPPORT FLAGS (SAPGUI FOR HTML / JAVA / WINDOWS)
*&    CREATES OR UPDATES, THEN TADIR + TRANSPORT ENTRY.
*&---------------------------------------------------------------------*
FORM make_tran.
  DATA: lt_tstc  TYPE STANDARD TABLE OF tstc,
        lt_tstct TYPE STANDARD TABLE OF tstct,
        lt_tstcp TYPE STANDARD TABLE OF tstcp,
        lt_tstcc TYPE STANDARD TABLE OF tstcc,
        lt_where TYPE tt_options,
        lv_subrc TYPE sy-subrc.

  lt_where = VALUE #( ( text = |TCODE = '{ lv_tran }'| ) ).

  " STEP 1: READ ALL TCODE TABLES FROM THE SOURCE SYSTEM
  lcl_util=>rfc_read( EXPORTING iv_table = 'TSTC'  it_where = lt_where
                      IMPORTING et_data  = lt_tstc  ev_subrc = lv_subrc ).
  IF lt_tstc IS INITIAL.
    lcl_util=>add_msg( iv_type = 'E' iv_text = |TRAN { lv_tran } - NOT FOUND IN SOURCE SYSTEM| ).
    RETURN.
  ENDIF.
  lcl_util=>rfc_read( EXPORTING iv_table = 'TSTCT' it_where = lt_where
                      IMPORTING et_data  = lt_tstct ev_subrc = lv_subrc ).
  lcl_util=>rfc_read( EXPORTING iv_table = 'TSTCP' it_where = lt_where
                      IMPORTING et_data  = lt_tstcp ev_subrc = lv_subrc ).
  lcl_util=>rfc_read( EXPORTING iv_table = 'TSTCC' it_where = lt_where
                      IMPORTING et_data  = lt_tstcc ev_subrc = lv_subrc ).

  READ TABLE lt_tstc INTO DATA(ls_tstc) INDEX 1.

  " WARN WHEN THE CALLED PROGRAM IS NOT (YET) IN THIS SYSTEM
  IF ls_tstc-pgmna IS NOT INITIAL.
    SELECT SINGLE name FROM trdir INTO @DATA(lv_pgm) WHERE name = @ls_tstc-pgmna.
    IF sy-subrc <> 0.
      lcl_util=>add_msg( iv_type = 'W' iv_text = |TRAN { lv_tran } - PROGRAM { ls_tstc-pgmna } DOES NOT EXIST IN THIS SYSTEM| ).
    ENDIF.
  ENDIF.

  SELECT SINGLE tcode FROM tstc INTO @DATA(lv_tcode) WHERE tcode = @lv_tran.
  DATA(lv_existed) = xsdbool( sy-subrc = 0 ).

  " STEP 2: WRITE LOCALLY (OLD ENTRIES REPLACED)
  CALL FUNCTION 'ENQUEUE_E_TABLE'
    EXPORTING
      tabname        = 'TSTC'
      varkey         = CONV rstable-varkey( lv_tran )
    EXCEPTIONS
      foreign_lock   = 1
      system_failure = 2
      OTHERS         = 3.
  IF sy-subrc <> 0.
    lcl_util=>add_msg( iv_type = 'E' iv_text = |TRAN { lv_tran } - LOCKED BY ANOTHER USER| ).
    RETURN.
  ENDIF.

  MODIFY tstc FROM TABLE lt_tstc.
  DELETE FROM tstct WHERE tcode = lv_tran.
  DELETE FROM tstcp WHERE tcode = lv_tran.
  DELETE FROM tstcc WHERE tcode = lv_tran.
  MODIFY tstct FROM TABLE lt_tstct.
  MODIFY tstcp FROM TABLE lt_tstcp.
  MODIFY tstcc FROM TABLE lt_tstcc.
  COMMIT WORK.

  CALL FUNCTION 'DEQUEUE_E_TABLE'
    EXPORTING
      tabname = 'TSTC'
      varkey  = CONV rstable-varkey( lv_tran ).

  " STEP 3: PACKAGE + TRANSPORT
  lcl_util=>create_tadir( iv_object = 'TRAN' iv_name = lv_tran ).
  lcl_util=>add_tr_obj( iv_pgmid = 'R3TR' iv_object = 'TRAN' iv_name = lv_tran ).

  IF lv_existed = abap_true.
    lcl_util=>add_msg( iv_type = 'S' iv_text = |TRAN { lv_tran } - UPDATED| ).
  ELSE.
    lcl_util=>add_msg( iv_type = 'S' iv_text = |TRAN { lv_tran } - CREATED| ).
  ENDIF.
ENDFORM.

*&---------------------------------------------------------------------*
*& FORM COPY_TABLES
*&---------------------------------------------------------------------*
*& GENERIC COPY OF CONTROL-TABLE ROWS FROM THE SOURCE SYSTEM:
*&   READ ROWS MATCHING IV_WHERE FROM EACH TABLE IN IV_TABLES (SPACE
*&   SEPARATED), DELETE THE LOCAL ROWS WITH THE SAME CONDITION, INSERT.
*&   IV_WHERE: CONDITIONS JOINED WITH ' AND ' (NO OTHER ' AND ' INSIDE).
*&   IV_QUIET: TABLES THAT MAY NOT EXIST IN EVERY RELEASE.
*&---------------------------------------------------------------------*
FORM copy_tables USING    iv_tables TYPE csequence
                          iv_where  TYPE csequence
                          iv_quiet  TYPE abap_bool
                 CHANGING cv_ok     TYPE abap_bool.
  DATA: lt_tabs    TYPE STANDARD TABLE OF string WITH EMPTY KEY,
        lt_parts   TYPE STANDARD TABLE OF string WITH EMPTY KEY,
        lt_options TYPE tt_options,
        lv_where   TYPE string,
        lv_tab     TYPE tabname,
        lv_subrc   TYPE sy-subrc,
        lr_tab     TYPE REF TO data.
  FIELD-SYMBOLS <lt_rows> TYPE STANDARD TABLE.

  cv_ok    = abap_true.
  lv_where = iv_where.

  " WHERE -> RFC_READ_TABLE OPTIONS (ONE CONDITION PER LINE, MAX 72)
  IF lv_where IS NOT INITIAL.
    SPLIT lv_where AT ' AND ' INTO TABLE lt_parts.
    LOOP AT lt_parts INTO DATA(lv_part).
      IF sy-tabix = 1.
        APPEND VALUE #( text = lv_part ) TO lt_options.
      ELSE.
        APPEND VALUE #( text = |AND { lv_part }| ) TO lt_options.
      ENDIF.
    ENDLOOP.
  ENDIF.

  SPLIT iv_tables AT space INTO TABLE lt_tabs.
  LOOP AT lt_tabs INTO DATA(lv_tabname).
    lv_tab = lv_tabname.

    TRY.
        CREATE DATA lr_tab TYPE STANDARD TABLE OF (lv_tab).
      CATCH cx_sy_create_data_error.
        IF iv_quiet = abap_false.
          lcl_util=>add_msg( iv_type = 'E' iv_text = |TABLE { lv_tab } DOES NOT EXIST IN THIS SYSTEM| ).
          cv_ok = abap_false.
        ENDIF.
        CONTINUE.
    ENDTRY.
    ASSIGN lr_tab->* TO <lt_rows>.

    lcl_util=>rfc_read( EXPORTING iv_table = lv_tab
                                  it_where = lt_options
                                  iv_quiet = iv_quiet
                        IMPORTING et_data  = <lt_rows>
                                  ev_subrc = lv_subrc ).
    IF lv_subrc <> 0 AND lv_subrc <> 2.
      IF iv_quiet = abap_false.
        cv_ok = abap_false.
      ENDIF.
      CONTINUE.
    ENDIF.

    " REPLACE: ROWS DELETED IN THE SOURCE ARE DELETED HERE TOO
    IF lv_where IS NOT INITIAL.
      DELETE FROM (lv_tab) WHERE (lv_where).
    ELSE.
      DELETE FROM (lv_tab).
    ENDIF.
    IF <lt_rows> IS NOT INITIAL.
      MODIFY (lv_tab) FROM TABLE <lt_rows>.
      IF sy-subrc <> 0.
        lcl_util=>add_msg( iv_type = 'E' iv_text = |TABLE { lv_tab } - ROWS NOT WRITTEN| ).
        cv_ok = abap_false.
      ENDIF.
    ENDIF.
  ENDLOOP.

  COMMIT WORK.
ENDFORM.

*&---------------------------------------------------------------------*
*& FORM MAKE_MSAG
*&---------------------------------------------------------------------*
*& 12. MESSAGE CLASS (IV_MSGNR INITIAL) OR ONE MESSAGE, ALL LANGUAGES
*&---------------------------------------------------------------------*
FORM make_msag USING iv_arbgb TYPE csequence
                     iv_msgnr TYPE csequence.
  DATA: lv_arbgb TYPE t100-arbgb,
        lv_msgnr TYPE t100-msgnr,
        lv_where TYPE string,
        lv_ok    TYPE abap_bool,
        lv_name  TYPE trobj_name.

  lv_arbgb = iv_arbgb.
  lv_msgnr = iv_msgnr.

  IF lv_msgnr IS INITIAL.
    lv_where = |ARBGB = '{ lv_arbgb }'|.
    PERFORM copy_tables USING 'T100A T100T T100 T100U' lv_where abap_false CHANGING lv_ok.
    IF lv_ok = abap_true.
      lcl_util=>create_tadir( iv_object = 'MSAG' iv_name = lv_arbgb ).
      lcl_util=>add_tr_obj( iv_pgmid = 'R3TR' iv_object = 'MSAG' iv_name = lv_arbgb ).
      lcl_util=>add_msg( iv_type = 'S' iv_text = |MSAG { lv_arbgb } - COPIED| ).
    ENDIF.
  ELSE.
    " MESSAGE CLASS MUST EXIST FOR A SINGLE MESSAGE
    SELECT SINGLE arbgb FROM t100a INTO @DATA(lv_exists) WHERE arbgb = @lv_arbgb.
    IF sy-subrc <> 0.
      PERFORM make_msag USING lv_arbgb space.
      RETURN.
    ENDIF.
    lv_where = |ARBGB = '{ lv_arbgb }' AND MSGNR = '{ lv_msgnr }'|.
    PERFORM copy_tables USING 'T100 T100U' lv_where abap_false CHANGING lv_ok.
    IF lv_ok = abap_true.
      lv_name = lv_arbgb.
      lv_name+20 = lv_msgnr.
      lcl_util=>add_tr_obj( iv_pgmid = 'LIMU' iv_object = 'MESS' iv_name = lv_name ).
      lcl_util=>add_msg( iv_type = 'S' iv_text = |MESS { lv_arbgb } { lv_msgnr } - COPIED| ).
    ENDIF.
  ENDIF.
ENDFORM.

*&---------------------------------------------------------------------*
*& FORM MAKE_TOBJ
*&---------------------------------------------------------------------*
*& 17. TABLE MAINTENANCE / TRANSPORT OBJECT (SE54 / SOBJ)
*&     E071 NAME = OBJECT NAME + OBJECT TYPE (LAST CHARACTER)
*&---------------------------------------------------------------------*
FORM make_tobj USING iv_name TYPE csequence.
  DATA: lv_full  TYPE string,
        lv_len   TYPE i,
        lv_obj   TYPE objh-objectname,
        lv_type  TYPE objh-objecttype,
        lv_where TYPE string,
        lv_ok    TYPE abap_bool,
        lv_ok2   TYPE abap_bool.

  lv_full = condense( CONV string( iv_name ) ).
  lv_len  = strlen( lv_full ) - 1.
  IF lv_len <= 0.
    RETURN.
  ENDIF.
  lv_obj  = condense( lv_full(lv_len) ).
  lv_type = lv_full+lv_len(1).

  lv_where = |OBJECTNAME = '{ lv_obj }' AND OBJECTTYPE = '{ lv_type }'|.
  PERFORM copy_tables USING 'OBJH OBJT OBJS' lv_where abap_false CHANGING lv_ok.
  PERFORM copy_tables USING 'OBJSL OBJM' lv_where abap_true CHANGING lv_ok2.

  " MAINTENANCE DIALOG SETTINGS (TABLE/VIEW)
  lv_where = |TABNAME = '{ lv_obj }'|.
  PERFORM copy_tables USING 'TDDAT TVDIR TVIMF' lv_where abap_true CHANGING lv_ok2.

  IF lv_ok = abap_true.
    lcl_util=>create_tadir( iv_object = 'TOBJ' iv_name = iv_name ).
    lcl_util=>add_tr_obj( iv_pgmid = 'R3TR' iv_object = 'TOBJ' iv_name = iv_name ).
    lcl_util=>add_msg( iv_type = 'S' iv_text = |TOBJ { lv_obj } ({ lv_type }) - COPIED| ).
  ENDIF.
ENDFORM.

  lt_methods    = CORRESPONDING #( lt_method_v ).

  lt_types      = CORRESPONDING #( lt_type_v ).
  lt_parameters = CORRESPONDING #( lt_param_v ).
  lt_exceps     = CORRESPONDING #( lt_excep_v ).
  lt_aliases    = CORRESPONDING #( lt_alias_v ).
  lt_typepusage = CORRESPONDING #( lt_typepls_v ).
  lt_friends    = CORRESPONDING #( lt_friends_v ).
  SORT lt_impls BY refclsname.
  DELETE ADJACENT DUPLICATES FROM lt_impls COMPARING refclsname.

  " STEP 3: SOURCE OF EVERY IMPLEMENTED METHOD (OWN, INTERFACE, REDEFINED)
  lcl_util=>rfc_read(
    EXPORTING iv_table  = 'TMDIR'
              it_where  = VALUE #( ( text = |CLASSNAME = '{ lv_class }'| ) )
              it_fields = VALUE #( ( 'METHODINDX' ) ( 'METHODNAME' ) )
    IMPORTING et_data   = lt_tmdir
              ev_subrc  = lv_subrc ).
  LOOP AT lt_tmdir INTO DATA(ls_tmdir) WHERE methodname IS NOT INITIAL AND methodindx <> 0.
    CLEAR: lt_meth_src, ls_msrc.
    lv_objname     = lv_class.
    lv_objname+30  = ls_tmdir-methodname.
    CALL FUNCTION 'SVRS_GET_VERSION_METH_40'
      EXPORTING
        destination           = p_rd
        object_name           = lv_objname
        versno                = '00000'
      IMPORTING
        info_line             = lv_info
      TABLES
        vsmodisrc             = lt_vsmodisrc
        repos_tab             = lt_meth_src
        vsmodilog             = lt_smodilog
        trdir_tab             = lt_trdir
      EXCEPTIONS
        no_version            = 1
        system_failure        = 2
        communication_failure = 3
        OTHERS                = 4.
    IF sy-subrc <> 0 OR lt_meth_src IS INITIAL.
      lcl_util=>add_msg( iv_type = 'W' iv_text = |CLAS { lv_class } - SOURCE OF { ls_tmdir-methodname } NOT READ| ).
      CONTINUE.
    ENDIF.
    ls_msrc-cpdname = ls_tmdir-methodname.
    IF line_exists( lt_redef_v[ mtdname = ls_tmdir-methodname ] ).
      ls_msrc-redefine = 'X'.
    ENDIF.
    LOOP AT lt_meth_src INTO DATA(ls_meth_line).
      APPEND ls_meth_line-line TO ls_msrc-source.
    ENDLOOP.
    APPEND ls_msrc TO lt_msrc.
  ENDLOOP.

  " STEP 4: CLASS-LOCAL DEFINITIONS / IMPLEMENTATIONS / MACROS / TESTS
  DATA(lv_ccdef) = CONV progname( cl_oo_classname_service=>get_ccdef_name( lv_class ) ).
  DATA(lv_ccimp) = CONV progname( cl_oo_classname_service=>get_ccimp_name( lv_class ) ).
  DATA(lv_ccmac) = CONV progname( cl_oo_classname_service=>get_ccmac_name( lv_class ) ).
  DATA(lv_ccau)  = CONV progname( cl_oo_classname_service=>get_ccau_name( lv_class ) ).
  PERFORM read_remote_source USING lv_ccdef CHANGING lt_locals_def.
  PERFORM read_remote_source USING lv_ccimp CHANGING lt_locals_imp.
  PERFORM read_remote_source USING lv_ccmac CHANGING lt_locals_mac.
  PERFORM read_remote_source USING lv_ccau  CHANGING lt_locals_au.

  " STEP 5: CREATE / OVERWRITE THE CLASS
  CALL FUNCTION 'SEO_CLASS_CREATE_COMPLETE'
    EXPORTING
      corrnr          = p_ltr
      devclass        = p_pk
      version         = lv_version
      overwrite       = 'X'
      suppress_dialog = 'X'
      method_sources  = lt_msrc
      locals_def      = lt_locals_def
      locals_imp      = lt_locals_imp
      locals_mac      = lt_locals_mac
      locals_au       = lt_locals_au
    IMPORTING
      korrnr          = lv_korrnr
    CHANGING
      class           = ls_class
      inheritance     = ls_inherit
      redefinitions   = lt_redefs
      implementings   = lt_impls
      attributes      = lt_attributes
      methods         = lt_methods
      events          = lt_events
      types           = lt_types
      parameters      = lt_parameters
      exceps          = lt_exceps
      aliases         = lt_aliases
      typepusages     = lt_typepusage
      friendships     = lt_friends
    EXCEPTIONS
      existing        = 1
      is_interface    = 2
      db_error        = 3
      component_error = 4
      no_access       = 5
      other           = 6
      OTHERS          = 7.
  IF sy-subrc <> 0.
    lcl_util=>add_msg( iv_type = 'E' iv_text = |CLAS { lv_class } - CLASS CREATE FAILURE| ).
    RETURN.
  ENDIF.
  COMMIT WORK.

  " STEP 6: TEXT SYMBOLS OF THE CLASS POOL
  DATA(lv_pool) = CONV progname( cl_oo_classname_service=>get_classpool_name( lv_class ) ).
  DATA: ls_prog_inf     TYPE rpy_prog,
        lt_incl         TYPE STANDARD TABLE OF rpy_repo,
        lt_s72          TYPE STANDARD TABLE OF abapsource,
        lt_s255         TYPE STANDARD TABLE OF abaptxt255,
        lt_textelements TYPE STANDARD TABLE OF textpool.
  CALL FUNCTION 'RPY_PROGRAM_READ' DESTINATION p_rd
    EXPORTING
      program_name          = lv_pool
      with_includelist      = ' '
      only_texts            = 'X'
    IMPORTING
      prog_inf              = ls_prog_inf
    TABLES
      include_tab           = lt_incl
      source                = lt_s72
      source_extended       = lt_s255
      textelements          = lt_textelements
    EXCEPTIONS
      OTHERS                = 1.
  IF sy-subrc = 0 AND lt_textelements IS NOT INITIAL.
    INSERT TEXTPOOL lv_pool FROM lt_textelements LANGUAGE sy-langu.
  ENDIF.

  PERFORM activate_wb USING 'CLAS' lv_class.
  lcl_util=>add_tr_obj( iv_pgmid = 'R3TR' iv_object = 'CLAS' iv_name = lv_class ).
  lcl_util=>add_msg( iv_type = 'S' iv_text = |CLAS { lv_class } - COPIED ({ lines( lt_msrc ) } METHODS)| ).
ENDFORM.

*&---------------------------------------------------------------------*
*& FORM MAKE_DYNP
*&---------------------------------------------------------------------*
*& 16a. ONE SCREEN: HEADER, FIELDS, FLOW LOGIC, PARAMETERS, TEXTS
*&      (READ: SVRS_GET_VERSION_DYNP_40, WRITE: EXPORT DYNPRO)
*&---------------------------------------------------------------------*
FORM make_dynp USING iv_prog TYPE csequence
                     iv_dnum TYPE csequence.
  DATA: lv_prog      TYPE progname,
        lv_dnum      TYPE d020s-dnum,
        lv_objname   TYPE vrsd-objname,
        lv_info      TYPE abaptext-line,
        lv_dynid     TYPE c LENGTH 44,
        lv_trname    TYPE trobj_name,
        lt_vsmodisrc TYPE STANDARD TABLE OF smodisrc,
        lt_vsmodilog TYPE STANDARD TABLE OF smodilog,
        lt_d020s     TYPE STANDARD TABLE OF d020s,
        lt_d020t     TYPE STANDARD TABLE OF d020t,
        lt_d021se    TYPE STANDARD TABLE OF d021se,
        lt_d021s     TYPE STANDARD TABLE OF d021s,
        lt_d021t     TYPE STANDARD TABLE OF d021t,
        lt_d022s     TYPE STANDARD TABLE OF d022s,
        lt_d023s     TYPE STANDARD TABLE OF d023s,
        ls_d020s     TYPE d020s.

  lv_prog       = iv_prog.
  lv_dnum       = iv_dnum.
  lv_objname    = lv_prog.
  lv_objname+40 = lv_dnum.

  CALL FUNCTION 'SVRS_GET_VERSION_DYNP_40'
    EXPORTING
      destination           = p_rd
      object_name           = lv_objname
      versno                = '00000'
    IMPORTING
      info_line             = lv_info
    TABLES
      vsmodisrc             = lt_vsmodisrc
      d020s_tab             = lt_d020s
      d020t_tab             = lt_d020t
      d021s_tab             = lt_d021se
      d021t_tab             = lt_d021t
      d022s_tab             = lt_d022s
      d023s_tab             = lt_d023s
      vsmodilog             = lt_vsmodilog
    EXCEPTIONS
      no_version            = 1
      system_failure        = 2
      communication_failure = 3
      OTHERS                = 4.
  READ TABLE lt_d020s INTO ls_d020s INDEX 1.
  IF sy-subrc <> 0.
    lcl_util=>add_msg( iv_type = 'E' iv_text = |DYNP { lv_prog } { lv_dnum } - READ FROM SOURCE FAILED| ).
    RETURN.
  ENDIF.

  lt_d021s = CORRESPONDING #( lt_d021se ).

  lv_dynid     = lv_prog.
  lv_dynid+40  = lv_dnum.
  EXPORT DYNPRO ls_d020s lt_d021s lt_d022s lt_d023s ID lv_dynid.
  IF sy-subrc <> 0.
    lcl_util=>add_msg( iv_type = 'E' iv_text = |DYNP { lv_prog } { lv_dnum } - SCREEN NOT WRITTEN| ).

  ENDIF.

  " SCREEN TITLE AND FIELD TEXTS (ALL LANGUAGES)
  DELETE FROM d020t WHERE prog = lv_prog AND dynr = lv_dnum.
  MODIFY d020t FROM TABLE lt_d020t.
  DELETE FROM d021t WHERE prog = lv_prog AND dynr = lv_dnum.
  MODIFY d021t FROM TABLE lt_d021t.
  COMMIT WORK.

  lv_trname    = lv_prog.
  lv_trname+40 = lv_dnum.
  lcl_util=>add_tr_obj( iv_pgmid = 'LIMU' iv_object = 'DYNP' iv_name = lv_trname ).
  lcl_util=>add_msg( iv_type = 'S' iv_text = |DYNP { lv_prog } { lv_dnum } - COPIED| ).
ENDFORM.

*&---------------------------------------------------------------------*
*& FORM MAKE_CUAD
*&---------------------------------------------------------------------*
*& 16b. GUI STATUS / TITLES / MENUS OF A PROGRAM (REMOTE-ENABLED FETCH)
*&---------------------------------------------------------------------*
FORM make_cuad USING iv_prog TYPE csequence.
  DATA: lv_prog   TYPE trdir-name,
        ls_adm    TYPE rsmpe_adm,
        lt_sta    TYPE STANDARD TABLE OF rsmpe_stat,
        lt_fun    TYPE STANDARD TABLE OF rsmpe_funt,
        lt_men    TYPE STANDARD TABLE OF rsmpe_men,
        lt_mtx    TYPE STANDARD TABLE OF rsmpe_mnlt,
        lt_act    TYPE STANDARD TABLE OF rsmpe_act,
        lt_but    TYPE STANDARD TABLE OF rsmpe_but,
        lt_pfk    TYPE STANDARD TABLE OF rsmpe_pfk,
        lt_set    TYPE STANDARD TABLE OF rsmpe_staf,
        lt_doc    TYPE STANDARD TABLE OF rsmpe_atrt,
        lt_tit    TYPE STANDARD TABLE OF rsmpe_titt,
        lt_biv    TYPE STANDARD TABLE OF rsmpe_buts,
        ls_tr_key TYPE trkey,
        lv_fg     TYPE string,
        lv_msg    TYPE c LENGTH 200.

  lv_prog = iv_prog.

  CALL FUNCTION 'RS_CUA_INTERNAL_FETCH' DESTINATION p_rd
    EXPORTING
      program               = lv_prog
      language              = sy-langu
      state                 = 'A'
    IMPORTING
      adm                   = ls_adm
    TABLES
      sta                   = lt_sta
      fun                   = lt_fun
      men                   = lt_men
      mtx                   = lt_mtx
      act                   = lt_act
      but                   = lt_but
      pfk                   = lt_pfk
      set                   = lt_set
      doc                   = lt_doc
      tit                   = lt_tit
      biv                   = lt_biv
    EXCEPTIONS
      not_found             = 1
      unknown_version       = 2
      system_failure        = 3 MESSAGE lv_msg
      communication_failure = 4 MESSAGE lv_msg
      OTHERS                = 5.
  IF sy-subrc = 1.
    RETURN.                           " PROGRAM HAS NO GUI STATUS
  ELSEIF sy-subrc <> 0.
    lcl_util=>add_msg( iv_type = 'E' iv_text = |CUAD { lv_prog } - READ FROM SOURCE FAILED { lv_msg }| ).
    RETURN.
  ENDIF.
  IF lt_sta IS INITIAL AND lt_tit IS INITIAL AND lt_men IS INITIAL.
    RETURN.
  ENDIF.

  " TRANSPORT KEY: FUNCTION GROUP FOR SAPL<FG>, OTHERWISE THE PROGRAM
  ls_tr_key-devclass = p_pk.
  ls_tr_key-sub_type = 'CUAD'.
  ls_tr_key-sub_name = lv_prog.
  IF lv_prog CP 'SAPL*' OR lv_prog CP '/*/SAPL*'.
    lv_fg = replace( val = CONV string( lv_prog ) sub = 'SAPL' with = '' ).
    ls_tr_key-obj_type = 'FUGR'.
    ls_tr_key-obj_name = lv_fg.
  ELSE.
    ls_tr_key-obj_type = 'PROG'.
    ls_tr_key-obj_name = lv_prog.
  ENDIF.

  sy-tcode = 'SE41' ##WRITE_OK.   " SAME WORKAROUND AS ABAPGIT (NOTE 2159455)
  CALL FUNCTION 'RS_CUA_INTERNAL_WRITE'
    EXPORTING
      program   = lv_prog
      language  = sy-langu
      tr_key    = ls_tr_key
      adm       = ls_adm
      state     = 'A'
    TABLES
      sta       = lt_sta
      fun       = lt_fun
      men       = lt_men
      mtx       = lt_mtx
      act       = lt_act
      but       = lt_but
      pfk       = lt_pfk
      set       = lt_set
      doc       = lt_doc
      tit       = lt_tit
      biv       = lt_biv
    EXCEPTIONS
      not_found = 1
      OTHERS    = 2.
  IF sy-subrc <> 0.
    lcl_util=>add_msg( iv_type = 'E' iv_text = |CUAD { lv_prog } - GUI STATUS NOT WRITTEN| ).
    RETURN.
  ENDIF.
  COMMIT WORK.

  lcl_util=>add_tr_obj( iv_pgmid = 'LIMU' iv_object = 'CUAD' iv_name = lv_prog ).
  lcl_util=>add_msg( iv_type = 'S' iv_text = |CUAD { lv_prog } - GUI STATUS COPIED| ).
ENDFORM.

*&---------------------------------------------------------------------*
*& FORM MAKE_TABU
*&---------------------------------------------------------------------*
*& 20. TABLE CONTENTS: THE KEYS RECORDED IN E071K OF THE SOURCE
*&     REQUEST (INCL. GENERIC KEYS WITH '*') ARE TURNED INTO WHERE
*&     CONDITIONS, THE ROWS ARE READ VIA RFC AND WRITTEN HERE. THE SAME
*&     KEYS ARE ADDED TO THE LOCAL REQUEST.
*&---------------------------------------------------------------------*
FORM make_tabu USING iv_name TYPE csequence.
  TYPES: BEGIN OF ty_e071k,
           trkorr     TYPE e071k-trkorr,
           pgmid      TYPE e071k-pgmid,
           object     TYPE e071k-object,
           objname    TYPE e071k-objname,
           mastertype TYPE e071k-mastertype,
           mastername TYPE e071k-mastername,
           tabkey     TYPE e071k-tabkey,
         END OF ty_e071k.
  DATA: lv_tab     TYPE tabname,
        lt_e071k   TYPE STANDARD TABLE OF ty_e071k,
        lt_options TYPE tt_options,
        lt_conds   TYPE STANDARD TABLE OF string WITH EMPTY KEY,
        lv_where   TYPE string,
        lv_key     TYPE string,
        lv_val     TYPE string,
        lv_pos     TYPE i,
        lv_len     TYPE i,
        lv_keylen  TYPE i,
        lv_ok      TYPE abap_bool,
        lv_rows    TYPE i,
        lv_subrc   TYPE sy-subrc.

  lv_tab = iv_name.

  SELECT SINGLE tabname FROM dd02l INTO @DATA(lv_exists)
    WHERE tabname = @lv_tab AND as4local = 'A'.
  IF sy-subrc <> 0.
    lcl_util=>add_msg( iv_type = 'E' iv_text = |TABU { lv_tab } - TABLE NOT ACTIVE IN THIS SYSTEM, COPY THE TABLE FIRST| ).
    RETURN.
  ENDIF.

  " KEYS OF THIS TABLE IN THE SOURCE REQUEST AND ITS TASKS
  LOOP AT gt_trkorrs INTO DATA(lv_tr).
    IF sy-tabix = 1.
      APPEND VALUE #( text = |( TRKORR = '{ lv_tr }'| ) TO lt_options.
    ELSE.
      APPEND VALUE #( text = |OR TRKORR = '{ lv_tr }'| ) TO lt_options.
    ENDIF.
  ENDLOOP.
  APPEND VALUE #( text = |) AND OBJECT = 'TABU'| ) TO lt_options.
  APPEND VALUE #( text = |AND OBJNAME = '{ lv_tab }'| ) TO lt_options.

  lcl_util=>rfc_read(
    EXPORTING iv_table  = 'E071K'
              it_where  = lt_options
              it_fields = VALUE #( ( 'TRKORR' ) ( 'PGMID' ) ( 'OBJECT' ) ( 'OBJNAME' )
                                   ( 'MASTERTYPE' ) ( 'MASTERNAME' ) ( 'TABKEY' ) )
    IMPORTING et_data   = lt_e071k
              ev_subrc  = lv_subrc ).
  SORT lt_e071k BY tabkey.
  DELETE ADJACENT DUPLICATES FROM lt_e071k COMPARING tabkey.
  IF lt_e071k IS INITIAL.
    lcl_util=>add_msg( iv_type = 'W' iv_text = |TABU { lv_tab } - NO KEYS (E071K) IN THE SOURCE REQUEST| ).
    RETURN.
  ENDIF.

  " KEY FIELDS OF THE TABLE IN KEY ORDER
  SELECT fieldname, position, leng, inttype, datatype
    FROM dd03l
    WHERE tabname = @lv_tab AND as4local = 'A' AND keyflag = 'X'
      AND fieldname NOT LIKE '.%'
    ORDER BY position
    INTO TABLE @DATA(lt_keys).

  LOOP AT lt_e071k INTO DATA(ls_e071k).
    CLEAR: lt_conds, lv_pos.
    lv_key    = ls_e071k-tabkey.
    lv_keylen = strlen( lv_key ).

    LOOP AT lt_keys INTO DATA(ls_keyfield).
      lv_len = ls_keyfield-leng.
      IF ls_keyfield-datatype = 'CLNT'.
        lv_pos = lv_pos + lv_len.        " CLIENT: RFC LOGON CLIENT / SY-MANDT
        CONTINUE.
      ENDIF.
      IF ls_keyfield-inttype NA 'CNDT' OR lv_pos >= lv_keylen.
        EXIT.                            " NON-CHARACTER KEY OR KEY ENDS HERE
      ENDIF.
      IF lv_pos + lv_len > lv_keylen.
        lv_len = lv_keylen - lv_pos.
      ENDIF.
      lv_val = lv_key+lv_pos(lv_len).
      IF lv_val CA '*'.
        " GENERIC KEY: PREFIX BEFORE '*' -> LIKE 'PREFIX%'
        SPLIT lv_val AT '*' INTO DATA(lv_prefix) DATA(lv_dummy).
        IF lv_prefix IS NOT INITIAL.
          REPLACE ALL OCCURRENCES OF `'` IN lv_prefix WITH `''`.
          APPEND |{ ls_keyfield-fieldname } LIKE '{ lv_prefix }%'| TO lt_conds.
        ENDIF.
        EXIT.
      ENDIF.
      REPLACE ALL OCCURRENCES OF `'` IN lv_val WITH `''`.
      APPEND |{ ls_keyfield-fieldname } = '{ lv_val }'| TO lt_conds.
      lv_pos = lv_pos + ls_keyfield-leng.
    ENDLOOP.

    CONCATENATE LINES OF lt_conds INTO lv_where SEPARATED BY ` AND `.
    PERFORM copy_tables USING lv_tab lv_where abap_false CHANGING lv_ok.
    IF lv_ok = abap_true.
      lv_rows = lv_rows + 1.
      APPEND VALUE #( pgmid = 'R3TR' object = 'TABU' objname = lv_tab
                      mastertype = 'TABU' mastername = lv_tab
                      tabkey = ls_e071k-tabkey ) TO gt_e071k.
    ENDIF.
  ENDLOOP.

  IF lv_rows > 0.
    APPEND VALUE #( pgmid = 'R3TR' object = 'TABU' obj_name = lv_tab objfunc = 'K' ) TO gt_e071.
    lcl_util=>add_msg( iv_type = 'S' iv_text = |TABU { lv_tab } - { lv_rows } KEY(S) COPIED| ).
  ENDIF.
ENDFORM.

*&---------------------------------------------------------------------*
*& FORM MAKE_SUSO
*&---------------------------------------------------------------------*
*& 21b. AUTHORIZATION OBJECT (SU21): DEFINITION (DB TABLE TOBJ - NOT THE
*&      TRANSPORT TYPE TOBJ), TEXTS, PERMITTED ACTIVITIES, SU24 DEFAULTS.
*&      CUSTOMER AUTHORIZATION FIELDS OF THE OBJECT ARE COPIED FIRST.
*&---------------------------------------------------------------------*
FORM make_suso USING iv_name TYPE csequence.
  TYPES: BEGIN OF ty_tobj,
           objct TYPE tobj-objct,
           fiel1 TYPE tobj-fiel1,
           fiel2 TYPE tobj-fiel2,
           fiel3 TYPE tobj-fiel3,
           fiel4 TYPE tobj-fiel4,
           fiel5 TYPE tobj-fiel5,
           fiel6 TYPE tobj-fiel6,
           fiel7 TYPE tobj-fiel7,
           fiel8 TYPE tobj-fiel8,
           fiel9 TYPE tobj-fiel9,
           fiel0 TYPE tobj-fiel0,
         END OF ty_tobj.
  DATA: lv_objct TYPE tobj-objct,
        lt_tobj  TYPE STANDARD TABLE OF ty_tobj,
        lv_where TYPE string,
        lv_subrc TYPE sy-subrc,
        lv_ok    TYPE abap_bool,
        lv_ok2   TYPE abap_bool.
  FIELD-SYMBOLS <lv_field> TYPE any.

  lv_objct = iv_name.

  " CUSTOMER FIELDS USED BY THE OBJECT MUST EXIST FIRST
  lcl_util=>rfc_read(
    EXPORTING iv_table  = 'TOBJ'
              it_where  = VALUE #( ( text = |OBJCT = '{ lv_objct }'| ) )
              it_fields = VALUE #( ( 'OBJCT' ) ( 'FIEL1' ) ( 'FIEL2' ) ( 'FIEL3' ) ( 'FIEL4' )
                                   ( 'FIEL5' ) ( 'FIEL6' ) ( 'FIEL7' ) ( 'FIEL8' ) ( 'FIEL9' ) ( 'FIEL0' ) )
    IMPORTING et_data   = lt_tobj
              ev_subrc  = lv_subrc ).
  READ TABLE lt_tobj INTO DATA(ls_tobj) INDEX 1.
  IF sy-subrc <> 0.
    lcl_util=>add_msg( iv_type = 'E' iv_text = |SUSO { lv_objct } - NOT FOUND IN SOURCE SYSTEM| ).
    RETURN.
  ENDIF.
  DATA: lv_comp  TYPE i,
        lv_afield TYPE authx-fieldname.
  DO 10 TIMES.
    lv_comp = sy-index + 1.                    " FIEL1 ... FIEL0
    ASSIGN COMPONENT lv_comp OF STRUCTURE ls_tobj TO <lv_field>.
    IF sy-subrc <> 0 OR <lv_field> IS INITIAL.
      CONTINUE.
    ENDIF.
    lv_afield = <lv_field>.
    IF lv_afield CP 'Z*' OR lv_afield CP 'Y*' OR lv_afield CP '/*'.
      SELECT SINGLE fieldname FROM authx INTO @DATA(lv_exists) WHERE fieldname = @lv_afield.
      IF sy-subrc <> 0.
        PERFORM make_auth USING lv_afield.
      ENDIF.
    ENDIF.
  ENDDO.

  lv_where = |OBJCT = '{ lv_objct }'|.
  PERFORM copy_tables USING 'TOBJ TOBJT TACTZ' lv_where abap_false CHANGING lv_ok.
  PERFORM copy_tables USING 'TOBJVOR TOBJVORDAT TOBJVORFLG' lv_where abap_true CHANGING lv_ok2.
  IF lv_ok = abap_true.
    lcl_util=>create_tadir( iv_object = 'SUSO' iv_name = lv_objct ).
    lcl_util=>add_tr_obj( iv_pgmid = 'R3TR' iv_object = 'SUSO' iv_name = lv_objct ).
    lcl_util=>add_msg( iv_type = 'S' iv_text = |SUSO { lv_objct } - COPIED| ).
  ENDIF.
ENDFORM.

*&---------------------------------------------------------------------*
*& FORM LOCK_KEYS_INTO_TR
*&---------------------------------------------------------------------*
*& RECORD TABLE CONTENTS (TABU + E071K KEYS) IN THE LOCAL REQUEST
*&---------------------------------------------------------------------*
FORM lock_keys_into_tr.
  IF gt_e071 IS INITIAL.
    RETURN.
  ENDIF.

  SORT gt_e071 BY obj_name.
  DELETE ADJACENT DUPLICATES FROM gt_e071 COMPARING obj_name.

  CALL FUNCTION 'TR_APPEND_TO_COMM_OBJS_KEYS'
    EXPORTING
      wi_trkorr = p_ltr
    TABLES
      wt_e071   = gt_e071
      wt_e071k  = gt_e071k
    EXCEPTIONS
      OTHERS    = 1.
  IF sy-subrc <> 0.
    lcl_util=>add_msg( iv_type = 'E' iv_text = |TABLE KEYS NOT ADDED TO REQUEST { p_ltr } (USE A TASK OR CUSTOMIZING REQUEST)| ).
  ELSE.
    lcl_util=>add_msg( iv_type = 'S' iv_text = |{ lines( gt_e071k ) } TABLE KEY(S) ADDED TO REQUEST { p_ltr }| ).
  ENDIF.
ENDFORM.

*&---------------------------------------------------------------------*
*& FORM DISPLAY_RETURN
*&---------------------------------------------------------------------*
*& RESULT LOG (ONE LINE PER OBJECT / ERROR)
*&---------------------------------------------------------------------*
FORM display_return.
  DATA: lt_fieldcat       TYPE slis_t_fieldcat_alv,
        ls_fieldcat       TYPE slis_fieldcat_alv,
        lt_return_display TYPE TABLE OF bapiret2.

  lt_return_display = t_return.

  CLEAR ls_fieldcat.
  ls_fieldcat-fieldname = 'TYPE'.
  ls_fieldcat-seltext_m = 'MESSAGE TYPE'.
  APPEND ls_fieldcat TO lt_fieldcat.

  CLEAR ls_fieldcat.
  ls_fieldcat-fieldname = 'MESSAGE'.
  ls_fieldcat-seltext_m = 'MESSAGE TEXT'.
  ls_fieldcat-outputlen = 120.
  APPEND ls_fieldcat TO lt_fieldcat.

  CALL FUNCTION 'REUSE_ALV_GRID_DISPLAY'
    EXPORTING
      i_callback_program = sy-repid
      it_fieldcat        = lt_fieldcat
    TABLES
      t_outtab           = lt_return_display
    EXCEPTIONS
      program_error      = 1
      OTHERS             = 2.
  IF sy-subrc <> 0.
    MESSAGE 'ERROR IN ALV DISPLAY' TYPE 'E'.
  ENDIF.
ENDFORM.

*&---------------------------------------------------------------------*
*& FORM LOCK_INTO_TR
*&---------------------------------------------------------------------*
*& RECORD ALL CREATED OBJECTS IN THE LOCAL TRANSPORT REQUEST P_LTR
*&---------------------------------------------------------------------*
FORM lock_into_tr.

  IF gt_objects IS INITIAL.
    RETURN.
  ENDIF.

  SORT gt_objects BY pgmid object obj_name.
  DELETE ADJACENT DUPLICATES FROM gt_objects COMPARING pgmid object obj_name.

  CALL FUNCTION 'TR_OBJECTS_CHECK'
    EXPORTING
      iv_no_show_option       = 'X'
    TABLES
      wt_ko200                = gt_objects
    EXCEPTIONS
      cancel_edit_other_error = 1
      show_only               = 2
      OTHERS                  = 3.
  IF sy-subrc <> 0.
    lcl_util=>add_msg( iv_type = 'E' iv_text = 'TR OBJECT CHECK FAILED' ).
    RETURN.
  ENDIF.

  CALL FUNCTION 'TRINT_OBJECTS_CHECK_AND_INSERT'
    EXPORTING
      iv_order          = p_ltr
      iv_with_dialog    = 'X'
      iv_no_show_option = 'X'
    CHANGING
      ct_ko200          = gt_objects
    EXCEPTIONS
      OTHERS            = 1.
  IF sy-subrc <> 0.
    lcl_util=>add_msg( iv_type = 'E' iv_text = |OBJECTS NOT ADDED TO REQUEST { p_ltr }| ).
  ELSE.
    lcl_util=>add_msg( iv_type = 'S' iv_text = |{ lines( gt_objects ) } OBJECT ENTRIES ADDED TO REQUEST { p_ltr }| ).
  ENDIF.

ENDFORM.

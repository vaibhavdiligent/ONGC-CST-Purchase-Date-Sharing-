PARAMETERS: p_rd TYPE rfcdes-rfcdest  .

*---------------------------------------------*
* START-OF-SELECTION
*---------------------------------------------*
START-OF-SELECTION.
  TRY.

      " DEFINE SELECTION CRITERIA FOR SPECIFIC TR
      CONCATENATE 'TRKORR = ''' p_tr '''' INTO ls_options-text.
      APPEND ls_options TO lt_options.

      " CALL RFC_READ_TABLE TO GET ALL FIELDS FROM E071
      CALL FUNCTION 'RFC_READ_TABLE' DESTINATION p_rd
        EXPORTING
          query_table          = 'E071'
        TABLES
          fields               = lt_fields
          data                 = lt_data
          options              = lt_options
        EXCEPTIONS
          table_not_available  = 1
          table_without_data   = 2
          option_not_valid     = 3
          field_not_valid      = 4
          not_authorized       = 5
          data_buffer_exceeded = 6
          OTHERS               = 7.

      IF sy-subrc = 0.
        " PROCESS THE RETURNED DATA
        LOOP AT lt_data INTO ls_data.
          CLEAR ls_e071.
          " MAP THE RAW DATA TO STRUCTURE (ASSUMING FIELDS ARE IN CORRECT ORDER)
          ls_e071-trkorr    = ls_data+0(20).
          ls_e071-as4pos    = ls_data+20(6).
          ls_e071-pgmid     = ls_data+26(4).
          ls_e071-object    = ls_data+30(4).
          ls_e071-obj_name  = ls_data+34(120).
          ls_e071-objfunc   = ls_data+154(1).
          ls_e071-lockflag  = ls_data+155(1).
          ls_e071-gennum    = ls_data+156(3).
          ls_e071-lang      = ls_data+159(1).
          ls_e071-activity  = ls_data+160(20).
          APPEND ls_e071 TO lt_e071.
        ENDLOOP.
      ENDIF.


    CATCH cx_root INTO DATA(lx_error).
      MESSAGE lx_error->get_text( ) TYPE 'E'.
  ENDTRY.


  IF  lt_e071 IS INITIAL.
    MESSAGE 'NO OBJECTS FOUND FOR THE GIVEN TRANSPORT REQUEST.' TYPE 'I'.
    EXIT.
  ENDIF.

* MAP RESULT TO INTERNAL TABLE WITH CHECKBOX FIELD
  LOOP AT lt_e071 INTO ls_e071.
    MOVE-CORRESPONDING ls_e071 TO wa_data.
    APPEND wa_data TO it_data.
  ENDLOOP.


  LOOP AT it_data ASSIGNING FIELD-SYMBOL(<fs_data>).
*    PERFORM check_object_exist.
    DATA : ls_tadir TYPE tadir.
    CALL FUNCTION 'TR_CHECK_EXIST'
      EXPORTING
        iv_pgmid             = <fs_data>-pgmid    " Key field E071-PGMID
        iv_object            = <fs_data>-object   " Key field E071-OBJECT
        iv_obj_name          = <fs_data>-obj_name " Key field E071-OBJ_NAME
        is_tadir             = ls_tadir              " Assigned TADIR key if known
      IMPORTING
        e_exist              = ev_exists          " Return value of object existence
      EXCEPTIONS
        tr_no_check_function = 1                " No check function exists for this object type
        OTHERS               = 2.

    IF ev_exists = 'X'.
      <fs_data>-activity = 'AVAILABLE'.
    ELSE.
      <fs_data>-activity = 'NOT AVAILABLE'.
    ENDIF.

    ev_exists = ''.
  ENDLOOP.


  PERFORM build_fieldcat.

  v_repid = sy-repid.
  wa_layout-box_fieldname = 'SEL'.


  CALL FUNCTION 'REUSE_ALV_GRID_DISPLAY'
    EXPORTING
      i_callback_program       = v_repid
      it_fieldcat              = it_fieldcat
      i_save                   = 'A'
      is_layout                = wa_layout
      i_callback_pf_status_set = 'SET_PF_STATUS'
      i_callback_user_command  = 'USER_COMMAND'
    TABLES
      t_outtab                 = it_data.

*---------------------------------------------*
* USER COMMAND HANDLING
*---------------------------------------------*
FORM user_command USING r_ucomm     LIKE sy-ucomm
                        rs_selfield TYPE slis_selfield.

  CASE r_ucomm.
    WHEN 'COPY'.
      READ TABLE it_data WITH KEY sel = 'X' TRANSPORTING NO FIELDS.
      IF sy-subrc = 0.
        PERFORM process_selected.
        IF t_return IS NOT INITIAL.
          PERFORM display_return.
        ELSE.
          MESSAGE 'Objects are created.' TYPE 'S'.
        ENDIF.

      ELSE.
        MESSAGE 'NO ROWS SELECTED.' TYPE 'I'.
        RETURN.
      ENDIF.
    WHEN '&BACK' OR '&EXIT'.
      LEAVE PROGRAM.
  ENDCASE.

ENDFORM.

*---------------------------------------------*
* PROCESS SELECTED ROWS
*---------------------------------------------*
FORM process_selected.

*--------------------------------------------------------------------*
* CREATE DDIC DOMAIN OBJECTS  .
*--------------------------------------------------------------------*
  LOOP AT it_data INTO DATA(wa_selected) WHERE sel = 'X'
                                         AND object = 'DOMA' .
    gv_pgmid = wa_selected-pgmid.

    lv_ddic = wa_selected-obj_name.
    PERFORM make_doma.
    WAIT UP TO 2 SECONDS.
  ENDLOOP.

*--------------------------------------------------------------------*

*--------------------------------------------------------------------*
* CREATE DDIC ELEMENT OBJECTS  .
*--------------------------------------------------------------------*


  LOOP AT it_data INTO wa_selected WHERE sel = 'X'
                                       AND object = 'DTEL' .
    gv_pgmid = wa_selected-pgmid.
    lv_ddic = wa_selected-obj_name.
    PERFORM make_detl.
    WAIT UP TO 2 SECONDS.
  ENDLOOP.

*--------------------------------------------------------------------*
* CREATE TABLE / STRUCTURE OBJECTS  .
*--------------------------------------------------------------------*
  LOOP AT it_data INTO wa_selected WHERE sel = 'X'
                                       AND object = 'TABL' .
    gv_pgmid = wa_selected-pgmid.
    lv_ddic = wa_selected-obj_name.
    PERFORM make_tabl.
    WAIT UP TO 2 SECONDS.
  ENDLOOP.
*--------------------------------------------------------------------*

*--------------------------------------------------------------------*
* CREATE PROGRAMS & INCLUDES .
*--------------------------------------------------------------------*

  LOOP AT it_data INTO wa_selected WHERE sel = 'X'
                                          AND ( object = 'INCL' OR object = 'PROG' OR object = 'REPS' )  .
    gv_pgmid = wa_selected-pgmid.
    lv_prog = wa_selected-obj_name.

    PERFORM make_prog.
    WAIT UP TO 2 SECONDS.
  ENDLOOP.

*--------------------------------------------------------------------*

*--------------------------------------------------------------------*
* CREATE FUNCATION GROUP & FUNCTION.
*--------------------------------------------------------------------*

  LOOP AT it_data INTO wa_selected WHERE sel = 'X'
                                   AND object = 'FUGR' .

    gv_pgmid = wa_selected-pgmid.
    lv_fgname = wa_selected-obj_name.

    "STEP - 1 CREATE FUNCTION GROUP.

    CALL FUNCTION 'CRMOST_FUGR_EXISTS'
      EXPORTING
        name       = lv_fgname
        objtype    = 'FUGR'
      IMPORTING
        bapireturn = ls_return.
    .
    "CHECK FUNCTION GROUP.
    IF ls_return IS NOT INITIAL.

      PERFORM make_functiongroup.

      PERFORM make_includes.

      WAIT UP TO 2 SECONDS.
    ENDIF.

*    IF gv_fugr IS INITIAL.
    "STEP - 2 GET FUNCTION NAME LIST
    CONCATENATE 'PNAME =' '''' 'SAPL' lv_fgname ''''
    INTO wa_options-text .
    APPEND wa_options TO it_options.

    wa_fields-fieldname = 'FUNCNAME'.
    APPEND wa_fields TO it_fields.

    CALL FUNCTION 'RFC_READ_TABLE' DESTINATION p_rd
      EXPORTING
        query_table          = 'TFDIR'
      TABLES
        options              = it_options
        fields               = it_fields
        data                 = it_data52
      EXCEPTIONS
        table_not_available  = 1
        table_without_data   = 2
        option_not_valid     = 3
        field_not_valid      = 4
        not_authorized       = 5
        data_buffer_exceeded = 6
        OTHERS               = 7.
    IF it_data52 IS NOT INITIAL.
      "STEP - 3 CREATE FUNCTION
      LOOP AT it_data52 INTO lv_fname.
        PERFORM make_func.
        WAIT UP TO 2 SECONDS.
      ENDLOOP.
    ENDIF.
*    ENDIF.

  ENDLOOP.

*--------------------------------------------------------------------*
*--------------------------------------------------------------------*
* CREATE TRAN OBJECTS  .
*--------------------------------------------------------------------*

  LOOP AT it_data INTO wa_selected WHERE sel = 'X'
                                   AND object = 'TRAN' .
    lv_tran = wa_selected-obj_name.
    gv_pgmid = wa_selected-pgmid.
    PERFORM make_tran.
    WAIT UP TO 2 SECONDS.
  ENDLOOP.

*--------------------------------------------------------------------*

  PERFORM lock_into_tr.


ENDFORM.

*&---------------------------------------------------------------------*
*& CREATE DDIC OBJECT DOMAIN
*&---------------------------------------------------------------------*
FORM make_doma .
  DATA :
    lv_vrsd       TYPE vrsd-objname,
    lv_rfcdes     TYPE rfcdes-rfcdest,
    lv_trobj_name TYPE trobj_name.


  DATA : ls_dd01v_wa_a LIKE  dd01v,
         ls_dd01v_wa_n LIKE  dd01v,
         ls_got_state  LIKE  dcdomaget,
         "  TABLES
         dd07v_tab_a   TYPE TABLE OF  dd07v,
         dd07v_tab_n   TYPE TABLE OF  dd07v,
         dcdomaget     TYPE dcdomaget.

  " STEP 1: GET DETAILS OF DOMAIN FROM REMOTE SYSTEM.
  dcdomaget-doma = 'M'.

  CALL FUNCTION 'DD_DOMA_GET' DESTINATION p_rd
    EXPORTING
      domain_name   = lv_ddic
      get_state     = dcdomaget
      langu         = sy-langu
      prid          = 0
      withtext      = 'X'
    IMPORTING
      dd01v_wa_a    = ls_dd01v_wa_a
      dd01v_wa_n    = ls_dd01v_wa_n
*     GOT_STATE     =
    TABLES
      dd07v_tab_a   = dd07v_tab_a
      dd07v_tab_n   = dd07v_tab_n
    EXCEPTIONS
      illegal_value = 1
      op_failure    = 2
      OTHERS        = 3.

  IF sy-subrc <> 0.

    APPEND VALUE #(       type       = 'E'
                           id         = sy-msgid
                           number     = sy-msgno
                           message_v1 = sy-msgv1
                           message_v2 = sy-msgv2
                           message_v3 = sy-msgv3
                           message_v4 = sy-msgv4
                           message = lv_ddic && '-'&& 'DOMAIN PUT FAILURE' ) TO t_return.
  ELSE.

    " STEP 2: CREATE OR UPDATE DOMAIN USING DDIF_DOMA_PUT
    CALL FUNCTION 'DD_DOMA_PUT'
      EXPORTING
        dd01v_wa            = ls_dd01v_wa_a
        domain_name         = lv_ddic
*       PRID                = 0
*       CTRL_DOMA_PUT       = 'N'
*       SAVE_ABAP_LANGUAGE_VERSION = ' '
      TABLES
        dd07v_tab           = dd07v_tab_a
      EXCEPTIONS
        illegal_value       = 1
        op_failure          = 2
        object_inconsistent = 3
        OTHERS              = 4.


    IF sy-subrc <> 0.

      APPEND VALUE #(       type       = 'E'
                             id         = sy-msgid
                             number     = sy-msgno
                             message_v1 = sy-msgv1
                             message_v2 = sy-msgv2
                             message_v3 = sy-msgv3
                             message_v4 = sy-msgv4
                             message = lv_ddic && '-'&&  'DOMAIN PUT FAILURE' ) TO t_return.

    ELSE.

      COMMIT WORK.
*STEP 3 : LOCK TABLE INTO TR SYSTEM.
      lv_trobj_name = lv_ddic.


      gv_objnam = lv_trobj_name.
      gv_objtyp = 'DOMA'.
      PERFORM add_tr.

      " STEP 4: ACTIVATE THE DOMAIN USING DDIF_DOMA_ACTIVATE
      CALL FUNCTION 'DDIF_DOMA_ACTIVATE'
        EXPORTING
          name      = lv_ddic
        EXCEPTIONS
          not_found = 1
          OTHERS    = 2.

    ENDIF.
  ENDIF.

  CLEAR :
       lv_vrsd  ,
       lv_rfcdes   ,
       lv_trobj_name ,
       ls_dd01v_wa_a ,
       ls_dd01v_wa_n ,
       ls_got_state  ,
       dd07v_tab_a   ,
       dd07v_tab_n   ,
       dcdomaget     .
ENDFORM.

  DATA : gt_objects     TYPE TABLE OF ko200.

*     GET_STATE      = 'M'

      withtext       = 'X'

    IF sy-subrc <> 0.
      APPEND VALUE #(       type       = 'E'

FORM make_func .

*DATA
  DATA :
    lv_vrsd      TYPE vrsd-objname,

    lv_info_line TYPE abaptext-line,
*"  TABLES
    lt_vsmodisrc TYPE TABLE OF smodisrc,

  DATA:

    wa_export_parameter   TYPE rsexp,

  DATA: lt_source        TYPE STANDARD TABLE OF rssource,

    EXPORTING
      destination           = p_rd
      object_name           = lv_fname
      versno                = '00000'
*     IV_NO_RELEASE_TRANSFORMATION = LV_INFO_LINE
    IMPORTING
      info_line             = lv_info_line
    TABLES
      vsmodisrc             = lt_vsmodisrc
      uincl_tab             = lt_uincl_tab
      dincl_tab             = lt_dincl_tab
      enlfd_tab             = lt_enlfd_tab
      funct_tab             = lt_funct_tab
      fupar_tab             = lt_fupar_tab
      vsmodilog             = lt_vsmodilog
      tfdir_tab             = lt_tfdir_tab
      tftit_tab             = lt_tftit_tab
      trdir_tab             = lt_trdir_tab
    EXCEPTIONS
      no_version            = 1
      system_failure        = 2
      communication_failure = 3
      OTHERS                = 4.
  IF sy-subrc <> 0.
    APPEND VALUE #(       type       = 'E'
                       id         = sy-msgid
                       number     = sy-msgno
                       message_v1 = sy-msgv1
                       message_v2 = sy-msgv2
                       message_v3 = sy-msgv3
                       message_v4 = sy-msgv4
                       message = 'FUNCATION PUT FAILURE' ) TO t_return.
  ELSE.

    " STEP 2: MAPPING IMPORT & EXPORT PARAMETER
    LOOP AT lt_fupar_tab INTO DATA(wa_fupar_tab) WHERE paramtype = 'I'.
      wa_import_parameter-parameter = wa_fupar_tab-parameter.
      wa_import_parameter-types     = wa_fupar_tab-type.
      IF wa_fupar_tab-type IS NOT INITIAL.
        wa_import_parameter-typ       = wa_fupar_tab-structure.
      ELSE.
        wa_import_parameter-dbfield   = wa_fupar_tab-structure.

      ENDIF.
      APPEND wa_tables_parameter TO lt_tables_parameter.
      CLEAR wa_tables_parameter.
      CLEAR wa_fupar_tab.
    ENDLOOP.



    TYPES: BEGIN OF ty_line,
             line TYPE string,
           END OF ty_line.

    DATA: "lt_source TYPE TABLE OF ty_line,    " Input lines
      lt_target TYPE TABLE OF ty_line,    " Output formatted lines
      wa_target TYPE ty_line.

    DATA: lv_input   TYPE string,
          lv_code    TYPE string,
          lv_comment TYPE string,
          lv_line    TYPE string,
          lv_word    TYPE string.

    DATA: it_words TYPE STANDARD TABLE OF string WITH EMPTY KEY.

    LOOP AT lt_source ASSIGNING FIELD-SYMBOL(<wa_source>).
      lv_input = <wa_source>-line.

      CLEAR: lv_line, lv_code, lv_comment, it_words.

      " Case 1: Full-line comment (starts with * or ")
      IF lv_input CP '*' OR lv_input CP '"'.
        lv_comment = lv_input+1. " Remove prefix
        SPLIT lv_comment AT space INTO TABLE it_words.

        CLEAR lv_line.
        LOOP AT it_words INTO lv_word.
          IF strlen( lv_line ) + strlen( lv_word ) + 1 > 71.
            wa_target-line = |{ lv_input(0) } { lv_line }|.
            APPEND wa_target TO lt_target.
            CLEAR lv_line.
          ENDIF.

          IF lv_line IS INITIAL.
            lv_line = lv_word.
          ELSE.
            CONCATENATE lv_line lv_word INTO lv_line SEPARATED BY space.
          ENDIF.
        ENDLOOP.

        IF lv_line IS NOT INITIAL.
          wa_target-line = |{ lv_input(0) } { lv_line }|.
          APPEND wa_target TO lt_target.
        ENDIF.

        " Case 2: Code line with inline comment
      ELSEIF lv_input CS '"'.
        SPLIT lv_input AT '"' INTO lv_code lv_comment.

        " Combine code + first part of comment
        wa_target-line = |{ lv_code }" { lv_comment }|.
        IF strlen( wa_target-line ) <= 72.
          APPEND wa_target TO lt_target.
        ELSE.
          " Code part on one line
          wa_target-line = lv_code.
          APPEND wa_target TO lt_target.

          " Now wrap the comment part on next lines with "*"
          SPLIT lv_comment AT space INTO TABLE it_words.
          CLEAR lv_line.

          LOOP AT it_words INTO lv_word.
            IF strlen( lv_line ) + strlen( lv_word ) + 1 > 71.
              wa_target-line = |* { lv_line }|.
              APPEND wa_target TO lt_target.
              CLEAR lv_line.
            ENDIF.

            IF lv_line IS INITIAL.
              lv_line = lv_word.
            ELSE.
              CONCATENATE lv_line lv_word INTO lv_line SEPARATED BY space.
            ENDIF.
          ENDLOOP.

          IF lv_line IS NOT INITIAL.
            wa_target-line = |* { lv_line }|.
            APPEND wa_target TO lt_target.
          ENDIF.
        ENDIF.

        " Case 3: Pure code (no comment)
      ELSE.
        wa_target-line = lv_input.
        APPEND wa_target TO lt_target.
      ENDIF.

    ENDLOOP.

    "TO UPDATE THE SHORT TEXT FOR FUNCTION MODULE
    READ TABLE lt_tftit_tab INTO DATA(wa_tftit_tab) WITH KEY funcname = lv_function_name.
    lv_short_text2 = wa_tftit_tab-stext.

    " STEP 4: TRY TO INSERT FM INTO ALREADY EXISET FM GROUP
    IF lt_source IS NOT INITIAL.

      DATA:
        lv_index          TYPE sy-tabix,
        lv_first_function TYPE sy-tabix,
        lv_last_function  TYPE sy-tabix,
        lv_first_endfunc  TYPE sy-tabix,
        lv_last_endfunc   TYPE sy-tabix.
      CLEAR lv_line.
      " Step 1: Find first and last occurrences of FUNCTION
      CLEAR: lv_first_function, lv_last_function.
      LOOP AT lt_source INTO lv_line.
        lv_index = sy-tabix.
        IF lv_line CS 'FUNCTION'.
          IF lv_first_function IS INITIAL.
            lv_first_function = lv_index. " Store first occurrence
          ENDIF.
          lv_last_function = lv_index. " Update last occurrence
        ENDIF.
      ENDLOOP.

      " Step 2: Find first and last occurrences of ENDFUNCTION
      CLEAR: lv_first_endfunc, lv_last_endfunc.
      LOOP AT lt_source INTO lv_line.
        lv_index = sy-tabix.
        IF lv_line CS 'ENDFUNCTION.'.
          IF lv_first_endfunc IS INITIAL.
            lv_first_endfunc = lv_index. " Store first occurrence
          ENDIF.
          lv_last_endfunc = lv_index. " Update last occurrence
        ENDIF.
      ENDLOOP.

      " Step 3: Delete the last occurrences first (to avoid index shifting)
      IF lv_last_endfunc > 0.
        DELETE lt_source INDEX lv_last_endfunc.
      ENDIF.
      IF lv_last_function > 0.
        DELETE lt_source INDEX lv_last_function.
      ENDIF.

      " Step 4: Delete the first occurrences (adjust for index shift)
      IF lv_first_endfunc > 0 AND lv_first_endfunc <> lv_last_endfunc.
        IF lv_last_endfunc > 0 AND lv_first_endfunc > lv_last_endfunc.
          lv_first_endfunc = lv_first_endfunc - 1. " Adjust index if last was deleted
        ENDIF.
        DELETE lt_source INDEX lv_first_endfunc.
      ENDIF.
      IF lv_first_function > 0 AND lv_first_function <> lv_last_function.
        IF lv_last_function > 0 AND lv_first_function > lv_last_function.
          lv_first_function = lv_first_function - 1. " Adjust index if last was deleted
        ENDIF.
        DELETE lt_source INDEX lv_first_function.
      ENDIF.



      CALL FUNCTION 'RS_FUNCTIONMODULE_INSERT'
        EXPORTING
          funcname                = lv_function_name
          function_pool           = lv_fgname   " FUNCTION GROUP MUST ALREADY EXIST
          short_text              = lv_short_text2
        TABLES
          source                  = lt_source
          import_parameter        = lt_import_parameter
          export_parameter        = lt_export_parameter
          tables_parameter        = lt_tables_parameter
          changing_parameter      = lt_changing_parameter
        EXCEPTIONS
          double_task             = 1
          error_message           = 2
          function_already_exists = 3
          invalid_function_pool   = 4
          invalid_name            = 5
          too_many_functions      = 6
          no_modify_permission    = 7
          no_show_permission      = 8
          enqueue_system_failure  = 9
          canceled_in_corr        = 10
          OTHERS                  = 11.
      IF sy-subrc <> 0.
        APPEND VALUE #(       type       = 'E'
                        id         = sy-msgid
                        number     = sy-msgno
                        message_v1 = sy-msgv1
                        message_v2 = sy-msgv2
                        message_v3 = sy-msgv3
                        message_v4 = sy-msgv4
                        message = lv_fname && '<-->'&& 'FUNCTION PUT FAILURE' ) TO t_return.
      ENDIF.
    ENDIF.
  ENDIF.

  CLEAR :

  lt_uincl_tab ,
  lt_dincl_tab ,
  lt_enlfd_tab ,

*&---------------------------------------------------------------------*
*& FORM MAKE_TRAN
*&---------------------------------------------------------------------*
*& CREATE TRANSECTION CODE
*&---------------------------------------------------------------------*
FORM make_tran .
  DATA: lt_data       TYPE TABLE OF tab512,
        lt_tstcp      TYPE TABLE OF tab512,
        lt_fields     TYPE TABLE OF rfc_db_fld,
        ls_tstc       TYPE tstc,
        ls_tstcp      TYPE tstcp,
        lv_trobj_name TYPE trobj_name,
        lt_options    TYPE TABLE OF rfc_db_opt.

  " Step 1: Prepare options for RFC_READ_TABLE to filter single TCODE
  CLEAR lt_options.
  APPEND VALUE #( text = |TCODE = '{ lv_tran }'| ) TO lt_options.
  lv_trobj_name = lv_tran.

  " Step 2: Read TCODE from remote system
  TRY.
      CALL FUNCTION 'RFC_READ_TABLE'
        DESTINATION p_rd
        EXPORTING
          query_table         = 'TSTC'
        TABLES
          data                = lt_data
          options             = lt_options
          fields              = lt_fields
        EXCEPTIONS
          table_not_available = 1
          table_without_data  = 2
          option_not_valid    = 3
          not_authorized      = 4
          OTHERS              = 5.

      IF sy-subrc <> 0.
        APPEND VALUE #( type       = 'E'
                        id         = sy-msgid
                        number     = sy-msgno
                        message_v1 = sy-msgv1
                        message_v2 = sy-msgv2
                        message_v3 = sy-msgv3
                        message_v4 = sy-msgv4
                        message    = |{ lv_tran } - TCODE PUT FAILURE| ) TO t_return.
        RETURN.
      ENDIF.

      " Step 3: Check if TCODE already exists in local system
      SELECT SINGLE tcode
        INTO @DATA(lv_tcode)
        FROM tstc
        WHERE tcode = @lv_tran.

      IF sy-subrc = 0.
        APPEND VALUE #( type       = 'E'
                        id         = sy-msgid
                        number     = sy-msgno
                        message_v1 = sy-msgv1
                        message_v2 = sy-msgv2
                        message_v3 = sy-msgv3
                        message_v4 = sy-msgv4
                        message    = |{ lv_tran } - TCODE ALREADY EXISTS IN LOCAL SYSTEM| ) TO t_return.
        RETURN.
      ENDIF.

      " Step 4: Map fields from RFC data to TSTC structure
      DATA(lv_offset) = 0.
      READ TABLE lt_data INTO DATA(ls_rfc_data) INDEX 1.
      LOOP AT lt_fields INTO DATA(ls_field).
        CASE ls_field-fieldname.
          WHEN 'TCODE'.
            ls_tstc-tcode = ls_rfc_data+lv_offset(ls_field-length).
          WHEN 'PGMNA'.
            ls_tstc-pgmna = ls_rfc_data+lv_offset(ls_field-length).
          WHEN 'CINFO'.
            ls_tstc-cinfo = ls_rfc_data+lv_offset(ls_field-length).
          WHEN 'DYPNO'.
            ls_tstc-dypno = ls_rfc_data+lv_offset(ls_field-length).
          WHEN 'MENUE'.
            ls_tstc-menue = ls_rfc_data+lv_offset(ls_field-length).
          WHEN 'ARBGB'.
            ls_tstc-arbgb = ls_rfc_data+lv_offset(ls_field-length).
        ENDCASE.
        lv_offset += ls_field-length.
      ENDLOOP.

      " Step 5: Insert TCODE into local system
      IF ls_tstc IS NOT INITIAL.
        CALL FUNCTION 'ENQUEUE_E_TABLE'
          EXPORTING
            tabname        = 'TSTC'
          EXCEPTIONS
            foreign_lock   = 1
            system_failure = 2
            OTHERS         = 3.
        IF sy-subrc = 0.
          INSERT tstc FROM ls_tstc.
          COMMIT WORK.
          CALL FUNCTION 'DEQUEUE_E_TABLE'
            EXPORTING
              tabname = 'TSTC'.
        ENDIF.
      ENDIF.

      " Step 6: Read and process transaction parameters (TSTCP)
      CLEAR: lt_tstcp, lt_fields, lv_offset.
      CALL FUNCTION 'RFC_READ_TABLE'
        DESTINATION p_rd
        EXPORTING
          query_table = 'TSTCP'
        TABLES
          data        = lt_tstcp
          options     = lt_options
          fields      = lt_fields
        EXCEPTIONS
          OTHERS      = 5.

      IF lt_tstcp IS NOT INITIAL.
        " Check if TCODE parameters already exist in local system
        SELECT SINGLE *
          INTO @DATA(wa_tstcp)
          FROM tstcp
          WHERE tcode = @lv_tran.
        IF sy-subrc = 0.
          APPEND VALUE #( type       = 'E'
                          id         = sy-msgid
                          number     = sy-msgno
                          message_v1 = sy-msgv1
                          message_v2 = sy-msgv2
                          message_v3 = sy-msgv3
                          message_v4 = sy-msgv4
                          message    = |{ lv_tran } - TCODE ALREADY EXISTS IN LOCAL SYSTEM| ) TO t_return.
          RETURN.
        ENDIF.

        " Map fields from RFC data to TSTCP structure
        READ TABLE lt_tstcp INTO ls_rfc_data INDEX 1.
        LOOP AT lt_fields INTO ls_field.
          CASE ls_field-fieldname.
            WHEN 'TCODE'.
              ls_tstcp-tcode = ls_rfc_data+lv_offset(ls_field-length).
            WHEN 'PARAM'.
              ls_tstcp-param = ls_rfc_data+lv_offset(ls_field-length).
          ENDCASE.
          lv_offset += ls_field-length.
        ENDLOOP.

        " Insert TCODE parameters into local system
        IF ls_tstcp IS NOT INITIAL.
          CALL FUNCTION 'ENQUEUE_E_TABLE'
            EXPORTING
              tabname        = 'TSTCP'
            EXCEPTIONS
              foreign_lock   = 1
              system_failure = 2
              OTHERS         = 3.
          IF sy-subrc = 0.
            INSERT tstcp FROM ls_tstcp.
            COMMIT WORK.
            CALL FUNCTION 'DEQUEUE_E_TABLE'
              EXPORTING
                tabname = 'TSTCP'.
          ENDIF.
        ENDIF.
      ENDIF.

      " Step 7: Lock table into transport request
      gv_objnam = lv_trobj_name.
      gv_objtyp = 'TRAN'.
      PERFORM add_tr.

    CATCH cx_root INTO DATA(lx_error).
      APPEND VALUE #( type    = 'E'
                      message = |Error processing TCODE { lv_tran }: { lx_error->get_text( ) }| ) TO t_return.
  ENDTRY.

  " Step 8: Clear variables
  CLEAR: lt_data, lt_fields, ls_tstc, lv_trobj_name, lt_options, lt_tstcp, ls_tstcp.

ENDFORM.
